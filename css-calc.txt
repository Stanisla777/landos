// Внутри component-income-with-additional-coefficient.vue

methods: {
  initializeFieldsIncomeFromStore() {
    const fields = {};
    
    this.array_all_income_period.forEach(tab => {
      // 1. Основной доход
      fields[tab.modelInput] = tab.value || '';
      
      // 2. Районный коэффициент (если он есть в структуре)
      if (tab.additionalCoefficient) {
        const baseCoeffKey = tab.additionalCoefficient.modelInput;
        fields[baseCoeffKey + '_district'] = tab.additionalCoefficient.districtCoefficient.value || '';
        fields[baseCoeffKey + '_northern'] = tab.additionalCoefficient.northernAllowance.value || '';
      }
    });
    
    this.fieldsIncomeMan = fields;
  },
}

mutations: {
  // ... ваши другие мутации ...

  // 1. МУТАЦИЯ ОБНОВЛЕНИЯ ПОЛЕЙ (теперь умеет работать с вложенностью)
  mutationInputIncomeMan(state, received_perem) {
    // received_perem выглядит так: { field_1: '50000', field_coefficient_1_district: '1.5', ... }
    
    Object.keys(received_perem).forEach(key => {
      const newValue = received_perem[key];

      // Ищем нужный год в массиве
      state.array_all_income_period_man.forEach(item => {
        // Вариант А: это основной доход
        if (item.modelInput === key) {
          item.value = newValue;
        }
        // Вариант Б: это районный коэффициент
        else if (item.additionalCoefficient && (item.additionalCoefficient.modelInput + '_district') === key) {
          item.additionalCoefficient.districtCoefficient.value = newValue;
        }
        // Вариант В: это северная надбавка
        else if (item.additionalCoefficient && (item.additionalCoefficient.modelInput + '_northern') === key) {
          item.additionalCoefficient.northernAllowance.value = newValue;
        }
      });
    });
  },

  // 2. МУТАЦИЯ ПЕРЕКЛЮЧЕНИЯ ТАБОВ (теперь очищает ВСЕ связанные поля)
  mutationTabTimeMan(state, received_perem) {
    const targetId = String(received_perem);
    const activeIndex = state.array_all_income_period_man.findIndex(
      item => String(item.id) === targetId
    );

    state.array_all_income_period_man.forEach((item, index) => {
      if (index === activeIndex) {
        item.isActive = true;
      } else {
        item.isActive = false;

        // ГЛАВНОЕ: если год идет ПОСЛЕ выбранного, очищаем ВСЮ сущность целиком
        if (index > activeIndex) {
          item.value = '';
          
          if (item.additionalCoefficient) {
            item.additionalCoefficient.districtCoefficient.value = '';
            item.additionalCoefficient.northernAllowance.value = '';
          }
        }
      }
    });
  },
}
