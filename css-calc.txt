               v-for="item2 in filteredVariants" 
               :key="item2.id"
               @click="selectItem(item2.label)">
            <p class="ds-multiselect__checkbox-label">{{ item2.label }}</p>
          </div>

          <div v-if="filteredVariants.length === 0 && isUserInteracted" class="no-results">
            <p>Нет совпадений</p>
          </div>
        </div>
      </div>
    </div>

<script>
export default {
  name: 'component-district-coefficient',
  props: {
    item: { type: Object, required: true },
    //  Стандартное имя пропа для v-model во Vue 3
    modelValue: { type: [String, Number, Array], default: '' }
  },
  data() {
    return {
      dataField: '',
      dropDown: false,
      isUserInteracted: false // Флаг для показа полного списка при инициализации
    }
  },
  computed: {
    filteredVariants() {
      const variants = this.item?.additionalCoefficient?.districtCoefficient?.variant || [];
      const searchQuery = String(this.dataField).trim();

      // Если пользователь еще не печатал или поле пустое, показываем всё
      if (!this.isUserInteracted || !searchQuery || searchQuery === '.') {
        return variants;
      }

      // Фильтрация по началу строки
      return variants.filter(item2 => String(item2.label).startsWith(searchQuery));
    }
  },
  watch: {
    //  Если значение изменилось извне (например, Store очистил поле при смене таба), обновляем локальное поле
    modelValue: {
      handler(newVal) {
        // Нормализуем: если пришел массив (от старого ds-multi-select), берем label, иначе строку
        if (Array.isArray(newVal) && newVal.length > 0) {
          this.dataField = String(newVal[0].label || newVal[0].value || '');
        } else {
          this.dataField = String(newVal || '');
        }
        // Сбрасываем флаг взаимодействия, чтобы при следующем открытии список был полным
        this.isUserInteracted = false; 
      },
      immediate: true
    }
  },
  methods: {
    logEvent() {
      this.isUserInteracted = true; // Пользователь начал печатать
      
      let val = this.dataField.replace(/[^\d.,]/g, '').replace(',', '.');
      const parts = val.split('.');
      
      if (parts.length === 2 && parts[1].length > 2) {
        val = parts[0] + '.' + parts[1].substring(0, 2);
      }

      const num = parseFloat(val);
      if (!isNaN(num)) {
        if (num < 1) val = '1';
        else if (num >= 3) val = '3'; // Или другое ограничение, если нужно
      }

      this.dataField = val;
      //  Отправляем обновленное значение родителю
      this.$emit('update:modelValue', this.dataField);
    },
    selectItem(label) {
      this.dataField = String(label);
      this.dropDown = false;
      this.isUserInteracted = true;
      //  Отправляем выбранное значение родителю
      this.$emit('update:modelValue', this.dataField);
    },
    clickdropDown() {
      this.dropDown = !this.dropDown;
    }
  }
}
</script>

<script>
import store from "../store";
import { DsInput } from '@ringsites/ui-components/DsInput';
import ComponentDistrictCoefficient from './component-district-coefficient.vue';
import { debounce } from "../../../modules/debounce";

export default {
  name: 'component-income-with-additional-coefficient',
  props: ['view'],
  data() {
    return {
      fieldsIncome: {},
      isInitialized: false,
      skipNextWatch: true,
    }
  },
  computed: {
    data_year() { return store.getters.DATAYEAR; },
    array_all_income_period() {
      return this.view === 'man' ? store.getters.ARRAYALLINCOMEPERIODMAN : store.getters.ARRAYALLINCOMEPERIODWIFE;
    },
    visibleBlockIncomePeriod() {
      const activeIndex = this.array_all_income_period.findIndex(tab => tab.isActive === true);
      if (activeIndex === -1) return [];
      return this.array_all_income_period.slice(0, activeIndex + 1);
    },
  },
  created() {
    this.debouncedIncomeUpdate = debounce((newFields) => {
      if (this.skipNextWatch || !this.isInitialized) return;

      const cleanedFields = {};
      for (const key in newFields) {
        const val = newFields[key];
        // Теперь val для коэффициентов будет просто строкой (например, "1.5"), 
        // а не массивом. Умная очистка это корректно обработает.
        if (Array.isArray(val) || (typeof val === 'object' && val !== null)) {
          cleanedFields[key] = val;
        } else {
          const cleanStr = String(val).replace(/\s/g, '');
          cleanedFields[key] = cleanStr === '' ? '' : (isNaN(parseFloat(cleanStr)) ? cleanStr : parseFloat(cleanStr));
        }
      }

      store.dispatch('ActionShowLoader', true);
      const actionName = this.view === 'man' ? 'ActionInputIncomeMan' : 'ActionInputIncomeWife';
      
      store.dispatch(actionName, cleanedFields).finally(() => {
        store.dispatch('ActionShowLoader', false);
      });
    }, 500);
  },
  methods: {
    // ⭐ Вынесли логику инициализации в отдельный метод, чтобы не дублировать код
    syncFieldsFromStore() {
      const fields = {};
      this.array_all_income_period.forEach(tab => {
        fields[tab.modelInput] = tab.value || '';
        
        if (tab.additionalCoefficient) {
          const baseCoeffKey = tab.additionalCoefficient.modelInput;
          
          // ⭐ НОРМАЛИЗАЦИЯ: Преобразуем массив (от старого селекта) в строку для нового инпута
          const distVal = tab.additionalCoefficient.districtCoefficient.value;
          fields[baseCoeffKey + '_district'] = Array.isArray(distVal) ? (distVal[0]?.label || '') : (distVal || '');

          const northVal = tab.additionalCoefficient.northernAllowance.value;
          fields[baseCoeffKey + '_northern'] = Array.isArray(northVal) ? (northVal[0]?.label || '') : (northVal || '');
        }
      });
      this.fieldsIncome = fields;
    }
  },
  mounted() {
    this.skipNextWatch = true;
    this.$nextTick(() => {
      this.syncFieldsFromStore();
      this.$nextTick(() => {
        this.skipNextWatch = false;
        this.isInitialized = true;
      });
    });
  },
  watch: {
    array_all_income_period: {
      handler() {
        this.skipNextWatch = true;
        this.$nextTick(() => {
          this.syncFieldsFromStore();
          this.$nextTick(() => {
            this.skipNextWatch = false;
            this.isInitialized = true;
          });
        });
      },
      deep: true
    },
    fieldsIncome: {
      handler(newValue) {
        if (this.skipNextWatch || !this.isInitialized) return;
        this.debouncedIncomeUpdate(newValue);
      },
      deep: true,
      immediate: false
    },
  },
  beforeUnmount() {
    if (this.debouncedIncomeUpdate) {
      this.debouncedIncomeUpdate.cancel();
    }
  },
  components: {
    DsInput,
    ComponentDistrictCoefficient
  },
};
</script>
