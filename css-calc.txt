<template>
  <div class="test-a__container marathon-new-last__test-vue-slider" ref="test_container">
    <div class="test-a__body" ref="BodyLabel">
      <div class="test-n-slider__line"></div>
      <div class="test-n-slider__line right js--slider-line-right"></div>
      
      <div class="test-n-slider test-a__slider" ref="ContainerSlider">
        <!-- 
          dragValue: хранит плавное дробное значение во время перетаскивания (step="0.01")
          @end: срабатывает и при отпускании мыши, и при тапе. Здесь мы "примагничиваем" значение.
        -->
        <Slider
          ref="slider"
          v-model="dragValue"
          :min="0"
          :max="optionsList.length - 1"
          :step="0.01"
          :tooltip="false"
          :height="4"
          :dot-size="24"
          :disabled="time_end"
          @end="onSliderRelease"
          class="custom-vue-slider"
        />

        <!-- Кастомные маркеры, позиционируемые равномерно -->
        <div class="test-n-slider__marks" :class="{ 'is-disabled': time_end }">
          <div
            v-for="(option, index) in optionsList"
            :key="option.id"
            class="test-n-slider__mark"
            :class="{ 'is-active': index === selectedIndex }"
            :style="{ left: getEvenPosition(index) + '%' }"
            @click.stop="selectByIndex(index)"
          >
            <span class="test-n-slider__mark-label">{{ option.name }}</span>
          </div>
        </div>
      </div>
    </div>

    <test-button
      :modal="modal"
      :status_internet="status_internet"
      :btn_next="btn_next"
      :status_btn="status_btn"
      :btn_prev="btn_prev"
      :error_sand_variant="error_sand_variant"
      :error_sand_variant_text="error_sand_variant_text"
    />
  </div>
</template>

<script>
import Slider from '@vueform/slider';
import store from '../store';
import eventBus from '../eventBus';
import TestButton from './TestButton.vue';

export default {
  name: 'TextNSlider',
  components: { 
    Slider,
    TestButton 
  },
  props: {
    param_component: {
      type: Object,
      required: true
    },
    modal: {
      type: Boolean,
      default: false
    }
  },
  data() {
    return {
      selectedIndex: 0,   // Логический индекс (0, 1, 2...) для бизнес-логики
      dragValue: 0,       // Визуальное плавное значение для слайдера (0.00, 0.34, 1.00...)
      answerValue: null,  // Реальный ID из options, который уходит в Store
      btn_next: true,
      btn_prev: true,
      count: 0,
      count_slide: 0,
    };
  },
  computed: {
    optionsList() {
      return this.param_component?.options || [];
    },
    error_sand_variant() {
      return store.getters.ERRORSANDVARIANT;
    },
    error_sand_variant_text() {
      return store.getters.ERRORSANDVARIANTTEXT;
    },
    status_internet() {
      return store.getters.STATUSINTERNET;
    },
    status_btn() {
      return store.getters.STATUSBTN;
    },
    time_end() {
      return store.getters.TIMEEND;
    },
    current_slide() {
      return store.getters.CURRENTSLIDE;
    },
  },
  watch: {
    current_slide() {
      const line = document.querySelectorAll('.js--slider-line-right');
      line.forEach(item => item.classList.remove('active'));
      
      if (this.optionsList.length > 0) {
        this.resetToFirstOption();
      }
    }
  },
  methods: {
    // Равномерное распределение маркеров
    getEvenPosition(index) {
      if (this.optionsList.length <= 1) return 0;
      return (index / (this.optionsList.length - 1)) * 100;
    },

    // ГЛАВНАЯ МАГИЯ: Срабатывает при отпускании (drag end) или тапе (click)
    onSliderRelease() {
      if (this.time_end || this.optionsList.length === 0) return;

      // 1. Округляем плавное значение до ближайшего целого индекса
      const nearestIndex = Math.round(this.dragValue);

      // 2. Принудительно устанавливаем оба значения. 
      // Это мгновенно "примагничивает" ползунок к центру маркера БЕЗ визуального отскока,
      // так как взаимодействие (drag/click) уже завершено.
      this.selectedIndex = nearestIndex;
      this.dragValue = nearestIndex;

      // 3. Обновляем состояние и отправляем в Store
      this.updateStateFromIndex(nearestIndex);
    },

    // Клик непосредственно по тексту маркера (обходной путь, если тап по треку не сработал)
    selectByIndex(index) {
      if (this.time_end) return;
      this.selectedIndex = index;
      this.dragValue = index; // Синхронизируем для плавности
      this.updateStateFromIndex(index);
    },

    updateStateFromIndex(index) {
      const selectedOption = this.optionsList[index];
      if (!selectedOption) return;

      this.answerValue = selectedOption.id;

      // Логика для правой линии (как в оригинальном коде)
      const line = document.querySelectorAll('.js--slider-line-right');
      if (this.optionsList.length - 1 === index) {
        line.forEach(item => {
          setTimeout(() => item.classList.add('active'), 200);
        });
      } else {
        line.forEach(item => item.classList.remove('active'));
      }

      // Отправка в Store (ваша исходная логика)
      store.dispatch('ActionVariantSlider', [this.param_component.id, this.answerValue]);
    },

    resetToFirstOption() {
      this.selectedIndex = 0;
      this.dragValue = 0;
      this.answerValue = this.optionsList[0].id;
      store.dispatch('ActionVariantSlider', [this.param_component.id, this.answerValue]);
    },

    autoHeight() {
      if (document.documentElement.clientWidth < 470) {
        const container = this.$refs.test_container;
        if (!container) return;
        const parent = container.closest('.js--test-cont');
        if (!parent) return;
        
        const title = parent.querySelector('.test-a__question');
        const title_height = title ? title.offsetHeight : 0;
        
        const sliderContainer = parent.querySelector('.test-a__container');
        if (sliderContainer) {
          sliderContainer.setAttribute('style', `min-height:calc(100vh - 32px - 26px - 24px - ${title_height}px - 65px)`);
        }
      }
    },

    updateScreenWidth() {
      this.autoHeight();
    },
    
    btnPrev() {
      this.count = 0;
      eventBus.emit('eventbtnPrev');
    },
  },
  mounted() {
    if (this.optionsList.length > 0) {
      this.resetToFirstOption();
    }
    
    this.updateScreenWidth();
    window.addEventListener('resize', this.updateScreenWidth);
  },
  created() {
    eventBus.on('slideReachBeginning', () => {
      this.btn_prev = false;
    });
    eventBus.on('slideFromEdge', () => {
      this.btn_prev = true;
    });
  },
  unmounted() {
    window.removeEventListener('resize', this.updateScreenWidth);
    eventBus.off('slideReachBeginning');
    eventBus.off('slideFromEdge');
  }
};
</script>
