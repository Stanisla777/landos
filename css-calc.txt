<script>
import store from "../store";
import { DsInput } from '@ringsites/ui-components/DsInput';
import { DsMultiSelect } from '@ringsites/ui-components';
import { debounce } from "../../../modules/debounce"; // Укажите правильный путь к вашему debounce

export default {
  name: 'component-income-with-additional-coefficient',
  props: {
    view: {
      type: String,
      default: 'man' // 'man' или 'wife'
    }
  },
  
  data() {
    return {
      fieldsIncome: {},
      isUpdatingFromStore: false, // Флаг-предохранитель
    }
  },

  created() {
    // 1. Инициализируем данные
    this.initializeFieldsIncomeFromStore();
    
    // 2. Создаем debounce-функцию ОДИН раз
    this.debouncedIncomeUpdate = debounce((newFields) => {
      // Последний рубеж обороны
      if (this.isUpdatingFromStore) return;

      const cleanedFields = {};
      for (const key in newFields) {
        let val = newFields[key];
        
        // Защита: если select вдруг вернет объект вместо строки/числа
        if (typeof val === 'object' && val !== null) {
          val = val.value || val.label || '';
        }
        
        // ⭐ ВАЖНО: Используем parseFloat вместо parseInt, 
        // чтобы коэффициенты типа "1.5" не превращались в "1"
        const cleanStr = String(val).replace(/\s/g, '');
        cleanedFields[key] = cleanStr === '' ? '' : (isNaN(parseFloat(cleanStr)) ? cleanStr : parseFloat(cleanStr));
      }

      // Динамически выбираем Action в зависимости от пропа view
      const actionName = this.view === 'man' ? 'ActionInputIncomeMan' : 'ActionInputIncomeWife';

      store.dispatch('ActionShowLoader', true);
      store.dispatch(actionName, cleanedFields)
        .finally(() => {
          store.dispatch('ActionShowLoader', false);
        });
    }, 500);
  },

  watch: {
    // === СЛУШАТЕЛЬ 1: Реагируем на изменения в Store (клик по табу) ===
    array_all_income_period: {
      handler() {
        this.isUpdatingFromStore = true;

        // ⭐ ОТМЕНЯЕМ таймер, если пользователь в этот момент что-то выбирал/печатал
        if (this.debouncedIncomeUpdate) {
          this.debouncedIncomeUpdate.cancel();
        }

        this.initializeFieldsIncomeFromStore();

        this.$nextTick(() => {
          this.isUpdatingFromStore = false;
        });
      },
      deep: true
    },

    // === СЛУШАТЕЛЬ 2: Реагируем на ввод/выбор пользователя ===
    fieldsIncome: {
      handler(newFields) {
        // ⭐ ГЛАВНОЕ ИСПРАВЛЕНИЕ: Синхронная проверка.
        // Прерываем выполнение НЕМЕДЛЕННО, не давая запуститься новому таймеру.
        if (this.isUpdatingFromStore) return;

        this.debouncedIncomeUpdate(newFields);
      },
      deep: true,
      immediate: false
    },
  },

  beforeUnmount() {
    // Очищаем память при уничтожении компонента
    if (this.debouncedIncomeUpdate) {
      this.debouncedIncomeUpdate.cancel();
    }
  },

  methods: {
    initializeFieldsIncomeFromStore() {
      const fields = {};

      this.array_all_income_period.forEach(tab => {
        // 1. Основной доход
        fields[tab.modelInput] = tab.value || '';

        // 2. Районный коэффициент и Северная надбавка
        if (tab.additionalCoefficient) {
          const baseCoeffKey = tab.additionalCoefficient.modelInput;
          fields[baseCoeffKey + '_district'] = tab.additionalCoefficient.districtCoefficient.value || '';
          fields[baseCoeffKey + '_northern'] = tab.additionalCoefficient.northernAllowance.value || '';
        }
      });

      this.fieldsIncome = fields;
    },
  },

  computed: {
    data_year() {
      return store.getters.DATAYEAR;
    },

    array_all_income_period() {
      if (this.view === 'man') {
        return store.getters.ARRAYALLINCOMEPERIODMAN;
      } else {
        return store.getters.ARRAYALLINCOMEPERIODWIFE;
      }
    },

    visibleBlockIncomePeriod() {
      const activeIndex = this.array_all_income_period.findIndex(tab => tab.isActive === true);
      if (activeIndex === -1) return [];
      return this.array_all_income_period.slice(0, activeIndex + 1);
    },
  },

  components: {
    DsInput,
    DsMultiSelect
  },
};
</script>
