 Компонент первоначального взноса (component-initial-payment)

inputValue(el) {
      const element = el.currentTarget;
      // Безопасно преобразуем значение в число, убирая пробелы
      let value = parseInt(element.value.replace(/\s/g, '')) || 0;

      // ГАРАНТИРУЕМ синхронизацию значения для расчётов
      this.dataFieldForCalculation = value;
      
      if (value < this.stgMin) {
        this.realtySlider.set(this.stgMin);
      } else {
        this.realtySlider.set(value);
      }
    },

    calculationPageLoad() {
      // БЕРЁМ АКТУАЛЬНОЕ значение из поля ввода, удаляя все пробелы
      const cleanValue = parseInt(String(this.dataField).replace(/\s/g, '')) || 0;
      
      // Синхронизируем внутреннюю переменную для надёжности
      this.dataFieldForCalculation = cleanValue;
      
      store.dispatch('ActionInitialFee', cleanValue);
    },

    inputValue(el) {

      // Синхронизируем значение

      } else {
        this.realtySlider.set(value);
      }
    },
