mounted() {
  this.$nextTick(() => {
    setTimeout(() => this.captchaInit(), 1500);
  });
},

methods: {
  captchaInit() {
    if (this.captcha_id !== null || !window.smartCaptcha) return;
    this.captcha_id = window.smartCaptcha.render('captcha-container-id', {
      sitekey: conf.smartcaptcha_key,
      invisible: true,
      callback: (token) => this.onCaptchaSuccess(token),
    });
  },

  handlerSendForm() {
    // ... валидация ...
    if (this.captcha_id === null) {
      this.$nextTick(() => this.captchaInit());
      return;
    }
    window.smartCaptcha.execute(this.captcha_id, this.send_argument);
  },
}
