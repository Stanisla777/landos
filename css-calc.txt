// УДАЛЯЕМ старый метод captchaInit() — он больше не нужен

handlerSendForm(argument) {
  // 1. Валидация полей
  this.validateAllInputs();
  if (!this.isValid) return;

  // 2. Проверка адреса
  if (!this.isAddressValid) {
    this.inputs[0].isNotValid = true;
    this.ocenkaMobiErrorText = 'Введите корректный адрес (город, улица, дом)';
    this.showError();
    return;
  }

  // 3. Блокируем кнопку и сохраняем аргумент (на всякий случай, если он нужен в шаблоне)
  this.send_argument = argument;
  this.btnSendPress = true;

  // 4. Запускаем капчу через миксин.
  // Миксин сам проверит, загрузился ли скрипт, и если нет — отправит запрос с token = null (Fallback)
  this.runCaptcha((token) => {
    // Благодаря замыканию переменная `argument` доступна здесь
    if (!argument) {
      this.apiConnect(token);
      this.handlerBuildInfo(false);
    } else {
      this.handlerBuildInfo(true, token);
    }
  });
},
