computed: {
  normalizedHeaders() { /* ваш код */ },
  normalizedRows() { /* ваш код */ },
  sortedRows() { /* ваш код */ },
  
  // Определяем тип каждой колонки на основе содержимого
  columnTypes() {
    const types = [];
    const rows = this.normalizedRows;
    
    if (rows.length === 0) return [];
    
    const colCount = this.normalizedHeaders.length;
    
    for (let colIndex = 0; colIndex < colCount; colIndex++) {
      let numberCount = 0;
      let dateCount = 0;
      let totalCount = 0;
      
      rows.forEach(row => {
        const cell = row.cells[colIndex];
        if (cell && cell.label) {
          const extracted = this.extractValue(cell.label);
          totalCount++;
          if (extracted.type === 'number') numberCount++;
          else if (extracted.type === 'date') dateCount++;
        }
      });
      
      // Колонка числовая, если больше половины значений — числа
      types.push({
        isNumeric: totalCount > 0 && numberCount > totalCount / 2,
        isDate: totalCount > 0 && dateCount > totalCount / 2
      });
    }
    
    return types;
  }
}
