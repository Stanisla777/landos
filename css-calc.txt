export default {
  data() {
    return {
      captchaWidgetId: null,
      captchaContainer: null,
      isSending: false,
    };
  },

  beforeUnmount() {
    this.destroyCaptcha();
  },

  methods: {
    handleSend() {
      if (this.isSending || !this.btn_active) return;
      this.isSending = true;

      if (!this.captchaWidgetId) {
        this.initCaptcha();
        setTimeout(() => this.executeCaptcha(), 100);
      } else {
        this.executeCaptcha();
      }
    },

    initCaptcha() {
      if (!window.smartCaptcha) {
        console.error('Скрипт капчи не загружен');
        this.isSending = false;
        return;
      }

      // 🔥 ПРОВЕРКА №1: Убедитесь, что ключ реально существует и это строка
      const sitekey = conf?.smartcaptcha_key;
      if (!sitekey || typeof sitekey !== 'string' || sitekey.trim() === '') {
        console.error('КРИТИЧЕСКАЯ ОШИБКА: smartcaptcha_key отсутствует или пуст! Это вызывает краш React Яндекса.');
        this.isSending = false;
        return;
      }

      const el = document.createElement('div');
      el.id = `yandex-captcha-ext-${Date.now()}`;
      // Убираем display: none, иногда Яндекс крашится при рендере в полностью скрытый элемент
      // Вместо этого сделаем его визуально незаметным, но присутствующим в потоке
      el.style.position = 'absolute';
      el.style.left = '-9999px';
      el.style.opacity = '0';
      
      document.body.appendChild(el);
      this.captchaContainer = el;

      try {
        this.captchaWidgetId = window.smartCaptcha.render(el.id, {
          sitekey: sitekey,
          invisible: true,
          callback: (token) => {
            // 🔥 ПРОВЕРКА №2: Даем React Яндекса завершить внутренний цикл (commit phase)
            // перед тем, как мы начнем отправку и возможное удаление DOM.
            setTimeout(() => {
              this.submitToApi(token);
            }, 0); // 0 мс достаточно, чтобы передать управление в конец очереди событий
          },
        });
      } catch (e) {
        console.error('Ошибка render:', e);
        this.destroyCaptcha();
        this.isSending = false;
      }
    },

    executeCaptcha() {
      if (this.captchaWidgetId !== null && window.smartCaptcha) {
        window.smartCaptcha.execute(this.captchaWidgetId);
      } else {
        this.isSending = false;
      }
    },

    submitToApi(token) {
      this.$refs.selectListWrapper?.classList.remove('open');
      this.closeBlock();
      
      store.dispatch('ActionGetResult', [this.oktmo, token])
        .then(() => console.log('Успешно'))
        .catch((err) => console.error('Ошибка:', err))
        .finally(() => {
          // Уничтожаем только ПОСЛЕ того, как запрос завершен
          this.destroyCaptcha();
          this.isSending = false;
        });
    },

    destroyCaptcha() {
      if (this.captchaWidgetId !== null && window.smartCaptcha) {
        try {
          window.smartCaptcha.reset(this.captchaWidgetId);
        } catch (e) {}
      }
      
      if (this.captchaContainer && this.captchaContainer.parentNode) {
        this.captchaContainer.parentNode.removeChild(this.captchaContainer);
      }

      this.captchaWidgetId = null;
      this.captchaContainer = null;
    }
  }
}
