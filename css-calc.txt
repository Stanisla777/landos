import store from "../store";
import { debounce } from "../../../modules/debounce";

// Добавляем второй параметр: имя computed-свойства в компоненте, откуда брать значение
export function sandToStoreAmountField(actionName, computedPropName) {
  return {
    data() {
      return {
        dataField: '',
        isInitialized: false,
        isUpdatingFromStore: false,
      };
    },

    created() {
      // В created мы ТОЛЬКО готовим debounce.
      // Никаких изменений данных здесь не делаем, чтобы избежать ошибки emitsOptions.
      this.debouncedIncomeUpdate = debounce((newValue) => {
        if (this.isUpdatingFromStore || !this.isInitialized) return;

        const cleanedFields = String(newValue).replace(/\s/g, '');
        store.dispatch('ActionShowLoader', true);

        const valueToSave = cleanedFields === '' ? '' : parseInt(cleanedFields, 10);

        store.dispatch(actionName, valueToSave)
          .finally(() => {
            store.dispatch('ActionShowLoader', false);
          });
      }, 500);
    },

    // ⭐ ВСЯ ИНИЦИАЛИЗАЦИЯ ТЕПЕРЬ ЗДЕСЬ (в миксине)
    mounted() {
      // Блокирую отправку данных обратно
      this.isUpdatingFromStore = true;

      this.$nextTick(() => {
        const val = this[computedPropName];
        if (val === 0 || val === '' || val === null || val === undefined) {
          this.dataField = '';
        } else {
          this.dataField = val;
        }
        this.isUpdatingFromStore = false;
        this.isInitialized = true;
      });
    },

    watch: {
      dataField: {
        handler(newValue) {
          if (this.isUpdatingFromStore || !this.isInitialized) return;
          this.debouncedIncomeUpdate(newValue);
        },
        immediate: false // Обязательно false
      }
    },

    beforeUnmount() {
      if (this.debouncedIncomeUpdate) {
        this.debouncedIncomeUpdate.cancel();
      }
    }
  };
}
