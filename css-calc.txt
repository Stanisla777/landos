      // Если такой элемент не найден — возвращаю пустой массив

  created() {

    this.debouncedIncomeUpdate = debounce((newFields) => {
      if (this.skipNextWatch || !this.isInitialized) return;

      const cleanedFields = {};
      for (const key in newFields) {
        const val = newFields[key];

        // Умная очистка: не ломаем массивы/объекты от select
        if (Array.isArray(val) || (typeof val === 'object' && val !== null)) {
          cleanedFields[key] = val;
        } else {
          const cleanStr = String(val).replace(/\s/g, '');
          if (cleanStr === '') {
            cleanedFields[key] = '';
          } else {
            const num = parseFloat(cleanStr);
            cleanedFields[key] = isNaN(num) ? cleanStr : num;
          }
        }
      }

      store.dispatch('ActionShowLoader', true);
      // Если флаг опущен, значит это реальный ввод пользователя. Отправляем в Store.
      if (this.view === 'man') {
        store.dispatch('ActionInputIncomeMan', cleanedFields);
      } else {
        store.dispatch('ActionInputIncomeWife', cleanedFields)
          .finally(() => {
            store.dispatch('ActionShowLoader', false);
          });
      }

    }, 500);
