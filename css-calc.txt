normalizedRows() {
  // Если нормализация выключена
  if (!this.normalizeColumn) {
    return (this.tableData.rows || []).map((row, rowIndex) => {
      // Поддерживается оба формата: массив и объект с cells
      if (Array.isArray(row)) {
        return {
          id: `row-${rowIndex}`,
          cells: row.map((cell, cellIndex) => ({
            ...(typeof cell === 'object' && cell !== null ? cell : { label: cell }),
            id: (typeof cell === 'object' && cell !== null ? cell.id : undefined) ?? `cell-${rowIndex}-${cellIndex}`
          }))
        };
      }
      return {
        id: row.id ?? `row-${rowIndex}`,
        ...row,
        cells: (row.cells || []).map((cell, cellIndex) => ({
          ...cell,
          id: cell.id ?? `cell-${rowIndex}-${cellIndex}`
        }))
      };
    });
  }

  const headerCount = this.normalizedHeaders.length;

  // ВАЖНО: убрали ранний возврат "if (headerCount === 0) return this.tableData.rows || [];"
  // Всегда нормализуем, даже если заголовков нет

  return (this.tableData.rows || []).map((row, rowIndex) => {
    // Поддерживаем формат массива: row может быть [{label: '...'}] или {cells: [...]}
    const cells = Array.isArray(row) ? row : (row.cells || []);

    // Если headerCount > 0, нормализуем по количеству заголовков
    let normalizedCells = headerCount > 0
      ? [...cells].slice(0, headerCount)
      : [...cells];

    if (process.env.NODE_ENV === 'development' && headerCount > 0 && cells.length !== headerCount) {
      console.warn(
        `[TableStory] Строка ${rowIndex}: ожидается ${headerCount} ячеек, получено ${cells.length}`
      );
    }

    // Добавляем пустые ячейки только если есть заголовки
    if (headerCount > 0) {
      while (normalizedCells.length < headerCount) {
        normalizedCells.push({
          id: `empty-${rowIndex}-${normalizedCells.length}`,
          label: '',
          link: '',
          icon: ''
        });
      }
    }

    // Нормализуем каждую ячейку (поддержка строк как ячеек)
    normalizedCells = normalizedCells.map((cell, cellIndex) => {
      if (typeof cell === 'string') {
        return {
          id: `cell-${rowIndex}-${cellIndex}`,
          label: cell
        };
      }
      return {
        ...cell,
        id: cell?.id ?? `cell-${rowIndex}-${cellIndex}`
      };
    });

    return {
      id: Array.isArray(row) ? `row-${rowIndex}` : (row.id ?? `row-${rowIndex}`),
      ...(Array.isArray(row) ? {} : row),
      cells: normalizedCells
    };
  });
}
