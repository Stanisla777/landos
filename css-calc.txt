/* === BESENDER CONFIG WAITER === */
(function () {
    var MAX_WAIT = 5000;
    var INTERVAL = 100;
    var waited = 0;

    function isConfigReady() {
        return typeof window.beesenderchatconfig !== 'undefined'
            && window.beesenderchatconfig !== null
            && window.beesenderchatconfig.srcUrl
            && window.beesenderchatconfig.connectorUrl;
    }

    function startWidget() {
        /* === ОРИГИНАЛЬНЫЙ КОД BESENDER === */
        ;(function (e) {
            // ... весь оригинальный код ...
        })({ /* ... */ });
        /* === КОНЕЦ ОРИГИНАЛЬНОГО КОДА === */
    }

    function waitForConfig() {
        if (isConfigReady()) {
            startWidget();
        } else if (waited < MAX_WAIT) {
            waited += INTERVAL;
            setTimeout(waitForConfig, INTERVAL);
        } else {
            // Конфиг не появился за 10 секунд — НЕ запускаем виджет
            console.error('[beesender] Config not found after ' + MAX_WAIT + 'ms. Widget NOT started.', {
                page: window.location.href,
                configExists: typeof window.beesenderchatconfig !== 'undefined',
                srcUrl: window.beesenderchatconfig ? window.beesenderchatconfig.srcUrl : '(config не определён)',
                connectorUrl: window.beesenderchatconfig ? window.beesenderchatconfig.connectorUrl : '(config не определён)',
                time: new Date().toISOString()
            });
            // Можно дополнительно отправить на сервер для мониторинга:
            // if (window.beesenderOnError) window.beesenderOnError({...});
        }
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', waitForConfig);
    } else {
        waitForConfig();
    }
})();
/* === END CONFIG WAITER === */
