UPDATE_INCOME_VALUE(state, { modelInput, newValue }) {
      const targetItem = state.block_income_period.find(item => item.modelInput === modelInput);
      if (targetItem) {
        targetItem.value = newValue;
      }
    }

initializeFieldsFromStore() {
      const fields = {}
      // Проходим по ВСЕМ табам (не только видимым, чтобы сохранить данные всех полей)
      this.allTabs.forEach(tab => {
        fields[tab.modelInput] = tab.value || ''
      })
      this.fieldsIncomeMan = fields
    },
