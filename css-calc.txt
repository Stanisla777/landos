methods: {
  initSwiper() {
    this.destroySwiper();

    if (!this.$refs.swiperContainer || this.flatSlides.length === 0) {
      return;
    }

    this.swiper = new Swiper(this.$refs.swiperContainer, {
      slidesPerView: 'auto',
      spaceBetween: 16,
      watchOverflow: true,
      observer: true,
      observeParents: true,
      grabCursor: true,
      keyboard: {
        enabled: true
      },
      on: {
        slideChange: (swiper) => {
          this.activeIndex = swiper.activeIndex;
          this.updateControls(swiper);
        },
        resize: (swiper) => {
          swiper.update();
          this.updateControls(swiper);
        },
        imagesReady: (swiper) => {
          swiper.update();
          this.updateControls(swiper);
        }
      }
    });

    this.activeIndex = this.swiper.activeIndex || 0;
    this.updateControls(this.swiper);
  },

  destroySwiper() {
    if (this.swiper && typeof this.swiper.destroy === 'function') {
      this.swiper.destroy(true, true);
    }
    this.swiper = null;
  },

  updateControls(swiper) {
    if (!swiper) return;
    
    this.isBeginning = Boolean(swiper.isBeginning);
    
    // Последний реальный слайд (не считая пустой слайд-заполнитель)
    const lastRealSlideIndex = this.flatSlides.length - 1;
    
    // Если активный слайд >= последнего реального, то isEnd = true
    this.isEnd = swiper.activeIndex >= lastRealSlideIndex;
  },

  nextSlide() {
    if (!this.swiper) return;
    
    // Не листаем, если уже на последнем реальном слайде
    const lastRealSlideIndex = this.flatSlides.length - 1;
    if (this.swiper.activeIndex < lastRealSlideIndex) {
      this.swiper.slideNext();
    }
  },

  prevSlide() {
    if (!this.swiper) return;
    this.swiper.slidePrev();
  }
}
