          <tr>
            <th
              v-for="(header, columnIndex) in normalizedHeaders"
              :key="header.id ?? columnIndex"
              scope="col"
            >
              <div
                :class="getColumnClass(columnIndex, true)"
                class="ds-table__table-th">

--------------------
const getColumnClass = (index, isHeader) => {
  const customAlign = props.additionalColumnParameters.find(item => item.column === index);
  if (customAlign) {
    return `align-${customAlign.align}` // вернёт 'align-right','align-center' или 'align-left'
  }
  if (props.alignAllColumn) {
    if ((props.alignAllColumn === 'right' || props.alignAllColumn === 'center') && index === 0) {
      return 'align-left'
    }
    return `align-${props.alignAllColumn}`
  }
  if (index === 0) return 'align-left'

  const shouldBeRight = isNumericColumn(index) || (isHeader && props.alignRightHeaderColumn)

  return shouldBeRight ? 'align-right' : 'align-left'

}
