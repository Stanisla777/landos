    initCaptcha() {
      if (!window.smartCaptcha) {
        console.error('Скрипт капчи не загружен');
        this.isSending = false;
        return;
      }

      // ключ реально существует и это строка
      const sitekey = conf?.smartcaptcha_key;
      if (!sitekey || typeof sitekey !== 'string' || sitekey.trim() === '') {
        console.error('КРИТИЧЕСКАЯ ОШИБКА: smartcaptcha_key отсутствует или пуст! Это вызывает краш React Яндекса.');
        this.isSending = false;
        return;
      }

      const el = document.createElement('div');
      el.id = `yandex-captcha-ext-${Date.now()}`;
      el.style.position = 'absolute';
      el.style.left = '-9999px';
      el.style.opacity = '0';

      document.body.appendChild(el);
      this.captchaContainer = el;

      try {
        this.captchaWidgetId = window.smartCaptcha.render(el.id, {
          sitekey: sitekey,
          invisible: false,
          callback: (token) => {
            // Даем React Яндекса завершить внутренний цикл (commit phase)
            // перед тем, как мы начнем отправку и возможное удаление DOM.
            setTimeout(() => {
              this.submitToApi(token);
            }, 0); // 0 мс достаточно, чтобы передать управление в конец очереди событий
          },
        });
      } catch (e) {
        console.error('Ошибка render:', e);
        this.destroyCaptcha();
        this.isSending = false;
      }
    },

    executeCaptcha() {
      console.log("Пытаюсь выполнить капчу с ID:",this.captchaWidgetId, typeof this.captchaWidgetId)
      if (this.captchaWidgetId !== null && window.smartCaptcha) {
        window.smartCaptcha.execute(this.captchaWidgetId);
      } else {
        this.isSending = false;
      }
    },
    destroyCaptcha() {
      if (this.captchaWidgetId !== null && window.smartCaptcha) {
        try {
          window.smartCaptcha.reset(this.captchaWidgetId);
        } catch (e) {}
      }

      if (this.captchaContainer && this.captchaContainer.parentNode) {
        this.captchaContainer.parentNode.removeChild(this.captchaContainer);
      }

      this.captchaWidgetId = null;
      this.captchaContainer = null;
    },
