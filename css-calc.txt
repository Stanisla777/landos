<script src="https://smartcaptcha.yandexcloud.net/captcha.js?render=explicit" defer></script>

export default {
  data() {
    return {
      captchaWidgetId: null,
      captchaContainer: null, // Будем хранить ссылку на реальный DOM-элемент
      isSending: false,
    };
  },

  // Гарантированная очистка при уходе со страницы
  beforeUnmount() { // beforeDestroy для Vue 2
    this.destroyCaptcha();
  },

  methods: {
    handleSend() {
      if (this.isSending || !this.btn_active) return;
      this.isSending = true;

      // Если капчи еще нет, создаем и инициализируем
      if (!this.captchaWidgetId) {
        this.initCaptcha();
        // Даем React Яндекса 50-100 мс на монтирование в новый узел
        setTimeout(() => {
          this.executeCaptcha();
        }, 100);
      } else {
        // Если уже есть, просто выполняем
        this.executeCaptcha();
      }
    },

    initCaptcha() {
      if (!window.smartCaptcha) {
        console.error('Скрипт капчи не загружен');
        this.isSending = false;
        return;
      }

      // 1. Создаем элемент ВНЕ Vue, прямо в body
      const el = document.createElement('div');
      // Уникальный ID на каждый раз
      el.id = `yandex-captcha-external-${Date.now()}-${Math.random().toString(36).substr(2, 5)}`;
      el.style.display = 'none'; // Он невидимый, прячем его на всякий случай
      
      document.body.appendChild(el);
      this.captchaContainer = el;

      // 2. Инициализируем
      try {
        this.captchaWidgetId = window.smartCaptcha.render(el.id, {
          sitekey: conf?.smartcaptcha_key,
          invisible: true,
          callback: (token) => {
            this.submitToApi(token);
          },
        });
      } catch (e) {
        console.error('Ошибка инициализации капчи:', e);
        this.destroyCaptcha();
        this.isSending = false;
      }
    },

    executeCaptcha() {
      if (this.captchaWidgetId !== null && window.smartCaptcha) {
        window.smartCaptcha.execute(this.captchaWidgetId);
      } else {
        console.error('Не удалось выполнить капчу: widgetId отсутствует');
        this.isSending = false;
      }
    },

    submitToApi(token) {
      // Ваша логика отправки
      this.$refs.selectListWrapper?.classList.remove('open');
      this.closeBlock();
      
      store.dispatch('ActionGetResult', [this.oktmo, token])
        .then(() => {
          console.log('Успешно');
        })
        .catch((err) => {
          console.error('Ошибка отправки:', err);
        })
        .finally(() => {
          // 3. ПОЛНОЕ УНИЧТОЖЕНИЕ после попытки отправки
          this.destroyCaptcha();
          this.isSending = false;
          // store.dispatch('ActionButtonActive', true); // Если нужно разблокировать кнопку
        });
    },

    destroyCaptcha() {
      // Сбрасываем виджет
      if (this.captchaWidgetId !== null && window.smartCaptcha) {
        try {
          window.smartCaptcha.reset(this.captchaWidgetId);
        } catch (e) {}
      }
      
      // Физически удаляем DOM-элемент из body, чтобы не оставлять мусор и "зомби" React
      if (this.captchaContainer && this.captchaContainer.parentNode) {
        this.captchaContainer.parentNode.removeChild(this.captchaContainer);
      }

      // Обнуляем ссылки
      this.captchaWidgetId = null;
      this.captchaContainer = null;
    }
  }
}
