clearBtn.addEventListener('click', (e) => {
  e.preventDefault();
  e.stopPropagation();
  
  const parentLabel = item.closest('label');
  
  // 1️⃣ Очищаем маску
  mask.value = '';
  
  // 2️⃣ 🔥 Удаляем ошибку формата (но не required — пусть появится, если поле обязательное и пустое)
  if (parentLabel) {
    // Находим только ошибку формата (не required)
    const formatError = parentLabel.querySelector('.js--input__error:not(.js--input__error_required)');
    if (formatError) {
      formatError.remove();
    }
    // Убираем стили ошибки, только если не осталось required-ошибки
    if (!parentLabel.querySelector('.js--input__error_required')) {
      parentLabel.classList.remove('input_error', 'ds-checkbox--error');
    }
  }
  
  // 3️⃣ Возвращаем фокус (опционально)
  item.focus();
});
