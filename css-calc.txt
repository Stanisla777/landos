handleOpen(event) {
  this.modalData = event.detail || {};
  this.parseDataMethod();
  
  this.$nextTick(() => {
    this.setupEventListeners();
    
    // 1. Сначала делаем модалку "существующей" для Vue (но она всё ещё скрыта через CSS opacity: 0)
    this.showModal = true;

    // 2. Ждём, пока браузер завершит текущий цикл отрисовки DOM
    requestAnimationFrame(() => {
      // 3. И только теперь, на следующем кадре, блокируем body.
      // Браузер уже "знает" о модалке, и фиксация body не вызовет резкого пересчёта вместе с её появлением.
      this.AddClassBody();
    });
  });
}
