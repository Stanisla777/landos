mutationInputIncomeMan(state, received_perem) {
      // 1. Получаем массив всех ключей из пришедшего объекта: ['field_1', 'field_2', 'field_3']
      Object.keys(received_perem).forEach(key => {

        // 2. Ищем в массиве store объект, у которого modelInput совпадает с ключом
        const targetItem = state.array_all_income_period_man.find(
          item => item.modelInput === key
        );

        // 3. Если такой объект найден, обновляем его значение
        if (targetItem) {
          targetItem.value = received_perem[key];
        }

      });
      console.log('state.array_all_income_period_man');
      console.log(state.array_all_income_period_man);
    }

watch:{

        store.dispatch('ActionInputIncomeMan', newFields)

    },
}
