// src/composables/useMaxValue.js
import { watch } from 'vue';

export function useMaxValue(modelRef, maxValue) {
  watch(modelRef, (newValue) => {
    if (!newValue) return;

    // Убираем пробелы для корректного сравнения чисел
    const cleanStr = String(newValue).replace(/\s/g, '');
    const num = parseFloat(cleanStr);

    if (!isNaN(num) && num > maxValue) {
      // Если превысили лимит, заменяем на максимальное значение 
      // и сразу форматируем его с пробелами (например, "60 000 000")
      modelRef.value = Number(maxValue).toLocaleString('ru-RU');
    }
  });
}

// src/composables/useDebounceStoreSync.js
import { ref, watch, onMounted, onBeforeUnmount, nextTick } from 'vue';
import store from "../store";
import { debounce } from "../../../modules/debounce";

export function useDebounceStoreSync(actionName, getInitialValue) {
  const dataField = ref('');
  const isInitialized = ref(false);
  const skipNextWatch = ref(true);

  // Создаем debounce-функцию
  const debouncedUpdate = debounce((newValue) => {
    if (skipNextWatch.value || !isInitialized.value) return;

    const cleanedFields = String(newValue).replace(/\s/g, '');
    store.dispatch('ActionShowLoader', true);

    // Используем parseInt, так как для стоимости недвижимости дроби обычно не нужны
    const valueToSave = cleanedFields === '' ? '' : parseInt(cleanedFields, 10);

    store.dispatch(actionName, valueToSave)
      .finally(() => {
        store.dispatch('ActionShowLoader', false);
      });
  }, 500);

  onMounted(() => {
    skipNextWatch.value = true;

    nextTick(() => {
      // Получаем значение: поддерживаем как функцию-геттер, так и ComputedRef
      const val = typeof getInitialValue === 'function' 
        ? getInitialValue() 
        : getInitialValue?.value;

      if (val === 0 || val === '' || val === null || val === undefined) {
        dataField.value = '';
      } else {
        // Если в Store число, форматируем его с пробелами для красивого отображения
        dataField.value = typeof val === 'number' ? val.toLocaleString('ru-RU') : String(val);
      }

      nextTick(() => {
        skipNextWatch.value = false;
        isInitialized.value = true;
      });
    });
  });

  watch(dataField, (newValue) => {
    if (skipNextWatch.value || !isInitialized.value) return;
    debouncedUpdate(newValue);
  });

  onBeforeUnmount(() => {
    debouncedUpdate.cancel();
  });

  // Возвращаем только то, что нужно компоненту
  return {
    dataField,
    isInitialized,
    skipNextWatch
  };
}

import { useMaxValue } from '../composables/useMaxValue';
