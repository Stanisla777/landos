function resetCaptcha(container, destroyDom = false) {
  const state = widgetStates.get(container);

  if (state && window.smartCaptcha) {
    try {
      window.smartCaptcha.reset(state.widgetId);
    } catch (e) {
      console.warn('[CaptchaCore] Ошибка при reset:', e);
    }
    widgetStates.delete(container);
  }

  if (destroyDom && container) {
    
    if (typeof container.remove === 'function') {
      container.remove();
    } else if (container.parentNode) {
      // Фоллбэк для старых браузеров
      container.parentNode.removeChild(container);
    } else {
      console.warn('[CaptchaCore] Контейнер уже отсоединён от DOM');
    }
  }
}
