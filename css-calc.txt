// Универсальная функция очистки поля
function clearFieldWithValidation(input, maskInstance) {
  const parentLabel = input.closest('label');
  
  // Очищаем маску
  if (maskInstance) {
    maskInstance.value = '';
  } else {
    input.value = '';
  }
  
  // "Умное" удаление ошибок
  if (parentLabel) {
    // Удаляем только ошибки формата (не required)
    parentLabel.querySelectorAll('.js--input__error:not(.js--input__error_required)')
      .forEach(err => err.remove());
    
    // Убираем стили ошибки, если не осталось required
    if (!parentLabel.querySelector('.js--input__error_required')) {
      parentLabel.classList.remove('input_error', 'ds-checkbox--error');
    }
  }
  
  // Возвращаем фокус
  input.focus();
}

// Использование в iMaskMail, iMaskTel и т.д.:
clearBtn.addEventListener('click', (e) => {
  e.preventDefault();
  e.stopPropagation();

});
