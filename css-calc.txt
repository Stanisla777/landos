watch: {
  fieldsIncomeMan: {
    // Используем обычную function, чтобы сохранить контекст this компонента
    handler: debounce(function (newFields) {
      // 1. ГЛАВНАЯ ЗАЩИТА: если обновление пришло из Store (клик по табу), прерываем выполнение
      if (this.isUpdatingFromStore) return;

      // 2. (Опционально) Очищаем значения объекта от пробелов перед отправкой
      const cleanedFields = {};
      for (const key in newFields) {
        // Превращаем "8 000 000" в 8000000. Если там уже число или пусто, будет корректно
        cleanedFields[key] = parseInt(String(newFields[key]).replace(/\s/g, ''), 10) || 0;
      }

      // 3. Показываем лоадер
      store.dispatch('ActionShowLoader', true);

      // 4. Отправляем очищенный объект в Store
      store.dispatch('ActionInputIncomeMan', cleanedFields)
        .then(() => {
          // 5. Прячем лоадер после завершения
          store.dispatch('ActionShowLoader', false);
        })
        .catch(() => {
          // На случай ошибки тоже нужно убрать лоадер
          store.dispatch('ActionShowLoader', false);
        });

    }, 500), // Задержка 500мс
    
    deep: true,
    immediate: false
  }
}

watch: {
  previousPayments: {
    handler: debounce(function (newValue) {
      // newValue выглядит так: { deduction_purchase: '1 000 000', deduction_interest: '500 000' }
      
      // 1. Создаем новый объект с очищенными числами
      const cleanedPayments = {};
      for (const key in newValue) {
        cleanedPayments[key] = parseInt(String(newValue[key]).replace(/\s/g, ''), 10) || 0;
      }

      // 2. Показываем лоадер
      store.dispatch('ActionShowLoader', true);

      // 3. Отправляем очищенный объект в Store
      store.dispatch('ActionInputPreviousPaymentsMan', cleanedPayments)
        .then(() => {
          store.dispatch('ActionShowLoader', false);
        })
        .catch(() => {
          store.dispatch('ActionShowLoader', false);
        });

    }, 500),
    
    deep: true,
    immediate: false
  }
}
