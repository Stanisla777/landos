import { watch, nextTick } from 'vue';

export function useMaxValue(modelRef, maxValue, inputRef = null) {
  watch(modelRef, async (newValue) => {
    if (!newValue) return;

    const cleanStr = String(newValue).replace(/\s/g, '');
    const num = parseFloat(cleanStr);
    
    // ⭐ Форматируем максимальное значение для сравнения
    const maxFormatted = Number(maxValue).toLocaleString('ru-RU');

    // Если текущее значение уже равно максимальному, не трогаем его
    if (cleanStr === String(maxValue) || newValue === maxFormatted) {
      return;
    }

    if (!isNaN(num) && num > maxValue) {
      const cappedValue = maxFormatted;

      // 1. Сбрасываем значение на максимальное
      modelRef.value = cappedValue;

      // 2. Исправляем позицию курсора
      if (inputRef && inputRef.value) {
        await nextTick();

        const nativeInput = inputRef.value.$el 
          ? inputRef.value.$el.querySelector('input') 
          : inputRef.value;

        if (nativeInput && typeof nativeInput.setSelectionRange === 'function') {
          nativeInput.focus();

          setTimeout(() => {
            const length = cappedValue.length;
            nativeInput.setSelectionRange(length, length);
          }, 0);
        }
      }
    }
  });
}
