.advanced-editor__accordion-item{
      &.hidden{
        max-height: 0;
        overflow: hidden;
        opacity: 0;
        display: block;
        padding-top: 0;
        padding-bottom: 0;
        //border-bottom-width: 0;
        border-bottom-color: transparent;
        transition: max-height 0.5s ease, opacity 0.5s ease;
      }
    }

btn.addEventListener('click', () => {
    const hiddenItems = parent.querySelectorAll('.js-accordion-parent.hidden');

    hiddenItems.forEach(item => {
      // Сохраняем оригинальную высоту для плавной анимации
      const originalHeight = item.scrollHeight;

      // Временные стили для анимации
      item.style.maxHeight = '0';
      item.style.opacity = '0';
      item.style.display = 'block';
      item.classList.remove('hidden');

      // Запуск анимации
      requestAnimationFrame(() => {
        item.style.transition = 'max-height 0.5s ease, opacity 0.5s ease';
        item.style.maxHeight = `${originalHeight}px`;
        item.style.opacity = '1';
      });

      // Очистка после анимации
      item.addEventListener('transitionend', () => {
        item.style.transition = '';
        item.style.maxHeight = '';
        item.style.opacity = '';
        item.style.display = '';
      }, { once: true });
    });

    // Скрываем кнопку после раскрытия
    btn.classList.add('btn-accordion-hidden');
  });
