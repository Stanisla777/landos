      disableTransition: false, // <-- НОВЫЙ ФЛАГ для управления анимацией
      btn_next: true,
      btn_prev: true,
      count: 0,
      count_slide: 0,
    };

        <Slider
          ref="slider"
          v-model="dragValue"
          :transition="disableTransition ? 'none' : 'all 0.2s ease'" 
          @update:modelValue="onValueChange"
          @dragstart="onDragStart"
          @dragend="onDragEnd"
          :min="0"
          :max="optionsList.length - 1"
          :step="0.01"
          :tooltip="false"
          :height="4"
          :dot-size="24"
          :disabled="time_end"
          class="test-n-slider__track"

    // Гарантируем, что при начале перетаскивания анимация всегда включена
    onDragStart() {
      this.isDragging = true;
      this.disableTransition = false;
    },

    onValueChange(newValue) {
      if (this.time_end || this.optionsList.length === 0) return;

      if (this.isDragging) {
        // Сценарий 1: Перетаскивание. Просто обновляем значение для плавности.
        this.dragValue = newValue;
        return;
      }

      // Сценарий 2: Чистый клик (tap) по треку.
      const targetIndex = Math.round(newValue);
      const clampedIndex = Math.max(0, Math.min(this.optionsList.length - 1, targetIndex));

      if (clampedIndex === this.selectedIndex) {
        // Кликнули рядом с текущим маркером -> НЕВИДИМЫЙ ОТКАТ
        // 1. Отключаем анимацию
        this.disableTransition = true;
        // 2. Мгновенно возвращаем значение на место
        this.dragValue = this.selectedIndex;
        // 3. Включаем анимацию обратно в следующем кадре, чтобы будущие действия были плавными
        this.$nextTick(() => {
          this.disableTransition = false;
        });
      } else {
        // Кликнули ближе к ДРУГОМУ маркеру -> ПЛАВНОЕ ПЕРЕМЕЩЕНИЕ
        this.selectedIndex = clampedIndex;
        this.dragValue = clampedIndex;
        this.updateStateFromIndex(clampedIndex);
      }
    },

    onDragEnd() {
      this.isDragging = false;
      
      // Логика "примагничивания" после окончания перетаскивания
      const targetIndex = Math.round(this.dragValue);
      const clampedIndex = Math.max(0, Math.min(this.optionsList.length - 1, targetIndex));

      if (clampedIndex !== this.selectedIndex) {
        // Дотянули до нового маркера
        this.selectedIndex = clampedIndex;
        this.dragValue = clampedIndex;
        this.updateStateFromIndex(clampedIndex);
      } else {
        // Начали тянуть, но отпустили рядом с тем же маркером -> НЕВИДИМЫЙ ОТКАТ
        this.disableTransition = true;
        this.dragValue = this.selectedIndex;
        this.$nextTick(() => {
          this.disableTransition = false;
        });
      }
    },

    selectByIndex(index) {
      if (this.time_end) return;
      this.selectedIndex = index;
      this.dragValue = index;
      this.updateStateFromIndex(index);
