computed: {
  normalizedRows() {
    const headerCount = this.tableData.headers?.length || 0;
    
    if (headerCount === 0) return [];
    
    return this.tableData.rows?.map((row, rowIndex) => {
      const cells = row.cells || [];
      const currentCount = cells.length;
      
      // Логируем в dev-режиме, если количество ячеек не совпадает
      if (process.env.NODE_ENV === 'development' && currentCount !== headerCount) {
        console.warn(
          `[TableStory] Строка ${rowIndex}: ожидается ${headerCount} ячеек, получено ${currentCount}`
        );
      }
      
      // Обрезаем лишние ячейки
      let normalizedCells = [...cells].slice(0, headerCount);
      
      // Добавляем пустые ячейки, если не хватает
      while (normalizedCells.length < headerCount) {
        normalizedCells.push({
          id: `empty-${rowIndex}-${normalizedCells.length}`,
          label: '',
          link: '',
          icon: ''
        });
      }
      
      // Всегда возвращаем объект с id и нормализованными cells
      return {
        id: row.id || `row-${rowIndex}`,
        ...row,
        cells: normalizedCells
      };
    }) || [];
  }
}
