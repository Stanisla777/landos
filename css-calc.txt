@update:modelValue="(newValue) => handleUserInput(item.modelInput, newValue)"

handleUserInput(modelInputKey, newValue) {
      // 1. Мгновенно обновляем локальное состояние, чтобы UI был отзывчивым
      this.fieldsIncomeMan[modelInputKey] = newValue;
      
      // 2. Отправляем в Store только измененное поле (лучше через commit, но можно и dispatch)
      // Если у тебя там action, который внутри делает commit:
      store.dispatch('ActionInputIncomeMan', { 
        key: modelInputKey, 
        value: newValue 
      });
      
      // ВАЖНО: Убедись, что твой ActionInputIncomeMan и мутация теперь принимают 
      // объект { key: 'field_1', value: '...' }, а не весь массив newFields.
    }

mutationInputIncomeMan(state, payload) {
      // 1. Ищем в массиве объект, у которого modelInput совпадает с переданным ключом
      const targetItem = state.array_all_income_period_man.find(
        item => item.modelInput === payload.key
      );

      // 2. Если такой элемент найден, обновляем только его значение
      if (targetItem) {
        targetItem.value = payload.value;
      }
    },
