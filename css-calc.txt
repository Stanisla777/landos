        const containerId = `captcha-reply-${Date.now()}-${Math.random().toString(36).substr(2, 5)}`;
        const container = CaptchaCore.createCaptchaContainer(replyForm, containerId);

function CommentsFormValidationAnswer() {

if (!errors.name && !errors.check && !errors.text) {
        // 🔥 ПОЛУЧАЕМ УЖЕ СОЗДАННЫЙ КОНТЕЙНЕР
        const containerId = form.dataset.captchaContainerId;
        if (!containerId) {
          console.error('Контейнер капчи не найден');
          return;
        }
        
        const container = form.querySelector(`#${containerId}`);
        if (!container) {
          console.error('Элемент контейнера капчи не найден');
          return;
        }
}

}

function removeReplyForm(form) {
  if (!form) return;
  
  // 🔥 СБРАСЫВАЕМ И УДАЛЯЕМ КОНТЕЙНЕР КАПЧИ
  const containerId = form.dataset.captchaContainerId;
  if (containerId) {
    const container = form.querySelector(`#${containerId}`);
    if (container) {
      CaptchaCore.resetCaptcha(container, true); // true = удалить DOM-элемент
    }
  }
  
  // Удаляем саму форму
  form.remove();
}
