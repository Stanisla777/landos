import { watch, nextTick } from 'vue';

export function useMaxValue(modelRef, maxValue, inputRef = null) {
  watch(modelRef, async (newValue, oldValue) => {
    if (!newValue) return;

    const cleanStr = String(newValue).replace(/\s/g, '');
    const num = parseFloat(cleanStr);
    const maxFormatted = Number(maxValue).toLocaleString('ru-RU');

    // Если значение уже равно максимальному, не трогаем его
    if (newValue === maxFormatted || cleanStr === String(maxValue)) {
      return;
    }

    // Если значение НЕ превышает максимум, тоже не трогаем
    if (isNaN(num) || num <= maxValue) {
      return;
    }

    // ⭐ ДОШЛИ СЮДА ЗНАЧИТ: значение действительно превысило максимум
    // и нужно его обрезать до максимального
    modelRef.value = maxFormatted;

    // Исправляем позицию курсора
    if (inputRef && inputRef.value) {
      await nextTick();

      const nativeInput = inputRef.value.$el 
        ? inputRef.value.$el.querySelector('input') 
        : inputRef.value;

      if (nativeInput && typeof nativeInput.setSelectionRange === 'function') {
        nativeInput.focus();
        setTimeout(() => {
          const length = maxFormatted.length;
          nativeInput.setSelectionRange(length, length);
        }, 0);
      }
    }
  });
}
