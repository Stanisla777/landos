mounted() {
  this.parseHistoryArray(this.arrayHistory);
  this.$nextTick(() => {
    this.initSwiper();
  });

  this.handleResize = this.debounce(() => {
    if (!this.swiper) return;

    // Блокируем свайп на время обновления
    const allowNext = this.swiper.allowSlideNext;
    const allowPrev = this.swiper.allowSlidePrev;
    this.swiper.allowTouchMove = false;

    // Принудительный пересчёт размеров и слайдов
    this.swiper.updateSize();
    this.swiper.updateSlides();

    // Используем двойной requestAnimationFrame,
    // чтобы дождаться завершения перерисовки браузера
    requestAnimationFrame(() => {
      requestAnimationFrame(() => {
        if (!this.swiper) return;
        
        this.swiper.updateProgress();
        this.swiper.update();
        this.updateControls(this.swiper);

        // Возвращаем свайп
        this.swiper.allowTouchMove = true;
        this.swiper.allowSlideNext = allowNext;
        this.swiper.allowSlidePrev = allowPrev;
      });
    });
  }, 300);

  window.addEventListener('resize', this.handleResize);
},
