visibleTabs() {
    // Ищем индекс первого элемента, у которого isActive === true
    const activeIndex = this.tabs.findIndex(tab => tab.isActive === true)
    
    // Если такой элемент не найден — возвращаем пустой массив
    if (activeIndex === -1) return []
    
    // Берём все элементы от начала до активного включительно
    return this.tabs.slice(0, activeIndex + 1)
  }
