import store from "../store";
import { debounce } from "../../../modules/debounce";

export function sandToStoreAmountField(actionName) {
  return {
    data() {
      return {
        dataField: 0,
        isInitialized: false,
        isUpdatingFromStore: false,
      };
    },

    created() {
      // Настраиваем debounce. Он готов к работе.
      this.debouncedIncomeUpdate = debounce((newValue) => {
        // Двойная защита: не отправляем, если обновление инициировано Store или компонент не готов
        if (this.isUpdatingFromStore || !this.isInitialized) return;

        const cleanedFields = String(newValue).replace(/\s/g, '');
        store.dispatch('ActionShowLoader', true);

        const valueToSave = cleanedFields === '' ? '' : parseInt(cleanedFields, 10);

        store.dispatch(actionName, valueToSave)
          .finally(() => {
            store.dispatch('ActionShowLoader', false);
          });
      }, 500);
    },

    mounted() {
      // Разрешаем реакцию на действия пользователя только после полной отрисовки
      this.isInitialized = true;
    },

    watch: {
      dataField: {
        handler(newValue) {
          if (this.isUpdatingFromStore || !this.isInitialized) return;
          this.debouncedIncomeUpdate(newValue);
        },
        immediate: false
      }
    },

    beforeUnmount() {
      if (this.debouncedIncomeUpdate) {
        this.debouncedIncomeUpdate.cancel();
      }
    }
  };
}
