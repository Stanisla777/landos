mutationAnswers(state, received_perem) {
      // 1. Базовые поля
      state.data_year = received_perem.dataYear;
      state.check_calculate_together = received_perem.calculateTogether;
      state.apartment_price = received_perem.apartmentPrice;
      state.interest_paid = received_perem.interestPaid;
      state.funds_used = received_perem.fundsUsed;

      // 2. Предыдущие вычеты (мужчины)
      if (received_perem.previousPaymentsMan) {
        state.previousPaymentsMan = {
          ...state.previousPaymentsMan,
          ...received_perem.previousPaymentsMan
        };
      }
      // 2. Предыдущие вычеты (женщины)
      if (received_perem.previousPaymentsWife) {
        state.previousPaymentsWife = {
          ...state.previousPaymentsWife,
          ...received_perem.previousPaymentsWife
        };
      }


      // 3. Динамическое обновление массива доходов (для всех лет!)
      if (received_perem.incomeMan && Array.isArray(received_perem.incomeMan)) {
        received_perem.incomeMan.forEach((incomingItem, index) => {
          // Проверяем, что в state есть элемент с таким индексом
          if (state.array_all_income_period_man[index]) {
            const stateItem = state.array_all_income_period_man[index];

            // Обновляем флаги и значение
            stateItem.isActive = incomingItem.isActive;
            stateItem.isShow = incomingItem.isShow;
            stateItem.value = incomingItem.value;

            // Обновляем коэффициенты, приводя их к формату, который ожидает ds-multi-select
            if (incomingItem.additionalCoefficient && stateItem.additionalCoefficient) {
              const distVal = incomingItem.additionalCoefficient.districtCoefficient;
              const northVal = incomingItem.additionalCoefficient.northernAllowance;

              // Формируем массив объектов, понятный для ds-multi-select
              if (distVal !== undefined) {
                stateItem.additionalCoefficient.districtCoefficient.value = [
                  { label: String(distVal), value: String(distVal) } // или { id: 1, label: String(distVal) } в зависимости от вашей UI-библиотеки
                ];
              }
              if (northVal !== undefined) {
                stateItem.additionalCoefficient.northernAllowance.value = [
                  { label: String(northVal), value: String(northVal) }
                ];
              }
            }
          }
        });
      }
    },

methods: {
  initializeFieldsIncomeFromStore() {
    const fields = {};

    this.array_all_income_period.forEach(tab => {
      // 1. Основной доход
      fields[tab.modelInput] = tab.value || '';

      // 2. Коэффициенты
      if (tab.additionalCoefficient) {
        const baseKey = tab.additionalCoefficient.modelInput;

        // Берем значение из Store. 
        // Если это массив (как мы сохранили в мутации), передаем его целиком для ds-multi-select.
        // Иначе передаем как есть.
        const distVal = tab.additionalCoefficient.districtCoefficient?.value;
        fields[baseKey + '_district'] = Array.isArray(distVal) ? distVal : (distVal || '');

        const northVal = tab.additionalCoefficient.northernAllowance?.value;
        fields[baseKey + '_northern'] = Array.isArray(northVal) ? northVal : (northVal || '');
      }
    });

    this.fieldsIncome = fields;
  }
