store.watch(
      state => state.data_year,
      (newYear) => {
        const currentYear = new Date().getFullYear();
        const diff = currentYear - newYear;

        let targetId = null;
        if (diff === 0) targetId = null;       // Текущий год
        else if (diff === 1) targetId = "1";   // Прошлый год
        else if (diff === 2) targetId = "2";   // Два года назад
        else if (diff >= 3) targetId = "3";    // Три и более лет назад

        // Вызываем нашу универсальную функцию для применения состояний
        updateIncomePeriodArray(store.state.array_all_income_period_man, targetId);
        updateIncomePeriodArray(store.state.array_all_income_period_wife, targetId);
      },
      { immediate: true } // <--- МАГИЯ ЗДЕСЬ: сработает мгновенно при создании Store
    );

mutationCalendarYear(state, received_perem) {
    // Просто меняем год. Watch в плагине автоматически отреагирует на это изменение
    // и вызовет updateIncomePeriodArray с правильными параметрами.
    state.data_year = received_perem;
  },
