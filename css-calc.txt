export default {
  data() {
    return {
      captchaId: null,
      isCaptchaReady: false,
    };
  },
  mounted() {
    // $nextTick гарантирует, что DOM-элемент уже существует
    this.$nextTick(() => {
      this.initCaptcha();
    });
  },
  beforeUnmount() { // или beforeDestroy для Vue 2
    // ОЧЕНЬ ВАЖНО: очищаем капчу при удалении компонента, 
    // чтобы React-инстанс Яндекс.Капчи не "завис" и не вызвал ошибку
    if (this.captchaId !== null && window.smartCaptcha) {
      window.smartCaptcha.reset(this.captchaId);
      this.captchaId = null;
    }
  },
  methods: {
    initCaptcha() {
      if (this.captchaId !== null || !window.smartCaptcha) return;

      // Получаем реальный DOM-элемент через ref
      const container = this.$refs.captchaContainer;
      if (!container) return;

      this.captchaId = window.smartCaptcha.render(container, {
        sitekey: conf?.smartcaptcha_key,
        invisible: true,
        callback: (token) => {
          this.sendingResultToApi(token);
        },
      });
      
      this.isCaptchaReady = true;
    },

    sendingResult() {
      if (!this.btn_active) return;

      store.dispatch('ActionStandingButton');
      store.dispatch('ActionButtonActive', false);
      
      // Если капча уже готова, просто выполняем её
      if (this.isCaptchaReady && this.captchaId !== null) {
        window.smartCaptcha.execute(this.captchaId);
      } else {
        // Если вдруг не успела инициализироваться, инициируем и выполняем
        this.initCaptcha();
        // Небольшая задержка, чтобы render успел отработать синхронно
        this.$nextTick(() => {
          if (this.captchaId !== null) {
            window.smartCaptcha.execute(this.captchaId);
          }
        });
      }
    },

    sendingResultToApi(token) {
      // Ваша логика отправки с токеном
      console.log('Token received:', token);
      
      // После успешной отправки рекомендуется сбросить капчу, 
      // чтобы её можно было пройти снова при ошибке валидации
      if (this.captchaId !== null && window.smartCaptcha) {
        window.smartCaptcha.reset(this.captchaId);
      }
    }
  }
}
