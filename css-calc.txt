handleOpen(event) {
  this.modalData = event.detail || {};
  this.parseDataMethod();
  
  this.$nextTick(() => {
    this.setupEventListeners();
    
    // 1. Сначала говорим Vue показать модалку (она появится, но фон ещё не зафиксирован)
    this.showModal = true;

    // 2. Даём Safari 50мс на спокойную отрисовку DOM-дерева модалки
    setTimeout(() => {
      // 3. И только теперь применяем фиксацию body. 
      // Конфликта перерисовки больше не будет.
      this.AddClassBody();
    }, 50); 
  });
}
