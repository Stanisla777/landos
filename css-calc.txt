<template>
  <div>
    <div 
      v-for="(item, index) in average_monthly_ncome_man" 
      :key="index" 
      class="input-story-container"
    >
      <DsInput 
        :placeholder="'Поле ' + (index + 1)" 
        v-model="fields[item]" 
      />
    </div>
  </div>
</template>

<script>
export default {
  data() {
    return {
      // Один объект вместо отдельных переменных
      fields: {
        field_1: '',
        field_2: '',
        field_3: ''
      }
    }
  },

  computed: {
    average_monthly_ncome_man() {
      return this.$store.getters.AVERAGEMONTHLYNCOMEMAN
    }
  },

  watch: {
    // Следим за объектом fields
    fields: {
      handler(newFields) {
        // Отправляем весь объект в store
        this.$store.dispatch('ActionInput', newFields)
      },
      deep: true // обязательно для отслеживания изменений внутри объекта
    }
  }
}
</script>
