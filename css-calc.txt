limit_init_fee(newVal) {
        // Если значение стало 0 или меньше (пользователь временно очистил поле),
        // мы просто игнорируем обновление. Слайдер сохранит свой старый диапазон,
        // визуального скачка не будет, и ошибки noUiSlider не возникнет.
        if (newVal <= 0) {
            return;
        }

        // Обновляем диапазон только тогда, когда пришло корректное положительное число
        this.realtySlider.updateOptions({
            range: {
                min: this.stgMin,
                max: newVal
            }
        });
    },
