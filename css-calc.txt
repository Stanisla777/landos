dataField(newValue) {
    // 1. Пропускаем пустое поле и промежуточные состояния ввода
    if (newValue === '' || newValue === '.' || newValue === null || newValue === undefined) {
      return;
    }

    const num = parseFloat(newValue);

    // 2. Если NaN (некорректный ввод)
    if (isNaN(num)) return;

    // 3. Ограничение снизу: меньше 1 → ставим 1
    if (num < 1) {
      this.dataField = '1';
      return;
    }

    // 4. ⭐ ИСПРАВЛЕНИЕ: больше ИЛИ РАВНО 3 → ставим 3
    // Это блокирует ввод точки после 3, так как 3. = 3, 3.0 = 3, 3.01 > 3
    if (num >= 3) {
      this.dataField = '3';
      return;
    }
  }
