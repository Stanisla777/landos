    // Разрешаем полю быть пустым во время ввода, не ломаем стейт промежуточными значениями
    calculationPageLoad() {
      let cleanValue = parseInt(String(this.dataField).replace(/\s/g, ''));

      // Если поле пустое, содержит 0 или не является числом — просто выходим.
      // Даем пользователю возможность полностью стереть значение и начать вводить новое.
      if (isNaN(cleanValue) || cleanValue === 0 || String(this.dataField).trim() === '') {
        return;
      }

      // Если число меньше минимума, но больше 0 (например, пользователь ввел "50"), 
      // тоже игнорируем, пока он не закончит ввод.
      if (cleanValue < this.stgMin) {
        return;
      }

      this.dataFieldForCalculation = cleanValue;
      store.dispatch('ActionRealPriceAppartment', cleanValue);
    },

    // А вот здесь, при ПОТЕРЕ ФОКУСА, мы жестко гарантируем минимальное значение
    lossFocus(el) {
      const element = el.currentTarget;
      let cleanValue = parseInt(element.value.replace(/\s/g, ''));

      // Если после ухода с поля оно пустое, равно 0 или меньше минимума
      if (isNaN(cleanValue) || cleanValue === 0 || cleanValue < this.stgMin) {
        this.dataField = this.stgMin.toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ');
        this.realtySlider.set(this.stgMin);
        store.dispatch('ActionRealPriceAppartment', this.stgMin);
      }
    },

@blur="lossFocus"

lossFocus(el) {
      const element = el.currentTarget;
      let cleanValue = parseInt(element.value.replace(/\s/g, ''));

      if (isNaN(cleanValue) || cleanValue === 0 || element.value.trim() === '') {
        this.dataField = '0'; // Явно ставим 0, чтобы исчез плейсхолдер
        this.dataFieldForCalculation = 0;
        this.realtySlider.set(0);
        store.dispatch('ActionMaternalCapital', 0);
      }
    },
