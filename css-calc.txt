mutations: {
  mutationCalendarYear(state, received_perem) {
    // 1. Сохраняем выбранный год
    state.data_year = received_perem;

    // 2. Вычисляем разницу между текущим годом и выбранным
    const currentYear = new Date().getFullYear();
    const diff = currentYear - received_perem;
    
    let targetId = null; // По умолчанию ни один таб не активен

    // 3. Определяем, какой таб должен быть активен по вашим правилам
    if (diff === 0) {
      targetId = null; // Текущий год: все isActive = false
    } else if (diff === 1) {
      targetId = "1"; // Прошлый год: активен 1-й таб
    } else if (diff === 2) {
      targetId = "2"; // Два года назад: активен 2-й таб
    } else if (diff >= 3) {
      targetId = "3"; // Три и более лет назад: активен 3-й таб
    }

    // 4. Применяем состояние к массиву табов
    state.array_all_income_period_man.forEach((item, index) => {
      if (targetId === null) {
        // Если текущий год, все табы неактивны
        item.isActive = false;
      } else {
        // Иначе активируем только нужный таб
        item.isActive = (item.id === targetId);

       
        // если мы автоматически активировали таб, очищаем данные у всех табов, 
        // которые идут ПОСЛЕ него в массиве.
        const activeIndex = state.array_all_income_period_man.findIndex(i => i.id === targetId);
        if (index > activeIndex) {
          item.value = '';
          if (item.additionalCoefficient) {
            item.additionalCoefficient.districtCoefficient.value = '';
            item.additionalCoefficient.northernAllowance.value = '';
          }
        }
      }
    });

    console.log('Год изменен на:', state.data_year, '| Разница лет:', diff, '| Активный таб ID:', targetId);
  },

  // ... ваши остальные мутации (mutationTabTimeMan и т.д.)
}
