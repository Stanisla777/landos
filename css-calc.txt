handleIncomeInput(key, value) {
    const MAX_LIMIT = 10000000;
    const cleanStr = String(value).replace(/\s/g, '');
    const num = parseFloat(cleanStr);

    let finalValue = value;

    // Если введено число и оно превышает лимит, сразу подменяем его на максимум
    if (!isNaN(num) && num > MAX_LIMIT) {
      finalValue = Number(MAX_LIMIT).toLocaleString('ru-RU'); // "10 000 000"
    }

    // Обновляем реактивный объект (в Vue 3 это работает без this.$set)
    this.fieldsIncome[key] = finalValue;

    // Запускаем debounce для сохранения в стор. 
    // Передаем копию объекта, чтобы debounce работал с актуальными данными.
    if (this.debouncedIncomeUpdate) {
      this.debouncedIncomeUpdate({ ...this.fieldsIncome });
    }
  },

watch: {
  // Реагирует на изменения в Store (клик по табу или первоначальная загрузка)
  array_all_income_period: {
    handler() {
      this.skipNextWatch = true;

      this.$nextTick(() => {
        const fields = {};
        this.array_all_income_period.forEach(tab => {
          // Форматируем число из стора для красивого отображения
          fields[tab.modelInput] = tab.value ? Number(tab.value).toLocaleString('ru-RU') : '';

          if (tab.additionalCoefficient) {
            const baseKey = tab.additionalCoefficient.modelInput;
            fields[baseKey + '_district'] = tab.additionalCoefficient.districtCoefficient.value || '';
            fields[baseKey + '_northern'] = tab.additionalCoefficient.northernAllowance.value || '';
          }
        });
        
        this.fieldsIncome = fields;
        
        this.$nextTick(() => {
          this.skipNextWatch = false;
          this.isInitialized = true;
        });
      });
    },
    deep: true
  }
  
  // Watcher на fieldsIncome для ввода пользователя УДАЛЕН. 
  // Эту задачу теперь элегантно выполняет @update:modelValue + handleIncomeInput
},

beforeUnmount() {
  if (this.debouncedIncomeUpdate) {
    this.debouncedIncomeUpdate.cancel();
  }
}
