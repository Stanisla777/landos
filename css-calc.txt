lastValidValue: '1'

dataField(newValue) {
    // Пропускаем пустое поле
    if (newValue === '' || newValue === '.' || newValue === null || newValue === undefined) {
      return;
    }

    const num = parseFloat(newValue);

    if (isNaN(num)) {
      // Некорректный ввод — возвращаемся к последнему валидному
      this.dataField = this.lastValidValue;
      return;
    }

    // Проверяем диапазон
    if (num < 1 || num >= 3) {
      // ⭐ ГЛАВНОЕ: Возвращаемся к последнему валидному значению
      this.dataField = this.lastValidValue;
      return;
    }

    // Если всё валидно — сохраняем как последнее валидное
    this.lastValidValue = newValue;
  }
