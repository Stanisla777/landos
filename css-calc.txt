import store from "../store";
import { debounce } from "../../../modules/debounce";

export function sandToStoreAmountField(actionName, computedPropName) {
  return {
    data() {
      return {
        dataField: '',
        isInitialized: false,
        skipNextWatch: true, // ⭐ НОВЫЙ ФЛАГ: Игнорировать первое срабатывание вотчера
      };
    },

    created() {
      this.debouncedIncomeUpdate = debounce((newValue) => {
        // Дополнительная защита: если по какой-то причине skipNextWatch не сработал
        if (this.skipNextWatch || !this.isInitialized) return;

        const cleanedFields = String(newValue).replace(/\s/g, '');
        store.dispatch('ActionShowLoader', true);

        const valueToSave = cleanedFields === '' ? '' : parseInt(cleanedFields, 10);

        store.dispatch(actionName, valueToSave)
          .finally(() => {
            store.dispatch('ActionShowLoader', false);
          });
      }, 500);
    },

    mounted() {
      // Гарантируем, что флаг поднят перед любым изменением
      this.skipNextWatch = true;

      this.$nextTick(() => {
        const val = this[computedPropName];
        
        if (val === 0 || val === '' || val === null || val === undefined) {
          this.dataField = '';
        } else {
          this.dataField = val; // Это изменение будет проигнорировано вотчером
        }

        // Снимаем блокировку только в СЛЕДУЮЩЕМ тике, когда вотчер уже точно отработал (и был пропущен)
        this.$nextTick(() => {
          this.skipNextWatch = false;
          this.isInitialized = true;
        });
      });
    },

    watch: {
      dataField: {
        handler(newValue) {
          // ⭐ ГЛАВНАЯ ЗАЩИТА: Если флаг поднят, молча выходим
          if (this.skipNextWatch || !this.isInitialized) return;
          
          this.debouncedIncomeUpdate(newValue);
        },
        immediate: false
      }
    },

    beforeUnmount() {
      if (this.debouncedIncomeUpdate) {
        this.debouncedIncomeUpdate.cancel();
      }
    }
  };
}

import { debounce } from "../../../modules/debounce"; // Импортируйте ваш debounce сюда

// Создаем дебаунс-функцию для тяжелого расчета ВНЕ watch, чтобы она была одной на всё приложение
const debouncedCalculateResult = debounce((storeInstance) => {
  storeInstance.commit('mutationCalculateResult');
}, 300); // 300 мс достаточно, чтобы "схлопнуть" все начальные инициализации в один вызов

export default createStore({
  // ... state, mutations, actions

  plugins: [
    store => {
      // 1. Watch для основных расчетов
      store.watch(
        state => [
          state.array_all_income_period_man, state.data_year, state.check_calculate_together,
          state.apartment_price, state.interest_paid, state.funds_used, state.previousPaymentsMan,
          state.check_district_coefficient_man, state.array_all_income_period_wife, 
          state.check_district_coefficient_wife, state.previousPaymentsWife
        ],
        (newValues) => {
          // newValues[0] - это array_all_income_period_man
          if (newValues[0] !== null) {
            // Вместо мгновенного commit, мы вызываем debounced-функцию.
            // Если 5 компонентов обновят стейт в течение 300мс, мутация сработает ТОЛЬКО ОДИН РАЗ!
            debouncedCalculateResult(store);
          }
        },
        { deep: true, immediate: false }
      );

      // 2. Watch для проверки на null (здесь debounce не обязателен, так как операция легкая, 
      // но можно добавить, если mutationSomeThingNull тяжелая)
      store.watch(
        state => [state.family_income, state.family_count, state.whom_counting, state.selected_region],
        ([family_income, family_count, whom_counting, selected_region]) => {
          if (family_income === null || family_count === null || whom_counting === null || selected_region === null) {
            store.commit('mutationSomeThingNull');
          }
        },
        { deep: true, immediate: false }
      );
    }
  ],
});
