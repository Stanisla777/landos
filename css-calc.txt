isSanitizing: false,

watch: {
  // 1. Реагирует на изменения в Store (клик по табу или загрузка)
  array_all_income_period: {
    handler() {
      this.skipNextWatch = true;
      this.isSanitizing = true; // Блокируем любые срабатывания во время синхронизации

      this.$nextTick(() => {
        const fields = {};
        this.array_all_income_period.forEach(tab => {
          fields[tab.modelInput] = tab.value ? Number(tab.value).toLocaleString('ru-RU') : '';

          if (tab.additionalCoefficient) {
            const baseKey = tab.additionalCoefficient.modelInput;
            fields[baseKey + '_district'] = tab.additionalCoefficient.districtCoefficient.value || '';
            fields[baseKey + '_northern'] = tab.additionalCoefficient.northernAllowance.value || '';
          }
        });
        
        this.fieldsIncome = fields;
        
        this.$nextTick(() => {
          this.skipNextWatch = false;
          this.isSanitizing = false; // Разблокируем после полной отрисовки
          this.isInitialized = true;
        });
      });
    },
    deep: true
  },

  // 2. Реагируем на ввод пользователя И применяем ограничение
  fieldsIncome: {
    handler(newValue) {
      // Если это программное изменение, компонент не готов или идет процесс санитизации — игнорируем
      if (this.skipNextWatch || !this.isInitialized || this.isSanitizing) {
        return;
      }

      const MAX_LIMIT = 10000000;
      let needsCorrection = false;
      
      // Создаем копию объекта, чтобы безопасно менять значения во время итерации
      const correctedFields = { ...newValue };

      // Получаем список разрешенных ключей (только те, что привязаны к modelInput в v-for)
      // Это защитит от случайной проверки полей коэффициентов (_district, _northern)
      const validModelInputs = this.visibleBlockIncomePeriod.map(item => item.modelInput);

      for (const key in correctedFields) {
        if (validModelInputs.includes(key)) {
          const cleanStr = String(correctedFields[key]).replace(/\s/g, '');
          const num = parseFloat(cleanStr);

          // Если введено число и оно больше лимита
          if (!isNaN(num) && num > MAX_LIMIT) {
            correctedFields[key] = Number(MAX_LIMIT).toLocaleString('ru-RU');
            needsCorrection = true;
          }
        }
      }

      // Если мы обнаружили превышение лимита
      if (needsCorrection) {
        // 1. Блокируем вотчер, чтобы не вызвать debouncedUpdate для этого программного изменения
        this.isSanitizing = true;
        this.skipNextWatch = true;

        // 2. Мгновенно перезаписываем значения в реактивном объекте
        Object.assign(this.fieldsIncome, correctedFields);

        // 3. Снимаем блокировку в следующем тике, чтобы пользователь мог продолжать ввод
        this.$nextTick(() => {
          this.isSanitizing = false;
          this.skipNextWatch = false;
        });

        // Прерываем выполнение функции. Мы не вызываем debouncedUpdate, 
        // так как это не действие пользователя, а наша автоматическая коррекция.
        return;
      }

      // 3. Если коррекция не потребовалась, работаем как обычно (отправляем в стор с задержкой)
      this.debouncedIncomeUpdate(newValue);
    },
    deep: true,
    immediate: false
  }
},
