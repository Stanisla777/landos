.advanced-editor {
  &__table-container {

    // ── первая колонка узкая (для нумерации) ──
    &.number {
      .advanced-editor__table-row {
        grid-template-columns:
          minmax(min-content, min-content)   // первая — строго по контенту
          repeat(var(--col-count, 1), minmax(min-content, var(--col-max)))
          1fr;
      }
      .advanced-editor__table-cell:first-child {
        text-align: center;
        padding-left: 12px;
        padding-right: 12px;
        white-space: nowrap;
      }
    }

    // ── все колонки равной ширины ──
    &.same-column {
      .advanced-editor__table-row {
        grid-template-columns: repeat(var(--col-count, 1), minmax(0, 1fr));
      }
    }

    // ── липкая шапка ──
    &.sticky-head {
      .advanced-editor__table-scroll {
        overflow-y: auto;
      }
      .advanced-editor__table-head {
        position: sticky;
        top: 0;
        z-index: 2;
        background: var(--thead-bg);
      }
    }

    // ── липкая первая колонка ──
    &.sticky-first {
      .advanced-editor__table-cell:first-child {
        position: sticky;
        left: 0;
        z-index: 1;
        background: #fff;
      }
      .advanced-editor__table-head .advanced-editor__table-cell:first-child {
        z-index: 3;
        background: var(--thead-bg);
      }
    }

    // ── зебра ──
    &.zebra {
      .advanced-editor__table-body .advanced-editor__table-row:nth-child(even) {
        background-color: #FAFAFA;
      }
    }
  }

  // ──────────────────────────────────────────────
  //  Скролл-контейнер
  // ──────────────────────────────────────────────
  &__table-scroll {
    overflow-x: auto;
    max-height: 400px;
    overflow-y: auto;

    // ── отступ скролла от таблицы 12px, только если скролл реально есть ──
    //  используем padding на контейнере + negative-margin на таблице,
    //  чтобы «пустого» отступа не было, когда скролла нет
    padding-bottom: 0;

    @-moz-document url-prefix() {
      scrollbar-color: #dedede transparent;
      scrollbar-width: thin;
    }

    &::-webkit-scrollbar { width: 5px; height: 5px; }
    &::-webkit-scrollbar-button { width: 0; height: 0; }
    &::-webkit-scrollbar-thumb {
      background: #dedede;
      border: 0 transparent;
      border-radius: 50px;
    }
    &::-webkit-scrollbar-track {
      background: transparent;
      border: 0 transparent;
      border-radius: 50px;
    }
    &::-webkit-scrollbar-corner { background: transparent; }

    // ── «дырка» между таблицей и скроллом только при наличии скролла ──
    //  scrollbar-gutter резервирует место под скролл всегда, поэтому
    //  отступ делаем через padding на самом скролле и компенсируем
    //  отрицательным margin у таблицы. Если скролла нет — padding
    //  визуально не проявится, т.к. таблица растянется.
    //  Практичнее: padding-right/bottom на скролле + margin-right/bottom
    //  на таблице = 0, но при появлении скролла он «отодвигается».
    //  Проще всего — использовать scrollbar-gutter: stable both-edges
    //  и padding. Ниже — рабочий вариант:
  }

  // ──────────────────────────────────────────────
  //  Сама «таблица»
  // ──────────────────────────────────────────────
  &__table {
    --col-min: 140px;
    --col-max: 409px;
    --col-count: 3;                       // ← задайте число колонок в EJS/инлайн-стилем
    --cell-border: 1px solid #EFEFEF;
    --thead-bg: #F4F4F4;

    display: flex;
    flex-direction: column;
    width: 100%;
    min-width: max-content;               // не сжиматься меньше контента → скролл
    font-family: $font-family;
    @include style-common-text;

    :is(p, ul, ol) { margin: 0; }

    a {
      font-size: inherit;
      font-family: inherit;
      color: inherit;
      line-height: inherit;
    }
  }

  // ──────────────────────────────────────────────
  //  Строка
  // ──────────────────────────────────────────────
  &__table-row {
    display: grid;
    // базовая раскладка: непоследние — по контенту, но ≤ col-max; последняя — 1fr
    grid-template-columns:
      repeat(calc(var(--col-count) - 1), minmax(min-content, var(--col-max)))
      1fr;
    // рамка снизу у всей строки (чтобы не дублировать на ячейках)
    border-bottom: var(--cell-border);

    &:last-child {
      border-bottom: none;
    }
  }

  // ──────────────────────────────────────────────
  //  Ячейка
  // ──────────────────────────────────────────────

    border-right: var(--cell-border);

      border-right: none;
    }

    & > *:first-child { margin-top: 0; }
  }

  // ──────────────────────────────────────────────
  //  Шапка

    .advanced-editor__table-cell {

        font-family: inherit !important;
        color: inherit !important;
      }
    }
  }
