<div v-else class="btn_s black" @click.prevent.stop="sendingResult" ref="btnSend" :class="preload_btn === true ? 'submitting' : ''">Проверить</div>


<form ref="captchaContainer" class="status-kadastr-block__form" @submit.prevent="handlerSendForm">
<input @input="onInput" />

<button type="submit" :class="isDisabled ? 'status-kadastr-block__submit_disabled' : ''" class="status-kadastr-block__submit">
        Проверить
      </button>

onSubmit(token){
      if(this.value.length === 0) return
      let key = conf.smartcaptcha_key
      this.isDisabled = true
      axios.post('', {
        kadastr: this.value,
        'smart-token': token,
      },{
        headers: {
          'Content-Type': 'application/json',
          'X-Bitrix-Csrf-Token': window.BX.bitrix_sessid(),
          Accept: 'application/json',
        },
      })
        .then((response) => {
          this.response = response.data.result
        })
        .catch((error) => {
          const result = {
            "status": '1',
            "message": error.response.data.description
          }
          this.response = result
        })
        .finally(() => {
          this.isDisabled = false
          // Сбрасываю капчу через миксин (без удаления DOM, чтобы можно было отправить снова)
          this.resetCaptcha(false);
        });
    },
