methods: {
  initSwiper() {
    this.destroySwiper();

    if (!this.$refs.swiperContainer || this.flatSlides.length === 0) {
      return;
    }

    this.swiper = new Swiper(this.$refs.swiperContainer, {
      slidesPerView: 'auto',
      spaceBetween: 16,
      watchOverflow: true,
      observer: true,
      observeParents: true,
      grabCursor: true,
      keyboard: {
        enabled: true
      },
      on: {
        slideChange: (swiper) => {
          this.activeIndex = swiper.activeIndex;
          this.updateControls(swiper);
        },
        progress: (swiper) => {
          this.updateControls(swiper);
        },
        resize: (swiper) => {
          swiper.update();
          this.updateControls(swiper);
        },
        imagesReady: (swiper) => {
          swiper.update();
          this.updateControls(swiper);
        }
      }
    });

    this.activeIndex = this.swiper.activeIndex || 0;
    this.updateControls(this.swiper);

    this.$nextTick(() => {
      if (this.swiper) {
        this.updateControls(this.swiper);
      }
    });
  },

  destroySwiper() {
    if (this.swiper && typeof this.swiper.destroy === 'function') {
      this.swiper.destroy(true, true);
    }
    this.swiper = null;
  },

  updateControls(swiper) {
    if (!swiper) return;

    this.isBeginning = Boolean(swiper.isBeginning);

    const lastRealSlideIndex = this.flatSlides.length - 1;
    const activeSlide = this.flatSlides[swiper.activeIndex];
    const lastSlide = this.flatSlides[lastRealSlideIndex];

    if (swiper.activeIndex >= lastRealSlideIndex) {
      this.isEnd = true;
      return;
    }

    if (activeSlide && lastSlide && activeSlide.year !== lastSlide.year) {
      this.isEnd = false;
      return;
    }

    const lastRealSlide = swiper.slides[lastRealSlideIndex];

    if (lastRealSlide) {
      const swiperRect = swiper.el.getBoundingClientRect();
      const slideRect = lastRealSlide.getBoundingClientRect();

      if (slideRect.right <= swiperRect.right + 1) {
        this.isEnd = true;
      } else {
        this.isEnd = false;
      }
    } else {
      this.isEnd = false;
    }
  },

  nextSlide() {
    if (!this.swiper) return;
    if (!this.isEnd) {
      this.swiper.slideNext();
    }
  },

  prevSlide() {
    if (!this.swiper) return;
    this.swiper.slidePrev();
  }
}
