inputValue(el) {
  const element = el.currentTarget;
  let value = parseInt(element.value.replace(/\s/g, '')) || 0;

  // === ДОБАВЛЕНО: Защита от ухода значения ниже минимального порога ===
  if (value < this.stgMin) {
    value = this.stgMin;
    // Визуально возвращаем минимальное значение в инпут с пробелами
    this.dataField = this.stgMin.toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ');
  }
  // ====================================================================

  this.dataFieldForCalculation = value;
  store.dispatch('ActionRealPriceAppartment', value);

  this.realtySlider.set(value);
},

lossFocus(el) {
  const element = el.currentTarget;
  const cleanValue = parseInt(element.value.replace(/\s/g, ''));
  
  // Если поле пустое или значение меньше минимального
  if (element.value === '' || cleanValue < this.stgMin) {
    this.dataField = this.stgMin.toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ');
    this.realtySlider.set(this.stgMin);
    // ИСПРАВЛЕНО: передаем просто число, а не массив [число, true], который мог ломать store
    store.dispatch('ActionRealPriceAppartment', this.stgMin);
  }
},

// ... внутри watch компонента component-initial-payment.vue

watch: {
  limit_init_fee() {
    // === ДОБАВЛЕНО: Защита от ошибки noUiSlider (min и max не могут быть равны) ===
    // Если лимит меньше или равен минимуму, делаем max искусственно на 1 шаг больше
    const safeMax = this.limit_init_fee <= this.stgMin 
      ? this.stgMin + this.stepApartment 
      : this.limit_init_fee;
    // ============================================================================

    this.realtySlider.updateOptions({
      range: {
        min: this.stgMin,
        max: safeMax
      }
    });

    // Дополнительная страховка: если текущее положение слайдера вдруг стало больше лимита, сбрасываем его
    if (parseInt(this.realtySlider.get()) > this.limit_init_fee) {
      this.realtySlider.set(this.limit_init_fee);
    }
  },
  
  init_fee() {
    if (this.checkbox_status === true) {
      this.realtySlider.set(this.init_fee);
    }
  },
  
  dataField() {
    this.calculationPageLoad();
  }
},
