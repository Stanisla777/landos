// eslint-disable-next-line import/prefer-default-export
export function debounce(func, delay) {
  let timeoutId;
  // eslint-disable-next-line func-names
  const debounced = function (...args) {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => func.apply(this, args), delay);
  };

  // Добавляем возможность отмены
  // eslint-disable-next-line func-names
  debounced.cancel = function () {
    clearTimeout(timeoutId);
  };

  return debounced;
}


<template>


  <div class="calculator__row">
    <div v-if="check_calculate_together" class="calculator__row-title-container m-b-24">
      <p class="calculator__row-title">Ваши данные</p>
    </div>
    <div :class="!check_calculate_together ? 'm-b-24' : ''" class="calculator__row-title-container">
      <p v-if="!check_calculate_together" class="calculator__row-title">Доходы</p>
      <p v-else class="calculator__row-title small">Доход</p>
    </div>
    <div class="calc-tax-deduc-new__block-subtitle small-grey">
      <p>За сколько лет хотите вернуть налог?</p>
    </div>
    <!--Табы за какой период 1год 2года 3года для мужа-->
    <componen-tabs-time-interval :view="'man'"/>

    <!--Чекбокс есть районные коэффициенты или северная надбавка-->
    <div class="calculator__row small-margin">
      <div class="input-story-container">
        <DsCheckbox
          v-model="dataFieldCheckbox"
          :label="'Есть районный коэффициент или северная надбавка '"
          :show-text="true"
          :size="'large'"
          :disabled="false"
          :indeterminate="false"
          :error="false"
          :error-text="'Обязательное поле'"
          :dark-mode="false"
          :locked="false"
        />
      </div>
    </div>

    <!--Доходы по годам, когда не выбран чекбокс Есть районный коэффициент-->
    <div v-if="!check_district_coefficient" class="calculator__row small-margin">
      <div class="calc-tax-deduc-new__row-grid-auto-column">
        <div v-for="(item,index) in visibleBlockIncomePeriod" :key="item.id"  class="input-story-container">
          <DsInput
            class="height-56"
            v-model="fieldsIncomeMan[item.modelInput]"
            :label="'Среднемесячный доход в ' + (data_year + index) + ' году, ₽'"
            preset="amount"
            labelPosition="inside"
            :unit="''"
            :maxlength="100"
            :autocomplete="'off'"
          />
        </div>
      </div>
    </div>

    <!--Доходы по годам, когда выбран чекбокс Есть районный коэффициент-->
    <div v-if="check_district_coefficient" class="calculator__row small-margin">
      <componentincome-with-additional-coefficient :view="'man'" />
    </div>

    <!--Предыдущие вычеты-->
    <div class="calculator__row small-margin">
      <div class="calculator__row-title-container">
        <p class="calculator__row-title">Предыдущие вычеты</p>
      </div>
      <div class="calc-tax-deduc-new__row-grid-auto-column">
        <div class="input-story-container">
          <DsInput
            class="height-56"
            v-model="previousPayments.deduction_purchase"
            :label="'Вычет за покупку недвижимости, ₽'"
            preset="amount"
            labelPosition="inside"
            :unit="''"
            :maxlength="100"
            :autocomplete="'off'"
          />
        </div>

        <div class="input-story-container">
          <DsInput
            class="height-56"
            v-model="previousPayments.deduction_interest"
            :label="'Вычет за проценты по ипотеке, ₽'"
            preset="amount"
            labelPosition="inside"
            :unit="''"
            :maxlength="100"
            :autocomplete="'off'"
          />
        </div>

        <div class="input-story-container">
          <DsInput
            class="height-56"
            v-model="previousPayments.other_deductions"
            :placeholder="'Другие вычеты, ₽'"
            :label="'Другие вычеты, ₽'"
            preset="amount"
            labelPosition="inside"
            :unit="''"
            :maxlength="100"
            :autocomplete="'off'"
          />
        </div>

      </div>
    </div>

  </div>

</template>
<script>
import store from "../store";
import { DsInput } from '@ringsites/ui-components/DsInput'
import { DsCheckbox } from '@ringsites/ui-components'
import ComponentincomeWithAdditionalCoefficient from './component-income-with-additional-coefficient';
import ComponenTabsTimeInterval from './component-tabs-time-interval';
import {debounce} from "../../../modules/debounce";


export default {
  name: 'component-income-man',
  mixins: [],
  data(){
    return {
      //чекбокс районный коэффициент или северная надбавка
      dataFieldCheckbox:false,
      //поля средний доход по годам, заполняется в created
      fieldsIncomeMan: {},
      isUpdatingFromStore:false, //флаг
      //предыдущие выплаты
      previousPayments : {
        deduction_purchase:'', //вычет за покупку
        deduction_interest:'', //вычет за проценты
        other_deductions:'', //другие вычеты
      }
    }
  },
  methods:{


    //заполнеия поля fieldsIncomeMan при загрузке из store
    initializeFieldsIncomeFromStore() {
      const fields = {}
      // Проходим по ВСЕМ табам (не только видимым, чтобы сохранить данные всех полей)
      this.array_all_income_period.forEach(tab => {
        fields[tab.modelInput] = tab.value || ''
      })
      this.fieldsIncomeMan = fields
    },

  },

  computed:{

    check_calculate_together() {
      return store.getters.CHECKCALCULATETOGETHER;
    },

    data_year() {
      return store.getters.DATAYEAR;
    },



    array_all_income_period(){
      return store.getters.ARRAYALLINCOMEPERIODMAN
    },

    visibleBlockIncomePeriod() {
      // Ищем индекс первого элемента, у которого isActive === true
      const activeIndex = this.array_all_income_period.findIndex(tab => tab.isActive === true)

      // Если такой элемент не найден — возвращаю пустой массив
      if (activeIndex === -1) return []

      // Беру все элементы от начала до активного включительно
      return this.array_all_income_period.slice(0, activeIndex + 1)
    },
    check_district_coefficient(){
      return store.getters.CHECKDISTRICTCOEFFICIENTMAN
    },

  },
  watch: {
    // 1. Следим за изменениями в Store (например, очистка полей при клике на таб) в полях с доходами
    array_all_income_period: {
      handler() {
        this.isUpdatingFromStore = true;

        // ⭐ ГЛАВНОЕ ИСПРАВЛЕНИЕ: Отменяем любой запланированный вызов debounce!
        // Если пользователь печатал, и мы кликнули на таб, старый таймер уничтожается.
        if (this.debouncedIncomeUpdate) {
          this.debouncedIncomeUpdate.cancel();
        }

        this.initializeFieldsIncomeFromStore();

        this.$nextTick(() => {
          this.isUpdatingFromStore = false;
        });
      },
      deep: true
    },

    // 2. Следим за изменениями в локальных полях доходов (когда печатает пользователь)
    fieldsIncomeMan: {
      handler(newFields) {
        // Просто передаем данные в нашу debounce-функцию
        this.debouncedIncomeUpdate(newFields);
      },
      deep: true,
      immediate: false
    },
    //Изменение чекбокса районный коэффициент
    dataFieldCheckbox() {
      store.dispatch('ActionCheckDistrictCoefficientMan', this.dataFieldCheckbox);
    },

    //изменения в предыдущих вычетах
    previousPayments: {
      handler: debounce(function (newValue) {
        // newValue выглядит так: { deduction_purchase: '1 000 000', deduction_interest: '500 000' }

        // 1. Создаем новый объект с очищенными числами
        const cleanedPayments = {};
        for (const key in newValue) {
          cleanedPayments[key] = parseInt(String(newValue[key]).replace(/\s/g, ''), 10) || 0;
        }

        // 2. Показываем лоадер
        store.dispatch('ActionShowLoader', true);

        // 3. Отправляем очищенный объект в Store
        store.dispatch('ActionInputPreviousPaymentsMan', cleanedPayments)
          .then(() => {
            store.dispatch('ActionShowLoader', false);
          })
          .catch(() => {
            store.dispatch('ActionShowLoader', false);
          });

      }, 500),

      deep: true,
      immediate: false
    },
  },
  mounted(){


  },
  created() {
    this.initializeFieldsIncomeFromStore();
    this.debouncedIncomeUpdate = debounce((newFields) => {
      // Эта проверка всё ещё нужна как последний рубеж обороны
      if (this.isUpdatingFromStore) return;

      const cleanedFields = {};
      for (const key in newFields) {
        cleanedFields[key] = parseInt(String(newFields[key]).replace(/\s/g, ''), 10) || 0;
      }

      store.dispatch('ActionShowLoader', true);

      // Используем .finally() вместо .then().catch() - это чище и гарантирует выполнение
      store.dispatch('ActionInputIncomeMan', cleanedFields)
        .finally(() => {
          store.dispatch('ActionShowLoader', false);
        });
    }, 500);
  },
  beforeUnmount() {
    if (this.debouncedIncomeUpdate) {
      this.debouncedIncomeUpdate.cancel();
    }
  },
  components: {
    DsInput,
    DsCheckbox,
    ComponentincomeWithAdditionalCoefficient,
    ComponenTabsTimeInterval
  },



};
</script>
