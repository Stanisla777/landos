import { watch, nextTick, unref } from 'vue';

export function useMaxValue(modelRef, maxValueRef, inputRef = null) {
  
  const applyLimit = async () => {
    const currentVal = modelRef.value;
    
    if (!currentVal && currentVal !== '0') return;

    const maxVal = unref(maxValueRef);
    
    if (maxVal === null || maxVal === undefined || maxVal === '') return;

    const cleanStr = String(currentVal).replace(/\s/g, '');
    const num = parseFloat(cleanStr);

    if (isNaN(num) || num <= maxVal) {
      return;
    }

    const maxFormatted = Number(maxVal).toLocaleString('ru-RU');
    
    if (currentVal === maxFormatted) return;

    // ⭐ Применяем то же решение, которое сработало в компоненте
    modelRef.value = maxFormatted;

    // Ждем следующего тика, чтобы Vue обновил реактивные данные
    await nextTick();

    // Работаем напрямую с активным DOM-элементом, минуя сложные внутренние механизмы компонента
    const activeInput = document.activeElement;
    
    if (activeInput && activeInput.tagName === 'INPUT' && typeof activeInput.setSelectionRange === 'function') {
      // Принудительно синхронизируем значение в DOM (на случай, если маска DsInput "отстала")
      activeInput.value = maxFormatted;
      activeInput.focus();
      
      // Ждем отрисовки кадра и ставим курсор в конец
      requestAnimationFrame(() => {
        activeInput.setSelectionRange(maxFormatted.length, maxFormatted.length);
      });
    }
  };

  // Следим за вводом пользователя
  watch(modelRef, () => {
    applyLimit();
  });

  // Следим за изменением самого лимита
  watch(
    () => unref(maxValueRef),
    () => {
      applyLimit();
    }
  );
}
