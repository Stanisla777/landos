&.hidden{
    max-height: 0;
    overflow: hidden;
    opacity: 0;
    display: block;
    padding-top: 0;
    padding-bottom: 0;
    border-bottom-color: transparent;
    
    // ВАЖНО: добавляем padding-bottom в transition
    transition: 
      max-height 0.5s ease, 
      opacity 0.5s ease, 
      padding-bottom 0.5s ease,
      padding-top 0.5s ease;
  }

-------------

btn.addEventListener('click', () => {
  const hiddenItems = parent.querySelectorAll('.js-accordion-parent.hidden');

  hiddenItems.forEach(item => {
    // 1. Сначала измеряем ПРАВИЛЬНУЮ высоту (с padding из CSS)
    // Временно убираем hidden стили для измерения
    item.style.transition = 'none';
    item.style.maxHeight = 'none';
    item.style.opacity = '1';
    item.style.paddingBottom = ''; // Берём из CSS (24px)
    item.style.paddingTop = ''; // Берём из CSS
    
    const originalHeight = item.scrollHeight;
    
    // 2. Возвращаем в начальное состояние для анимации
    item.style.maxHeight = '0';
    item.style.opacity = '0';
    item.style.paddingBottom = '0';
    item.style.paddingTop = '0';
    
    // 3. Убираем класс
    item.classList.remove('hidden');
    
    // 4. Принудительный reflow
    void item.offsetHeight;
    
    // 5. Запускаем анимацию
    requestAnimationFrame(() => {
      item.style.transition = 'max-height 0.5s ease, opacity 0.5s ease, padding-bottom 0.5s ease, padding-top 0.5s ease';
      item.style.maxHeight = `${originalHeight}px`;
      item.style.opacity = '1';
      item.style.paddingBottom = ''; // Плавно возвращается к CSS значению
      item.style.paddingTop = ''; // Плавно возвращается к CSS значению
    });

    // 6. Очистка после анимации
    item.addEventListener('transitionend', function handler(e) {
      if (e.propertyName !== 'max-height') return;
      item.style.transition = '';
      item.style.maxHeight = '';
      item.style.opacity = '';
      item.style.paddingBottom = '';
      item.style.paddingTop = '';
      item.removeEventListener('transitionend', handler);
    });
  });

  btn.classList.add('btn-accordion-hidden');
});
