// Извлечение числа и исходного маркера из текста
extractNumberAndMarker(text) {
  if (!text) return { number: null, originalMarker: '' };
  
  // Очищаем от HTML-тегов перед анализом
  const cleaned = String(text).trim().replace(/<[^>]*>/g, '').trim();
  
  // Если после очистки ничего не осталось
  if (!cleaned) return { number: null, originalMarker: '' };
  
  // Извлекаем числовую часть и маркер
  // Числовая часть: цифры, пробелы, точки, запятые, минус
  const match = cleaned.match(/^([\d\s.,-]+)\s*(.*)$/);
  
  if (!match) return { number: null, originalMarker: cleaned };
  
  const numberPart = match[1].trim();
  const originalMarker = match[2].trim();
  
  // Парсим число
  const cleanedNumber = numberPart.replace(/\s/g, '').replace(/[^\d.,-]/g, '');
  if (!cleanedNumber || isNaN(cleanedNumber)) {
    return { number: null, originalMarker: cleaned };
  }
  
  const number = parseFloat(cleanedNumber.replace(',', '.'));
  if (isNaN(number)) {
    return { number: null, originalMarker: cleaned };
  }
  
  return { number, originalMarker };
}
