 limit_init_fee(newVal){
        // Защита от ошибки noUiSlider: min и max не могут быть равны
        const min = this.stgMin;
        const max = newVal > min ? newVal : min + this.stepApartment;
        
        this.realtySlider.updateOptions({
            range: {
                min: min,
                max: max
            }
        });
    },
