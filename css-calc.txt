safeJsonParse(value) {
      if (!value) return {};
      if (typeof value === 'object' && value !== null && !Array.isArray(value)) {
        return value;
      }
      if (typeof value === 'string') {
        try {
          return JSON.parse(value);
        } catch (e) {
          console.warn('Ошибка парсинга JSON:', e);
          return {};
        }
      }
      return {};
    }
