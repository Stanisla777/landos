
function formatIncomeForBackend(incomeArray) {
  if (!Array.isArray(incomeArray)) return [];

  // Вспомогательная функция для безопасного извлечения значения коэффициента
  const extractCoeffValue = (coeffObj) => {
    if (!coeffObj || !coeffObj.value) return "";
    
    // Если value - это массив (как в нашем случае: [{id: 1, label: '0.1'}])
    if (Array.isArray(coeffObj.value) && coeffObj.value.length > 0) {
      // Берем label (или value) первого элемента и превращаем в строку
      return String(coeffObj.value[0].label || coeffObj.value[0].value || "");
    }
    
    // Если вдруг value уже строка или число
    return String(coeffObj.value);
  };

  return incomeArray.map(item => ({
    isActive: item.isActive,
    isShow: item.isShow,
    value: item.value,
    additionalCoefficient: {
      districtCoefficient: extractCoeffValue(item.additionalCoefficient?.districtCoefficient),
      northernAllowance: extractCoeffValue(item.additionalCoefficient?.northernAllowance)
    }
  }));
}

state.answers_to_send.incomeMan = formatIncomeForBackend(state.array_all_income_period_man);
state.answers_to_send.incomeWife = formatIncomeForBackend(state.array_all_income_period_wife);
