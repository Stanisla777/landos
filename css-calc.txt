function iMaskMail() {
  const input_status = document.querySelectorAll('.js--validate-mask-email');
  const maskOptions = {
    mask: /^[a-z0-9@.!#$%&'*+\\/=?^_"`{|}~-]*$/i
    // enum: ' '
  };
  for (const item of input_status) {
    // Уничтожаю старую маску, если есть
    if (maskInstances.has(item)) {
      maskInstances.get(item).destroy();
      maskInstances.delete(item);
    }
    // Создаю новую и сохраняем ссылку
    const mask = IMask(item, maskOptions);
    maskInstances.set(item, mask);
  }
}
