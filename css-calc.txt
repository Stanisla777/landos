// 🔹 РОВНО 5 БЛОКОВ: 
  // 1-й блок (лев) = высота 1+2 блоков (прав)
  // 2-й блок (лев) = высота 3-го блока (прав)
  &:has(> .item:nth-child(5):last-child) {
    grid-template-rows: repeat(3, auto);
    > .item:nth-child(1) { grid-row: 1 / 3; }   // Занимает строки 1 и 2
    > .item:nth-child(2) { grid-row: 3 / 4; }   // Занимает строку 3
    > .item:nth-child(n+3) { grid-column: 2; }  // Автоматически встанут в строки 1, 2, 3
  }

  // 🔹 РОВНО 6 БЛОКОВ:
  // 1-й блок (лев) = высота 1,2,3 блоков (прав)
  // 2-й блок (лев) = высота 4-го блока (прав)
  // (в вашем ТЗ было повторение "первый блок...". Сделал логичный сплит 3+1. 
  //  Если нужно 2+2, см. комментарий ниже 👇)
  &:has(> .item:nth-child(6):last-child) {
    grid-template-rows: repeat(4, auto);
    > .item:nth-child(1) { grid-row: 1 / 4; }   // Занимает строки 1, 2, 3
    > .item:nth-child(2) { grid-row: 4 / 5; }   // Занимает строку 4
    > .item:nth-child(n+3) { grid-column: 2; }  // Автоматически встанут в строки 1, 2, 3, 4
  }
