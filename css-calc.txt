:model-value="dataField"
@update:model-value="onModelUpdate"

methods: {
  onModelUpdate(value) {
    this.isUserInteracted = true;

    // Крестик или внешняя очистка — пробрасываем пустое значение как есть,
    // без валидации (иначе min/max вернёт '1' или '0.1' и поле не очистится!)
    if (value === '' || value === null || value === undefined) {
      this.dataField = '';
      this.$emit('update:modelValue', '');
      return;
    }

    let val = String(value).replace(/[^\d.,]/g, '').replace(',', '.');
    const parts = val.split('.');

    // Ограничение до 2 знаков после точки
    if (parts.length === 2 && parts[1].length > 2) {
      val = parts[0] + '.' + parts[1].substring(0, 2);
    }

    const num = parseFloat(val);

    if (!isNaN(num)) {
      if (this.view === 'district') {
        if (num < 1) val = '1';
        else if (num >= 3) val = '3';
      } else {
        if (num < 0.1) val = '0.1';
        else if (num >= 100) val = '100';
      }
    }

    this.dataField = val;
    this.$emit('update:modelValue', this.dataField);
  }
}
