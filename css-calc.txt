import { createStore } from 'vuex';

export default createStore({
  state: {
    // ... ваши существующие поля ...
  },
  getters: {
    // ... ваши существующие геттеры ...
    
    // ДОБАВЛЕНО: Сырые числовые значения для удобной синхронизации компонентов
    MATERNALCAPITAL_RAW(state) {
      return state.maternal_capital;
    },
    PERSONALFUNDS_RAW(state) {
      return state.personal_funds;
    },
  },
  mutations: {
    mutationPriceAppartment(state, received_perem) {
      // 1. Устанавливаем цену (она уже приходит >= 625000 из компонента, но добавляем страховку)
      let price = parseInt(received_perem);
      if (isNaN(price) || price === 0) {
        price = 625000; 
      }
      state.appartment_price = price;

      // 2. Ограничиваем материнский капитал, если он стал больше стоимости недвижимости
      if (state.maternal_capital > state.appartment_price) {
        state.maternal_capital = state.appartment_price;
      }

      // 3. Ограничиваем первоначальный взнос стоимостью недвижимости
      // (Если ПВ был 2 000 000, а цена стала 625 000, ПВ станет 625 000. Если ПВ был 200 000, он останется 200 000)
      if (state.initialfee > state.appartment_price) {
        state.initialfee = state.appartment_price;
      }

      // 4. КЛЮЧЕВОЙ МОМЕНТ: Синхронизируем личные средства
      // Личные средства всегда равны: Первоначальный взнос минус Материнский капитал
      state.personal_funds = state.initialfee - state.maternal_capital;

      // 5. Защита от отрицательных личных средств (на случай форс-мажора)
      if (state.personal_funds < 0) {
        state.personal_funds = 0;
        // Если мат. капитал вдруг оказался больше текущего ПВ, выравниваем ПВ по мат. капиталу
        state.initialfee = state.maternal_capital;
      }

      // 6. Обновляем отображаемые значения (с пробелами и знаком рубля) для всех затронутых полей
      state.initialfee_show = state.initialfee.toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ') + ' ₽';
      state.maternal_capital_show = state.maternal_capital.toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ') + ' ₽';
      state.personal_funds_show = state.personal_funds.toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ') + ' ₽';
    },

    // ... остальные мутации без изменений ...

component-maternal-capital.vue

  computed: {
    limitmaterialcapital() {
      return store.getters.LIMITMATERIALCAPITAL;
    },
    maternal_capital_show() {
      return store.getters.MATERNALCAPITALSHOW;
    },
    // ДОБАВЛЕНО: сырое значение для отслеживания изменений
    maternal_capital_raw() {
      return store.getters.MATERNALCAPITAL_RAW;
    }
  },
  watch: {
    // ДОБАВЛЕНО: Синхронизация инпута и слайдера, если значение изменилось извне (из store)
    maternal_capital_raw(newVal) {
      const currentVal = parseInt(String(this.dataField).replace(/\s/g, '')) || 0;
      if (currentVal !== newVal) {
        this.dataField = newVal.toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ');
        this.realtySlider.set(newVal);
      }
    },
    limitmaterialcapital() {
      if (this.limitmaterialcapital < this.maximumMaternityCapital && this.limitmaterialcapital !== 0) {
        this.realtySlider.updateOptions({
          range: {
            min: this.stgMin,
            max: this.limitmaterialcapital
          }
        });
      } else if (this.limitmaterialcapital >= this.maximumMaternityCapital) {
        this.realtySlider.updateOptions({
          range: {
            min: this.stgMin,
            max: this.maximumMaternityCapital
          }
        });
      } else if (this.limitmaterialcapital === 0) {
        this.$refs.realtyInput.classList.add('unactive');
      }
      
      if (this.limitmaterialcapital !== 0 && this.$refs.realtyInput.classList.contains('unactive')) {
        this.$refs.realtyInput.classList.remove('unactive');
      }
    },
    dataField() {
      this.calculationPageLoad();
    }
  },

  watch: {
    // ДОБАВЛЕНО: Синхронизация инпута и слайдера при изменении из store

        this.$refs.realtyInput.classList.add('unactive');
