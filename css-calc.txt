        return typeof window.beesenderchatconfig !== 'undefined'
            && window.beesenderchatconfig !== null
            && window.beesenderchatconfig.srcUrl
            && window.beesenderchatconfig.connectorUrl;
    }

    function startWidget() {

        });

        /* === ОРИГИНАЛЬНЫЙ КОД ЗАКАНЧИВАЕТСЯ === */
    } // конец startWidget

    function waitForConfig() {
        if (isConfigReady()) {
            startWidget();
        } else if (waited < MAX_WAIT) {
            waited += INTERVAL;
            setTimeout(waitForConfig, INTERVAL);
        } else {
            console.error('[beesender-debug] Config not found after ' + MAX_WAIT + 'ms, starting widget anyway');
            startWidget();
        }
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', waitForConfig);
    } else {
        waitForConfig();
    }
})();
