data() {
  return {
    captchaId: null,
    // Генерируем действительно уникальный ID для каждого экземпляра
    containerId: `yandex-captcha-${Math.random().toString(36).substr(2, 9)}`,
    isMobile: null,
    flag: 0,
  };
},

initCaptcha() {
  if (this.captchaId !== null) return;
  if (!window.smartCaptcha) {
    console.warn('[SmartCaptcha] Скрипт не загружен');
    return;
  }

  const container = document.getElementById(this.containerId);
  if (!container) return;

  // 🔥 КЛЮЧЕВОЙ ШАГ: Полная очистка контейнера перед рендером.
  // Это убивает любые "зомби" React-деревья, которые могли там остаться,
  // и предотвращает ошибки гидратации #418 / #423.
  container.innerHTML = ''; 

  try {
    this.captchaId = window.smartCaptcha.render(this.containerId, {
      sitekey: conf?.smartcaptcha_key,
      invisible: true,
      callback: (token) => {
        this.sendingResultToApi(token);
      },
    });
    console.log('[SmartCaptcha] Успешно, ID виджета:', this.captchaId);
  } catch (e) {
    console.error('[SmartCaptcha] Ошибка render:', e);
  }
},

beforeUnmount() { 
  if (this.captchaId !== null && window.smartCaptcha) {
    try {
      window.smartCaptcha.reset(this.captchaId);
    } catch (e) {
      // Игнорируем
    }
    this.captchaId = null;
  }
},

<div :id="containerId" class="captcha-element" v-once></div>
