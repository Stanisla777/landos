const maskInstances = new Map();

function iMaskMail() {
  const input_status = document.querySelectorAll('.js--validate-mask-email');
  const maskOptions = {
    mask: /^[a-z0-9@.!#$%&'*+\\/=?^_"`{|}~-]*$/i
    // enum: ' '
  };
  for (const item of input_status) {
    // Уничтожаю старую маску, если есть
    if (maskInstances.has(item)) {
      maskInstances.get(item).destroy();
      maskInstances.delete(item);
    }
    // Создаю новую и сохраняем ссылку
    const mask = IMask(item, maskOptions);
    maskInstances.set(item, mask);

    // Обработчик изменения: добавляем/убираем класс у родителя
    const toggleActiveClass = () => {
      const parent = item.closest('.js--story-input-field');
      if (!parent) return;

      // mask.value — актуальное значение маски (не item.value!)
      if (mask.value.length > 0) {
        parent.classList.add('active');
      } else {
        parent.classList.remove('active');
      }
    };

    // Подписываемся на событие 'accept' (срабатывает при каждом изменении)
    mask.on('accept', toggleActiveClass);

    // Проверяю начальное состояние (на случай автозаполнения браузера)
    toggleActiveClass();


  }
}

function iMaskReset(form = null){
  const selector = '.js--validate-mask-tel, .js--validate-mask-name, .js--validate-mask-email, .js--validate-mask-message';
  const inputs = form
    ? form.querySelectorAll(selector)
    : document.querySelectorAll(selector);

  for (const input of inputs) {
    if (maskInstances.has(input)) {
      const mask = maskInstances.get(input);
      mask.updateValue();
      if (mask.value !== '') {
        mask.value = '';
      }
    }
  }
}
