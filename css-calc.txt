  methods: {
    parseHistoryArray(data) {
      if (!data) return;
      // Если уже объект — используем как есть
      if (typeof data === 'object' && data !== null && !Array.isArray(data)) {
        this.objectHistory = data
        return;

    debounce(fn, delay) {
      let timer = null;
      return (...args) => {
        clearTimeout(timer);
        timer = setTimeout(() => fn.apply(this, args), delay);
      };
    },

    initSwiper() {
      this.destroySwiper();

      if (!this.$refs.swiperContainer || this.flatSlides.length === 0) {
        return;
      }

      this.swiper = new Swiper(this.$refs.swiperContainer, {
        slidesPerView: 'auto',
        spaceBetween: 16,
        watchOverflow: true,
        // observer: true,

            // Обновляем после завершения анимации,
            // чтобы getBoundingClientRect вернул финальные значения
            this.updateControls(swiper);
          },
          imagesReady: (swiper) => {
            swiper.update();
            this.updateControls(swiper);
          }
        }
      });

      this.activeIndex = this.swiper.activeIndex || 0;
      this.updateControls(this.swiper);

      this.$nextTick(() => {
        if (this.swiper) {
          this.updateControls(this.swiper);
        }
      });
    },

    destroySwiper() {
      if (this.swiper && typeof this.swiper.destroy === 'function') {
        this.swiper.destroy(true, true);
      }
      this.swiper = null;
    },

    updateControls(swiper) {
      if (!swiper) return;

      this.isBeginning = Boolean(swiper.isBeginning);

      const lastRealSlideIndex = this.flatSlides.length - 1;
      const activeSlide = this.flatSlides[swiper.activeIndex];
      const lastSlide = this.flatSlides[lastRealSlideIndex];

      if (swiper.activeIndex >= lastRealSlideIndex) {
        this.isEnd = true;
      } else if (activeSlide && lastSlide && activeSlide.year !== lastSlide.year) {
        this.isEnd = false;
      } else {
        const lastRealSlide = swiper.slides[lastRealSlideIndex];

        if (lastRealSlide) {
          const swiperRect = swiper.el.getBoundingClientRect();
          const slideRect = lastRealSlide.getBoundingClientRect();

          this.isEnd = slideRect.right <= swiperRect.right + 1;
        } else {
          this.isEnd = false;
        }
      }

      // Блокируем свайп мышкой и пальцем
      swiper.allowSlideNext = !this.isEnd;
      swiper.allowSlidePrev = !this.isBeginning;
      console.log('this.isEnd');
      console.log(this.isEnd);
    },

    nextSlide() {
      if (!this.swiper) return;
      if (!this.isEnd) {
        this.swiper.slideNext();
      }
    },

    prevSlide() {
      if (!this.swiper) return;
      this.swiper.slidePrev();
    }
  },

  computed: {
    flatSlides() {
      if (!Array.isArray(this.objectHistory.years)) {
        return [];
      }

      return this.objectHistory.years.reduce((acc, year) => {
        const yearTitle = year && year.title !== undefined ? String(year.title) : '';
        const items = Array.isArray(year.items) ? year.items : [];

        items.forEach((item, index) => {
          acc.push({
            year: yearTitle,
            title: item.title || '',
            description: item.description || '',
            picture: item.picture || '',
            isFirstInYear: index === 0,
            isLastInYear: index === items.length - 1
          });
        });

        return acc;
      }, []);
    }
  },

  watch: {
    history: {

      handler() {
        this.activeIndex = 0;
        this.$nextTick(() => {
          this.initSwiper();

      if (!this.swiper) return;

      // Принудительный пересчёт размеров и слайдов
      this.swiper.updateSize();
      this.swiper.updateSlides();
      this.swiper.updateProgress();
      this.swiper.update();

      this.updateControls(this.swiper);
    }, 300);

    window.addEventListener('resize', this.handleResize);
  },

  beforeUnmount() {
    window.removeEventListener('resize', this.handleResize);
    this.destroySwiper();
