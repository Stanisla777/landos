function updateSize() {
  const containers = document.querySelectorAll('.js--img-observe');
  containers.forEach(container => {
    //сбрасываю стили, чтобы получить реальные размеры
    container.style.width = '';
    container.style.height = '';

    // Получаю текущие размеры
    const w = Math.floor(container.clientWidth);
    const h = Math.floor(container.clientHeight);

    // Устанавливаю новые размеры
    container.style.width = `${w}px`;
    container.style.height = `${h}px`;
  });
}



export default function accordeonHeight(el) {

  const parent = document.querySelector('.js--questions-answers-accordion');
  const btn = document.querySelector('.js--btn-questions-answers');

  if (!parent || !btn) return;

  const items = parent.querySelectorAll('.js-accordion-parent');

  // Если элементов 5 или меньше - скрываем кнопку
  if (items.length <= 5) {
    btn.classList.add('btn-accordion-hidden');
    return;
  }

  // Скрываем элементы, начиная с 6-го
  items.forEach((item, index) => {
    if (index >= 5) {
      item.classList.add('hidden');
    }
  });


  // Обработчик клика на кнопку
  btn.addEventListener('click', () => {
    const hiddenItems = parent.querySelectorAll('.js-accordion-parent.hidden');

    hiddenItems.forEach(item => {
      // Измеряю правильную высоту (временно показываю)
      item.style.transition = 'none';
      item.style.maxHeight = 'none';
      item.style.opacity = '1';
      item.style.paddingBottom = '';
      item.style.paddingTop = '';
      item.classList.remove('hidden');

      const originalHeight = item.scrollHeight;

      // Возвращаю в сжатое состояние
      item.style.maxHeight = '0';
      item.style.opacity = '0';
      item.style.paddingBottom = '0';
      item.style.paddingTop = '0';

      void item.offsetHeight; // reflow

      // Анимирую
      requestAnimationFrame(() => {
        item.style.transition =
          'max-height 0.5s ease, opacity 0.5s ease, padding-bottom 0.5s ease, padding-top 0.5s ease';

        item.style.maxHeight = `${originalHeight}px`;
        item.style.opacity = '1';
        item.style.paddingBottom = '';
        item.style.paddingTop = '';
      });

      // Очистка после анимации
      item.addEventListener('transitionend', function handler(e) {
        if (e.propertyName !== 'max-height') return;
        item.style.transition = '';
        item.style.maxHeight = '';
        item.style.opacity = '';
        item.style.paddingBottom = '';
        item.style.paddingTop = '';
        item.removeEventListener('transitionend', handler);
      });
    });

    btn.classList.add('btn-accordion-hidden');
  });

  const resizeObserver = new ResizeObserver(updateSize);

  // Нахожу все элементы и подписываюсь на их изменения
  const containers = document.querySelectorAll('.js--img-observe');
  containers.forEach(container => {
    resizeObserver.observe(container);
  });

  // Также подписываюсь на изменение размера окна на всякий случай
  window.addEventListener('resize', updateSize);

  // Инициализация при первом запуске
  updateSize();
}
