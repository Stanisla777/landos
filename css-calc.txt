import {

function CommentsFormValidation() {
  const form = document.querySelector('.js--form-comments');
  
  if (!form) return;

  form.addEventListener('submit', (event) => {
    event.preventDefault();
    
    const button = form.querySelector('.form-comments__button');
    if (button) {
      button.classList.add('submitting');
      button.setAttribute('disabled', 'disabled');
    }

    if (!errorsValid.name && !errorsValid.check && !errorsValid.text) {
      // Создаём контейнер для капчи внутри формы
      const container = createCaptchaContainer(form, 'captcha-comment-main');
      
      // Запускаем капчу через наш модуль
      runCaptcha(container, (token) => {
        let formData = new FormData(form);
        
        // Добавляем токен (реальный или fallback)
        if (token !== null && token !== undefined) {
          formData.append('smart-token', token);
        }
        
        sendComment(form, formData, errorsValid);
      }, {
        sitekey: conf?.smartcaptcha_key,
        
        // Подписываемся на события капчи
        onWidgetCreated: (widgetId) => {
          // Пользователь закрыл окно капчи
          subscribeCaptcha(container, 'challenge-hidden', () => {
            console.log('Пользователь закрыл окно капчи');
            if (button) {
              button.classList.remove('submitting');
              button.removeAttribute('disabled');
            }
            resetCaptcha(container, false);
          });

          // Токен истёк
          subscribeCaptcha(container, 'token-expired', () => {
            console.warn('Токен капчи истёк');
            if (button) {
              button.classList.remove('submitting');
              button.removeAttribute('disabled');
            }
            resetCaptcha(container, false);
          });
        }
      });
    }
  });
}

function sendComment(form, formData, errorsValid) {
  const button = form.querySelector('.form-comments__button');
  
  let request = BX.ajax.runComponentAction('local:comments.form', 'save', {
    mode: 'class',
    data: formData
  });

  request.then(function(response) {
    commentsPopupShow(response.data.title, response.data.message, response.status);
    form.reset();
    
    if (button) {
      button.classList.add('form-comments__button_disabled');
      button.classList.remove('submitting');
      button.removeAttribute('disabled');
    }
    
    errorsValid = {
      name: false,
      text: false,
      check: false
    };
    
    updateCountComments(response);
    updateCommentsList(response.data.id, 0);
    
    // Сбрасываем капчу после успешной отправки
    const container = form.querySelector('.captcha-element');
    if (container) {
      resetCaptcha(container, false);
    }
    
  }, function(response) {
    commentsPopupShow('Ошибка', response.errors, response.status);
    
    // Разблокируем кнопку при ошибке
    if (button) {
      button.classList.remove('submitting');
      button.removeAttribute('disabled');
    }
    
    // Сбрасываем капчу при ошибке
    const container = form.querySelector('.captcha-element');
    if (container) {
      resetCaptcha(container, false);
    }
  });
}
