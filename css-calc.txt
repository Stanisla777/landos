function updateIncomeArrayFromAnswers(stateArray, incomingArray) {
  if (!incomingArray || !Array.isArray(incomingArray)) return;

  incomingArray.forEach((incomingItem, index) => {
    if (stateArray[index]) {
      const stateItem = stateArray[index];

      // 1. Обновляем базовые поля
      stateItem.isActive = incomingItem.isActive;
      stateItem.isShow = incomingItem.isShow;
      stateItem.value = incomingItem.value;

      // 2. Обновляем коэффициенты
      if (incomingItem.additionalCoefficient && stateItem.additionalCoefficient) {
        const distVal = incomingItem.additionalCoefficient.districtCoefficient;
        const northVal = incomingItem.additionalCoefficient.northernAllowance;

        // Вспомогательная функция для поиска нужного объекта в variant по label
        const findMatchingVariant = (variants, targetValue) => {
          if (!variants || targetValue === undefined || targetValue === null || targetValue === '') {
            return null;
          }
          // Ищем объект в variant, у которого label совпадает с пришедшим значением
          return variants.find(v => String(v.label).trim() === String(targetValue).trim()) || null;
        };

        // --- Районный коэффициент ---
        if (distVal !== undefined) {
          const matchedDist = findMatchingVariant(
            stateItem.additionalCoefficient.districtCoefficient.variant,
            distVal
          );
          // Если нашли совпадение, записываем его в массив (как ожидает ds-multi-select)
          // Если нет, записываем пустой массив, чтобы сбросить выбор
          stateItem.additionalCoefficient.districtCoefficient.value = matchedDist ? [matchedDist] : [];
        }

        // --- Северная надбавка ---
        if (northVal !== undefined) {
          const matchedNorth = findMatchingVariant(
            stateItem.additionalCoefficient.northernAllowance.variant,
            northVal
          );
          stateItem.additionalCoefficient.northernAllowance.value = matchedNorth ? [matchedNorth] : [];
        }
      }
    }
  });
}
