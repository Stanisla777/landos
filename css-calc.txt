/**
 * Универсальная функция для обновления массива доходов из data-answers
 * @param {Array} stateArray - Массив из state (man или wife)
 * @param {Array} incomingArray - Массив из пришедших данных (received_perem)
 */
function updateIncomeArrayFromAnswers(stateArray, incomingArray) {
  if (!incomingArray || !Array.isArray(incomingArray)) return;

  incomingArray.forEach((incomingItem, index) => {
    if (stateArray[index]) {
      const stateItem = stateArray[index];

      // 1. Обновляем базовые поля
      stateItem.isActive = incomingItem.isActive;
      stateItem.isShow = incomingItem.isShow;
      stateItem.value = incomingItem.value;

      // 2. Обновляем коэффициенты
      if (incomingItem.additionalCoefficient && stateItem.additionalCoefficient) {
        const distVal = incomingItem.additionalCoefficient.districtCoefficient;
        const northVal = incomingItem.additionalCoefficient.northernAllowance;

        // Вспомогательная функция для поиска нужного объекта в variant по label
        const findMatchingVariant = (variants, targetValue) => {
          if (!variants || targetValue === undefined || targetValue === null || targetValue === '') {
            return null;
          }
          // Ищем объект в variant, у которого label совпадает с пришедшим значением
          return variants.find(v => String(v.label).trim() === String(targetValue).trim()) || null;
        };

        // --- Районный коэффициент ---
        if (distVal !== undefined) {
          const matchedDist = findMatchingVariant(
            stateItem.additionalCoefficient.districtCoefficient.variant, 
            distVal
          );
          // Если нашли совпадение, записываем его в массив (как ожидает ds-multi-select)
          // Если нет, записываем пустой массив, чтобы сбросить выбор
          stateItem.additionalCoefficient.districtCoefficient.value = matchedDist ? [matchedDist] : [];
        }

        // --- Северная надбавка ---
        if (northVal !== undefined) {
          const matchedNorth = findMatchingVariant(
            stateItem.additionalCoefficient.northernAllowance.variant, 
            northVal
          );
          stateItem.additionalCoefficient.northernAllowance.value = matchedNorth ? [matchedNorth] : [];
        }
      }
    }
  });
}

mutationAnswers(state, received_perem) {
  // 1. Базовые поля
  state.data_year = received_perem.dataYear;
  state.check_calculate_together = received_perem.calculateTogether;
  state.apartment_price = received_perem.apartmentPrice;
  state.interest_paid = received_perem.interestPaid;
  state.funds_used = received_perem.fundsUsed;

  // 2. Предыдущие вычеты (мужчина)
  if (received_perem.previousPaymentsMan) {
    state.previousPaymentsMan = {
      ...state.previousPaymentsMan,
      ...received_perem.previousPaymentsMan
    };
  }
  
  // 3. Предыдущие вычеты (женщина)
  if (received_perem.previousPaymentsWife) {
    state.previousPaymentsWife = {
      ...state.previousPaymentsWife,
      ...received_perem.previousPaymentsWife
    };
  }

  // 4. Динамическое обновление доходов с помощью нашей функции!
  updateIncomeArrayFromAnswers(state.array_all_income_period_man, received_perem.incomeMan);
  updateIncomeArrayFromAnswers(state.array_all_income_period_wife, received_perem.incomeWife);
}

methods: {
  initializeFieldsIncomeFromStore() {
    const fields = {};

    this.array_all_income_period.forEach(tab => {
      // 1. Основной доход
      fields[tab.modelInput] = tab.value || '';

      // 2. Коэффициенты
      if (tab.additionalCoefficient) {
        const baseKey = tab.additionalCoefficient.modelInput;
        
        // Мы передаем массив объектов напрямую, так как ds-multi-select ожидает именно это
        fields[baseKey + '_district'] = tab.additionalCoefficient.districtCoefficient.value || [];
        fields[baseKey + '_northern'] = tab.additionalCoefficient.northernAllowance.value || [];
      }
    });

    this.fieldsIncome = fields;
  }
}
