<template>
  <div class="input-story-container ds-input__field-wrapper feed_back__story-wr-select input-emty select-list__wrapper js--select js--openlist-container js--only-mobile-fixed"
       :class="dropDown ? 'open' : ''"
       @click="clickdropDown">
       
    <div class="select__background" :class="dropDown ? 'show-vue' : ''"></div>
    
    <DsInput
      class="height-56 z11"
      ref="Input"
      v-model="dataField"
      :label="view === 'district' ? item.additionalCoefficient?.districtCoefficient?.placeholder : item.additionalCoefficient?.northernAllowance?.placeholder"
      preset="decimal"
      labelPosition="inside"
      @input="logEvent"
      :unit="''"
      :maxlength="5" 
      :autocomplete="'off'"
    />
    
    <div class="ds-multiselect__dropdown z-16 select-list__selection-window modal-special-styles">
      <div class="select-list__head">
        <p>{{ view === 'district' ? item.additionalCoefficient?.districtCoefficient?.placeholder : item.additionalCoefficient?.northernAllowance?.placeholder }}</p>
        <div class="select-list__head-close js--select-list-close" @click.stop="dropDown = false">
          <svg width="10" height="10" viewbox="0 0 10 10" fill="none" xmlns="http://www.w3.org/2000/svg">
            <path fill-rule="evenodd" clip-rule="evenodd" d="M0.209209 0.209209C0.488155 -0.0697365 0.940416 -0.0697365 1.21936 0.209209L5 3.98985L8.78064 0.209209C9.05958 -0.0697365 9.51184 -0.0697365 9.79079 0.209209C10.0697 0.488155 10.0697 0.940416 9.79079 1.21936L6.01015 5L9.79079 8.78064C10.0697 9.05958 10.0697 9.51184 9.79079 9.79079C9.51184 10.0697 9.05958 10.0697 8.78064 9.79079L5 6.01015L1.21936 9.79079C0.940416 10.0697 0.488155 10.0697 0.209209 9.79079C-0.0697365 9.51184 -0.0697365 9.05958 0.209209 8.78064L3.98985 5L0.209209 1.21936C-0.0697365 0.940416 -0.0697365 0.488155 0.209209 0.209209Z" fill="#252628"></path>
          </svg>
        </div>
      </div>

      <div class="feed_back__story-wr-search js--scroll-wrap">
        <div class="feed_back__story-scroll-inner js--scroll">
          <div class="ds-multiselect__option-item"
               v-for="item2 in filteredVariants"
               :key="item2.id"
               @click="selectItem(item2.label, $event)">
            <p class="ds-multiselect__checkbox-label">{{ item2.label }}</p>
          </div>
          
          <div v-if="filteredVariants.length === 0 && isUserInteracted" class="no-results">
            <p>Нет совпадений</p>
          </div>
        </div>

        <div class="feed_back__story-scrollbar js--scrollbar" aria-hidden="true">
          <div class="feed_back__story-scrollbar-thumb js--scrollbar-thumb"></div>
        </div>
      </div>
    </div>
  </div>
</template>

<script>
import { DsInput } from '@ringsites/ui-components/DsInput';
import ScrollbarThumb from '../../../modules/scrollbarThumb.js';

export default {
  name: 'component-district-coefficient',
  props: {
    item: { type: Object, required: true },
    modelValue: { type: [String, Number, Array], default: '' },
    view: { type: String, default: 'district' } // 'district' или 'northern'
  },
  data() {
    return {
      dataField: '',
      dropDown: false,
      isUserInteracted: false,
    }
  },
  computed: {
    filteredVariants() {
      // 1. Динамически выбираем массив variant в зависимости от view
      const variants = this.view === 'district' 
        ? (this.item?.additionalCoefficient?.districtCoefficient?.variant || [])
        : (this.item?.additionalCoefficient?.northernAllowance?.variant || []);

      const searchQuery = String(this.dataField).trim();

      // 2. Если пользователь еще не печатал или поле пустое, отдаем полный список
      if (!this.isUserInteracted || !searchQuery || searchQuery === '.') {
        return variants;
      }

      // 3. Фильтрация по началу строки
      return variants.filter(item2 => {
        return String(item2.label).startsWith(searchQuery);
      });
    }
  },
  watch: {
    // ⭐ КЛЮЧЕВОЕ ИСПРАВЛЕНИЕ:
    modelValue: {
      handler(newVal) {
        let normalizedVal = '';
        if (Array.isArray(newVal) && newVal.length > 0) {
          normalizedVal = String(newVal[0].label || newVal[0].value || '');
        } else {
          normalizedVal = String(newVal || '');
        }

        // Обновляем поле и сбрасываем флаг ТОЛЬКО если значение пришло извне 
        // и отличается от текущего (например, Store очистил поле при смене таба).
        // Если пользователь просто печатает, родитель вернет то же значение, 
        // условие this.dataField !== normalizedVal будет false, и флаг НЕ сбросится!
        if (this.dataField !== normalizedVal) {
          this.dataField = normalizedVal;
          if (normalizedVal === '') {
            this.isUserInteracted = false;
          }
        }
      },
      immediate: true
    }
  },
  methods: {
    clickdropDown(event) {
      // Простой и надежный тоггл: если клик был не по инпуту и не по дропдауну, закрываем
      if (event.target.closest('.ds-input__field') || event.target.closest('.ds-multiselect__dropdown')) {
        this.dropDown = !this.dropDown;
      } else {
        this.dropDown = false;
      }
    },
    selectItem(label, event) {
      this.dataField = String(label);
      this.dropDown = false;
      this.isUserInteracted = true;
      
      const element = event.currentTarget;
      const parent = element.closest('.js--select');
      if (parent?.querySelector('.ds-multiselect__option-item.active')) {
        parent.querySelector('.ds-multiselect__option-item.active').classList.remove('active');
      }
      element.classList.add('active');
      
      this.$emit('update:modelValue', this.dataField);
    },
    logEvent(event) {
      this.isUserInteracted = true; // Пользователь начал взаимодействовать
      
      let val = event.target.value.replace(/[^\d.,]/g, '').replace(',', '.');
      const parts = val.split('.');
      
      // Ограничение до 2 знаков после точки
      if (parts.length === 2 && parts[1].length > 2) {
        val = parts[0] + '.' + parts[1].substring(0, 2);
      }

      const num = parseFloat(val);

      if (!isNaN(num)) {
        if (this.view === 'district') {
          if (num < 1) val = '1';
          else if (num >= 3) val = '3'; // Если введено 3 или больше, фиксируем на 3
        } else {
          if (num < 0.1) val = '0.1';
          else if (num >= 100) val = '100';
        }
      }

      this.dataField = val;
      this.$emit('update:modelValue', this.dataField);
    }
  },
  mounted() {
    ScrollbarThumb();
    // ⭐ УДАЛЕНО: жесткое присвоение districtCoefficient. 
    // Теперь за начальную инициализацию отвечает watch с immediate: true, 
    // который корректно работает и для 'district', и для 'northern'.
  },
  components: {
    DsInput
  }
};
</script>

<style scoped>
.no-results {
  padding: 10px;
  text-align: center;
  color: #999;
  font-size: 14px;
}
</style>
