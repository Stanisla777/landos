<script>
import store from "../store";
import { DsInput } from '@ringsites/ui-components/DsInput';
import { DsCheckbox } from '@ringsites/ui-components';
import ComponentincomeWithAdditionalCoefficient from './component-income-with-additional-coefficient';
import ComponenTabsTimeInterval from './component-tabs-time-interval';
import { debounce } from "../../../modules/debounce";

export default {
  name: 'component-income-man',

  data() {
    return {
      dataFieldCheckbox: false,
      fieldsIncomeMan: {},
      isUpdatingFromStore: false, // Флаг-предохранитель
      previousPayments: {
        deduction_purchase: '',
        deduction_interest: '',
        other_deductions: '',
      }
    }
  },

  created() {
    // 1. Инициализируем данные при загрузке
    this.initializeFieldsIncomeFromStore();

    // 2. Создаем debounce-функцию ОДИН раз.
    // Это позволяет нам использовать метод .cancel() при необходимости.
    this.debouncedIncomeUpdate = debounce((newFields) => {
      // Последний рубеж обороны (на всякий случай)
      if (this.isUpdatingFromStore) return;

      const cleanedFields = {};
      for (const key in newFields) {
        cleanedFields[key] = parseInt(String(newFields[key]).replace(/\s/g, ''), 10) || 0;
      }

      store.dispatch('ActionShowLoader', true);
      store.dispatch('ActionInputIncomeMan', cleanedFields)
        .finally(() => {
          store.dispatch('ActionShowLoader', false);
        });
    }, 500);
  },

  watch: {
    // === СЛУШАТЕЛЬ 1: Реагируем на изменения в Store (например, клик по табу) ===
    array_all_income_period: {
      handler() {
        // 1. Поднимаем флаг, чтобы заблокировать отправку данных обратно
        this.isUpdatingFromStore = true;

        // 2. ОТМЕНЯЕМ таймер debounce, если пользователь в этот момент печатал
        if (this.debouncedIncomeUpdate) {
          this.debouncedIncomeUpdate.cancel();
        }

        // 3. Обновляем локальные поля из Store
        this.initializeFieldsIncomeFromStore();

        // 4. Сбрасываем флаг после того, как Vue обновил DOM
        this.$nextTick(() => {
          this.isUpdatingFromStore = false;
        });
      },
      deep: true
    },

    // === СЛУШАТЕЛЬ 2: Реагируем на ввод пользователя в поля доходов ===
    fieldsIncomeMan: {
      handler(newFields) {
        // ⭐ ГЛАВНОЕ ИСПРАВЛЕНИЕ: Синхронная проверка.
        // Если обновление инициировано Store (см. Слушатель 1), мы прерываем выполнение
        // НЕМЕДЛЕННО, не давая запуститься новому таймеру debounce.
        if (this.isUpdatingFromStore) return;

        this.debouncedIncomeUpdate(newFields);
      },
      deep: true,
      immediate: false
    },

    // === СЛУШАТЕЛЬ 3: Чекбокс районного коэффициента ===
    dataFieldCheckbox() {
      store.dispatch('ActionCheckDistrictCoefficientMan', this.dataFieldCheckbox);
    },

    // === СЛУШАТЕЛЬ 4: Предыдущие вычеты (тут цикл Store->Component не грозит, поэтому inline debounce ок) ===
    previousPayments: {
      handler: debounce(function (newValue) {
        const cleanedPayments = {};
        for (const key in newValue) {
          cleanedPayments[key] = parseInt(String(newValue[key]).replace(/\s/g, ''), 10) || 0;
        }

        store.dispatch('ActionShowLoader', true);
        store.dispatch('ActionInputPreviousPaymentsMan', cleanedPayments)
          .finally(() => {
            store.dispatch('ActionShowLoader', false);
          });
      }, 500),
      deep: true,
      immediate: false
    },
  },

  beforeUnmount() {
    // Очищаем память при уходе со страницы
    if (this.debouncedIncomeUpdate) {
      this.debouncedIncomeUpdate.cancel();
    }
  },

  methods: {
    initializeFieldsIncomeFromStore() {
      const fields = {};

      this.array_all_income_period.forEach(tab => {
        // 1. Основной доход
        fields[tab.modelInput] = tab.value || '';

        // 2. Дополнительные коэффициенты (ОБЯЗАТЕЛЬНО, чтобы они не пропадали!)
        if (tab.additionalCoefficient) {
          const baseKey = tab.additionalCoefficient.modelInput;
          fields[baseKey + '_district'] = tab.additionalCoefficient.districtCoefficient.value || '';
          fields[baseKey + '_northern'] = tab.additionalCoefficient.northernAllowance.value || '';
        }
      });

      this.fieldsIncomeMan = fields;
    },
  },

  computed: {
    check_calculate_together() {
      return store.getters.CHECKCALCULATETOGETHER;
    },
    data_year() {
      return store.getters.DATAYEAR;
    },
    array_all_income_period() {
      return store.getters.ARRAYALLINCOMEPERIODMAN;
    },
    visibleBlockIncomePeriod() {
      const activeIndex = this.array_all_income_period.findIndex(tab => tab.isActive === true);
      if (activeIndex === -1) return [];
      return this.array_all_income_period.slice(0, activeIndex + 1);
    },
    check_district_coefficient() {
      return store.getters.CHECKDISTRICTCOEFFICIENTMAN;
    },
  },

  components: {
    DsInput,
    DsCheckbox,
    ComponentincomeWithAdditionalCoefficient,
    ComponenTabsTimeInterval
  },
};
</script>
