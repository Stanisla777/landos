props: {
  answers: {
    type: [Object, String],
    default: () => ({}),
    validator: (value) => {
      // 1. Явно разрешаем пустую строку, null или undefined 
      // (атрибут может отсутствовать или быть пустым)
      if (value === '' || value === null || value === undefined) {
        return true;
      }

      // 2. Если это уже объект — ок
      if (typeof value === 'object' && value !== null && !Array.isArray(value)) {
        return true;
      }

      // 3. Если это строка — пробуем распарсить
      if (typeof value === 'string') {
        try {
          const parsed = JSON.parse(value);
          return typeof parsed === 'object' && parsed !== null;
        } catch (e) {
          // Если строка не является валидным JSON, считаем это ошибкой
          return false;
        }
      }

      return false;
    }
  }
}
