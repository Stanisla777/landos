onModelUpdate(value) {
  console.log('[onModelUpdate] IN  value =', JSON.stringify(value));
  console.log('[onModelUpdate] dataField BEFORE =', JSON.stringify(this.dataField));

  this.isUserInteracted = true;

  if (value === '' || value === null || value === undefined) {
    this.dataField = '';
    this.$emit('update:modelValue', '');
    console.log('[onModelUpdate] OUT (clear) dataField =', JSON.stringify(this.dataField));
    return;
  }

  let val = String(value).replace(/[^\d.,]/g, '').replace(',', '.');
  console.log('[onModelUpdate] normalized val =', JSON.stringify(val));

  const parts = val.split('.');
  if (parts.length === 2 && parts[1].length > 2) {
    val = parts[0] + '.' + parts[1].substring(0, 2);
    console.log('[onModelUpdate] after trim =', JSON.stringify(val));
  }

  if (val === '' || val === '.' || val.endsWith('.')) {
    console.log('[onModelUpdate] INTERMEDIATE — emit only, dataField untouched =', JSON.stringify(this.dataField));
    this.$emit('update:modelValue', val);
    return;
  }

  const num = parseFloat(val);
  if (!isNaN(num)) {
    if (this.view === 'district') {
      if (num < 1) val = '1';
      else if (num >= 3) val = '3';
    } else {
      if (num < 0.1) val = '0.1';
      else if (num >= 100) val = '100';
    }
  }

  this.dataField = val;
  console.log('[onModelUpdate] OUT dataField =', JSON.stringify(this.dataField));
  this.$emit('update:modelValue', this.dataField);
}
