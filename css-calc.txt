computed: {
    //проверка объекта и нормализация, ,
    //если к-во объектов для шапки и body отличаются, чтобы таблица не ломалась. Вывожу в консоль

    // Нормализация шапки — только если включена
    normalizedHeaders() {
      // Если нормализация выключена
      if (!this.normalizeColumn) {
        // Если headers есть — возвращаем как есть
        if (this.tableData.headers && this.tableData.headers.length > 0) {
          return this.tableData.headers;
        }
        // Если headers нет, но headerAlways = true — генерируем пустые
        if (this.headerAlways) {
          const maxCells = this.getMaxCellsCount();
          return Array.from({ length: maxCells }, (_, i) => ({
            id: `auto-header-${i}`,
            label: '',
            sortable: false
          }));
        }
        return [];
      }

      // Нормализация включена
      const hasHeaders = this.tableData.headers && this.tableData.headers.length > 0;
      const maxCellsInRows = this.getMaxCellsCount();

      if (!hasHeaders && !this.headerAlways) {
        return [];
      }

      const targetCount = hasHeaders
        ? Math.max(this.tableData.headers.length, maxCellsInRows)
        : maxCellsInRows;

      if (targetCount === 0) return [];

      let headers = hasHeaders ? [...this.tableData.headers] : [];

      // Добавляю пустые заголовки, если не хватает
      while (headers.length < targetCount) {
        headers.push({
          id: `empty-header-${headers.length}`,
          label: '',
          sortable: false
        });
      }
      // Нормализую существующие заголовки
      return headers.map((header, index) => ({
        id: header.id ?? `header-${index}`,
        label: header.label || '',
        sortable: header.sortable ?? false,
        ...header
      }));
    },

    // Нормализация строк для body — только если включена
    normalizedRows() {
      // Если нормализация выключена
      if (!this.normalizeColumn) {
        return (this.tableData.rows || []).map(row => {
          // Поддерживается оба формата: массив и объект с cells
          if (Array.isArray(row)) {
            return { cells: row };
          }
          return row;
        });
      }

      const headerCount = this.normalizedHeaders.length;

      if (headerCount === 0) return this.tableData.rows || [];

      return (this.tableData.rows || []).map((row, rowIndex) => {
        const cells = row.cells || [];
        const currentCount = cells.length;

        if (process.env.NODE_ENV === 'development' && currentCount !== headerCount) {
          console.warn(
            `[TableStory] Строка ${rowIndex}: ожидается ${headerCount} ячеек, получено ${currentCount}`
          );
        }

        // Обреза. лишние ячейки (хотя теперь это не должно происходить)
        let normalizedCells = [...cells].slice(0, headerCount);

        // ДобавляЮ пустые ячейки, если не хватает
        while (normalizedCells.length < headerCount) {
          normalizedCells.push({
            id: `empty-${rowIndex}-${normalizedCells.length}`,
            label: '',
            link: '',
            icon: ''
          });
        }

        return {
          id: row.id ?? `row-${rowIndex}`,
          ...row,
          cells: normalizedCells.map((cell, cellIndex) => ({
            ...cell,
            id: cell.id ?? `cell-${rowIndex}-${cellIndex}`
          }))
        };
      });
    },

    sortedRows() {
      // Если сортировка выключена — возвращаются исходные данные
      if (!this.sortingRow || this.activeSortColumn === null || this.sortDirection === null) {
        return this.normalizeColumn ? this.normalizedRows : (this.tableData.rows || []);
      }

      const rows = this.normalizeColumn ? this.normalizedRows : (this.tableData.rows || []);
      const col = this.activeSortColumn;
      const direction = this.sortDirection === 'asc' ? 1 : -1;

      return [...rows].sort((a, b) => {
        const valA = this.extractValue(a.cells[col]?.label || '');
        const valB = this.extractValue(b.cells[col]?.label || '');
        // Если типы разные — сравниваем как строки
        if (valA.type !== valB.type) {
          return direction * String(valA.value).localeCompare(String(valB.value), 'ru');
        }
        // Числа и даты сравниваются численно
        if (valA.type === 'number' || valA.type === 'date') {
          return direction * (valA.value - valB.value);
        }
        // Строки — по алфавиту с русской локалью
        return direction * valA.value.localeCompare(valB.value, 'ru');
      });
    },

    columnTypes() {
      // Если numberRight = false, не делаем вычисления
      if (!this.numberRight) return [];

      const types = [];
      const rows = this.normalizeColumn ? this.normalizedRows : (this.tableData.rows || []);
      const headers = this.normalizeColumn ? this.normalizedHeaders : (this.tableData.headers || []);
      const colCount = headers.length;

      if (rows.length === 0 || colCount === 0) return [];

      for (let colIndex = 0; colIndex < colCount; colIndex++) {
        let numberCount = 0;
        let totalCount = 0;

        rows.forEach(row => {
          const cell = row.cells[colIndex];
          if (cell && cell.label) {
            const extracted = this.extractValue(cell.label);
            totalCount++;
            if (extracted.type === 'number') numberCount++;
          }
        });

        types.push({
          isNumeric: totalCount > 0 && numberCount > totalCount / 2
        });
      }

      return types;
    },

    hasNestedContent() {
      const rows = this.normalizeColumn ? this.normalizedRows : (this.tableData.rows || []);

      return rows.some(row => {
        const label = row.nesting?.label;
        return label && String(label).trim() !== '';
      });
    }
  },
  methods: {
    // Вспомогательный метод для получения максимального количества ячеек
    getMaxCellsCount() {
      const rows = this.tableData.rows || [];
      return rows.reduce((max, row) => {
        const cellsCount = (row.cells || []).length;
        return Math.max(max, cellsCount);
      }, 0);
    },

    //открытие вложенности
    toggleExpand() {
      this.isExpanded = !this.isExpanded;
    },

    isNumericColumn(columnIndex) {
      if (!this.numberRight) return false;
      return this.columnTypes[columnIndex]?.isNumeric || false;
    },

    // // Форматирование числа для отображения
    formatNumber(value, colIndex) {
      // Если колонка не числовая — сразу возвращаем
      if (!this.isNumericColumn(colIndex)) return value;

      // Если оба флага выключены — сразу возвращаем
      if (!this.breakThousandths && !this.numberMarker) return value;

      let formatted = value;

      if (this.breakThousandths) {
        const extracted = this.extractValue(value);
        if (extracted.type === 'number') {
          formatted = extracted.value.toLocaleString('ru-RU').replace(/,/g, ' ');
        }
      }

      // Добавление маркера
      if (this.numberMarker && this.numberMarker.trim()) {
        formatted = `${formatted} ${this.numberMarker}`;
      }

      return formatted;
    },

    // Получение значения для отображения (с форматированием)
    getDisplayValue(cell, colIndex) {
      if (!cell || !cell.label) return '';

      // Если numberRight выключен — сразу возвращаем label
      if (!this.numberRight) return cell.label;

      return this.formatNumber(cell.label, colIndex);
    },

    //метод для проверки ссылки, если https:// - добавить target="_blank"
    isExternalLink(link) {
      if (!link) return false;
      return link.startsWith('http://') || link.startsWith('https://');
    },

    //метод для проверки какого вида картинка в ключе icon - это элемент svg или путь к кратинке
    isImagePath(icon) {
      if (!icon || typeof icon !== 'string') return false;

      const trimmed = icon.trim();

      // Если это inline SVG (начинается с тега <svg)
      if (trimmed.toLowerCase().startsWith('<svg')) {
        return false;
      }

      // Если содержит расширения изображений
      const imageExtensions = ['.svg', '.jpg', '.jpeg', '.png', '.webp', '.gif', '.avif'];
      const lowerIcon = trimmed.toLowerCase();

      if (imageExtensions.some(ext => lowerIcon.includes(ext))) {
        return true;
      }

      // Если это путь к ресурсу (начинается с /, ./, ../ или http)
      if (
        trimmed.startsWith('/') ||
        trimmed.startsWith('./') ||
        trimmed.startsWith('../') ||
        trimmed.startsWith('http://') ||
        trimmed.startsWith('https://')
      ) {
        return true;
      }

      // По умолчанию  SVG
      return false;
    },

    //методы для сортировки
    handleSort(columnIndex) {
      if (this.activeSortColumn !== columnIndex) {
    // Клик по новой колонке — начинаем с 'asc'
        this.activeSortColumn = columnIndex;
        this.sortDirection = 'asc';
      } else if (this.sortDirection === 'asc') {
    // Второй клик — переключаем на 'desc'
        this.sortDirection = 'desc';
      } else {
    // Третий клик — сброс к исходному порядку
        this.activeSortColumn = null;
        this.sortDirection = null;
      }
    },

    extractValue(text) {
      if (!text) return { type: 'string', value: '' };

      const trimmed = String(text).trim();

      const dateMatch = trimmed.match(/^(\d{2})\.(\d{2})\.(\d{4})$/);
      if (dateMatch) {
        const [, day, month, year] = dateMatch;
        return {
          type: 'date',
          value: new Date(year, month - 1, day).getTime()
        };
      }

      const cleaned = trimmed.replace(/\s/g, '').replace(/[^\d.,-]/g, '');
      if (cleaned && !isNaN(cleaned)) {
        const number = parseFloat(cleaned.replace(',', '.'));
        if (!isNaN(number)) {
          return { type: 'number', value: number };
        }
      }

      return { type: 'string', value: trimmed.toLowerCase() };
    }
  }
