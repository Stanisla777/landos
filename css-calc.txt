mutationTabTimeMan(state, received_perem) {
    // 1. Приводим к строке для надежного сравнения (так как из DOM часто приходит строка)
    const targetId = String(received_perem);
    
    // 2. Находим индекс элемента, на который кликнули
    const activeIndex = state.array_all_income_period_man.findIndex(
      item => String(item.id) === targetId
    );

    // 3. Проходим по всему массиву и применяем логику
    state.array_all_income_period_man.forEach((item, index) => {
      if (index === activeIndex) {
        // Это тот элемент, на который кликнули
        item.isActive = true;
        // Его value НЕ трогаем, оставляем как есть
      } else {
        // Это все остальные элементы
        item.isActive = false;
        
        // ГЛАВНОЕ УСЛОВИЕ: если индекс текущего элемента больше индекса активного,
        // значит он идет ПОСЛЕ него в массиве. Очищаем его.
        if (index > activeIndex) {
          item.value = ''; 
        }
      }
    });
  }
