created(){
    eventBus.on('eventbtnNext', () => {
      // 1. Логическая защита: если тест уже завершен или отправлен, прерываем выполнение
      if (this.finish_window || this.window_successful_sending) {
        return;
      }

      // 2. Безопасный вызов (Optional Chaining): если элемента нет, ошибка не выбросится
      this.$refs.BtnNext?.click();

      // 3. Переход слайдера, если он инициализирован
      if (mySwiper) {
        mySwiper.slideNext();
      }
    });
  },

  // 4. ОЧИСТКА (Best Practice для Vue 3 и mitt):
  // Удаляем слушатель событий при уничтожении компонента, чтобы избежать утечек памяти 
  // и срабатывания "мертвых" коллбэков.
  unmounted() {
    eventBus.off('eventbtnNext');
  },
