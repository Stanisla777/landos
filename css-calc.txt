formatNumber(value, colIndex) {
  if (!this.isNumericColumn(colIndex)) return value;
  if (!this.breakThousandths && !this.numberMarker) return value;

  // Сохраняем исходные теги
  const tagsMatch = String(value).match(/^(<[^>]*>)(.*?)(<\/[^>]*>)$/);
  const hasTags = tagsMatch !== null;
  const openTag = hasTags ? tagsMatch[1] : '';
  const closeTag = hasTags ? tagsMatch[3] : '';
  const innerContent = hasTags ? tagsMatch[2] : String(value);

  // Анализируем только внутренний контент
  const { number, originalMarker } = this.extractNumberAndMarker(innerContent);
  
  if (number === null) return value;

  let formatted;
  if (this.breakThousandths) {
    formatted = number.toLocaleString('ru-RU').replace(/,/g, ' ');
  } else {
    formatted = String(number);
  }

  let marker;
  if (this.numberMarker && this.numberMarker.trim()) {
    marker = this.numberMarker.trim();
  } else {
    marker = originalMarker;
  }

  if (marker) {
    formatted = `${formatted} ${marker}`;
  }

  // Возвращаем с исходными тегами
  return hasTags ? `${openTag}${formatted}${closeTag}` : formatted;
}
