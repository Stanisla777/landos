dataField(newValue) {
    // 1. Пропускаем пустое поле и промежуточные состояния ввода
    if (newValue === '' || newValue === '.' || newValue === null || newValue === undefined) {
      return;
    }

    const num = parseFloat(newValue);

    // 2. Если NaN (например, пользователь ввёл только точку и что-то ещё некорректное)
    if (isNaN(num)) return;

    // 3. Ограничение снизу: меньше 1 → ставим 1
    if (num < 1) {
      this.dataField = '1';
      return;
    }

    // 4. Ограничение сверху: больше 3 → ставим 3
    if (num > 3) {
      this.dataField = '3';
      return;
    }
  }
