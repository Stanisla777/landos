    // Универсальная логика "примагничивания"
    processSnap(value) {
      const targetIndex = Math.round(value);
      const clampedIndex = Math.max(0, Math.min(this.optionsList.length - 1, targetIndex));

      if (clampedIndex === this.selectedIndex) {
        // Сценарий А: Кликнули или отпустили рядом с текущим маркером.
        // requestAnimationFrame синхронизирует возврат с перерисовкой экрана,
        // перебивая внутреннюю анимацию библиотеки до того, как глаз успеет заметить движение.
        requestAnimationFrame(() => {
          this.dragValue = this.selectedIndex;
        });
      } else {
        // Сценарий Б: Кликнули или дотянули до НОВОГО маркера.
        // Принимаем изменение и отправляем в Store.
        this.selectedIndex = clampedIndex;
        this.dragValue = clampedIndex;
        this.updateStateFromIndex(clampedIndex);
      }
    },
