import {
  createCaptchaContainer,
  runCaptcha,
  resetCaptcha
} from './captchaCore.js';


function initCaptchaForForm(form, containerId) {
  // Создаем контейнер внутри формы
  const container = createCaptchaContainer(form, containerId);

  // Запускаем капчу
  runCaptcha(container, (token) => {
    // Вызываем отправку формы с токеном (или null при fallback)
    submitForm(form, token);
  }, {
    sitekey: conf?.smartcaptcha_key
  });
}



function formSubmission() {
  const btn = document.querySelectorAll('.js--btn-universal-form');
  for(let item_btn = 0; item_btn<btn.length; item_btn++){

    const parent_form = btn[item_btn].closest('.js--universal-form');

    btn[item_btn].onclick=(e)=>{
      // const captcha_id =  captchaInit(parent_form,item_btn)
      e.preventDefault();
      e.stopPropagation();
      let count_required = 0
      const element = e.target

      // Уникальный ID контейнера для этой формы
      const containerId = `yandex-smart-captcha-${item_btn}`;

      if(parent_form){
       if(count_required===0){
          //вот тут команды в случае если всё хорошо
          btn[item_btn].classList.add('submitting')
          const _form_button =  parent_form.querySelector('.js--btn-universal-form')
          if (_form_button){
            _form_button.setAttribute('disabled','disabled')
          }
          // Запускаем капчу
          initCaptchaForForm(parent_form, containerId);
          return true;
        }
     }
   }
  }
}
