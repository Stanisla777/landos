export default {
  name: 'component-apartment-price',
  data(){
    return {
      realtySlider: '',
      dataField: 0,
      dataFieldForCalculation: 0,
      subsidies: 0,
      stepApartment: 1,
      stgMin: 625000,
      stgMiddle: 27000000,
      stgMax: 56000000,
      input_salary: false,
      isDragging: false // <--- ДОБАВЛЕНО
    }
  },
  methods:{
    initRealtySlider() {
      this.realtySlider = noUiSlider.create(this.$refs.mortgagePrice, {
        start: [this.stgMin],
        connect: 'lower',
        step: this.stepApartment,
        initial: this.stgMin,
        range: {
          min: this.stgMin,
          max: this.stgMax
        },
      });

      this.realtySlider.on('update', (val, handle) => {
        this.dataField = parseInt(val[handle]).toFixed(0).toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ')
        this.dataFieldForCalculation = parseInt(val[handle]).toFixed(0)
      });

      // <--- ДОБАВЛЕНО: Отслеживание начала и конца перетаскивания
      this.realtySlider.on('start', () => { this.isDragging = true; });
      this.realtySlider.on('end', () => { this.isDragging = false; });
    },
    
    // ... (методы inputField, keyUp, inputPast, lossFocus, inputFocus оставляем как были) ...

    // ИСПРАВЛЕННЫЙ МЕТОД: Защита от промежуточных значений
    calculationPageLoad() {
      let cleanValue = parseInt(String(this.dataField).replace(/\s/g, '')) || 0;

      // ЕСЛИ ПОЛЬЗОВАТЕЛЬ ТОЛЬКО НАЧАЛ ВВОДИТЬ ЧИСЛО (например, "8" или "62"), 
      // МЫ ИГНОРИРУЕМ ОБНОВЛЕНИЕ СТОРА, ЧТОБЫ НЕ СЛОМАТЬ ПЕРВОНАЧАЛЬНЫЙ ВЗНОС
      if (cleanValue > 0 && cleanValue < this.stgMin) {
        return; 
      }

      // Если поле полностью очищено (0), принудительно ставим минимум
      if (cleanValue === 0) {
        cleanValue = this.stgMin;
        this.dataField = this.stgMin.toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ');
      }

      this.dataFieldForCalculation = cleanValue;
      store.dispatch('ActionRealPriceAppartment', cleanValue);
    },
  },
  mounted(){
    this.initRealtySlider()
  },
  watch:{
    dataField(){
      this.calculationPageLoad()
    }
  },
};

component-maternal-capital.vue

export default {
  name: 'v-component-maternal-capital',
  props:['tooltipMaternityCapital','maximumMaternityCapital'],
  data(){
    return {
      // ... ваши существующие поля ...
      isDragging: false // <--- ДОБАВЛЕНО
    }
  },
  methods:{
    initRealtySlider() {
      this.realtySlider = noUiSlider.create(this.$refs.mortgagePrice, {
        start: [this.start],
        connect: 'lower',
        step: this.stepApartment,
        initial: this.stgMin,
        range: {
          min: this.stgMin,
          max: this.maximumMaternityCapital
        },
      });
      this.realtySlider.on('update', (val, handle) => {
        this.dataField = parseInt(val[handle]).toFixed(0).toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ')
        this.dataFieldForCalculation = parseInt(val[handle]).toFixed(0)
      });

      // <--- ДОБАВЛЕНО
      this.realtySlider.on('start', () => { this.isDragging = true; });
      this.realtySlider.on('end', () => { this.isDragging = false; });
    },
    // ... (остальные методы без изменений) ...
  },
  computed:{
    limitmaterialcapital(){ return store.getters.LIMITMATERIALCAPITAL },
    maternal_capital_show(){ return store.getters.MATERNALCAPITALSHOW },
    maternal_capital_raw(){ return store.getters.MATERNALCAPITAL_RAW } // <--- Убедитесь, что этот геттер есть в store.js
  },
  watch:{
    maternal_capital_raw(newVal) {
      // <--- ИСПРАВЛЕНО: Синхронизируем только если не идет перетаскивание
      if (!this.isDragging) {
        const currentVal = parseInt(String(this.dataField).replace(/\s/g, '')) || 0;
        if (currentVal !== newVal) {
          this.dataField = newVal.toString().replace(/(\d)(?=(\d{3})+$)/g, '$1 ');
          this.realtySlider.set(newVal);
        }
      }
    },
    limitmaterialcapital() {
      if (this.limitmaterialcapital < this.maximumMaternityCapital && this.limitmaterialcapital !== 0) {
        this.realtySlider.updateOptions({ range: { min: this.stgMin, max: this.limitmaterialcapital } });
      } else if (this.limitmaterialcapital >= this.maximumMaternityCapital) {
        this.realtySlider.updateOptions({ range: { min: this.stgMin, max: this.maximumMaternityCapital } });
      } else if (this.limitmaterialcapital === 0) {
        this.$refs.realtyInput.classList.add('unactive')
      }
      
      if (this.limitmaterialcapital !== 0 && this.$refs.realtyInput.classList.contains('unactive')) {
        this.$refs.realtyInput.classList.remove('unactive')
      }
    },
    dataField(){
      this.calculationPageLoad()
    }
  },
  // ...
