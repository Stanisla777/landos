1. Компонент "Материнский капитал" (v-component-maternal-capital)
Найдите метод calculationPageLoad и замените его на этот:

    calculationPageLoad() {
      // Берём актуальное значение из поля ввода, удаляя все пробелы
      const cleanValue = parseInt(String(this.dataField).replace(/\s/g, '')) || 0;
      
      // Синхронизируем внутреннюю переменную для надёжности
      this.dataFieldForCalculation = cleanValue;
      
      store.dispatch('ActionMaternalCapital', cleanValue);
    },
