logEvent(event) {
    this.isUserInteracted = true;

    // 1. Безопасно получаем значение. 
    // Некоторые UI-библиотеки при очистке передают просто строку "" или null, а не объект event.
    const rawVal = typeof event === 'string' ? event : (event?.target?.value ?? '');
    
    let val = rawVal.replace(/[^\d.,]/g, '').replace(',', '.');

    // ⭐ 2. ЯВНАЯ ОБРАБОТКА ОЧИСТКИ (КРЕСТИК)
    // Если значение пустое, немедленно очищаем dataField и прерываем функцию.
    // Это гарантирует, что крестик сработает мгновенно и без побочных эффектов.
    if (val === '') {
      this.dataField = '';
      this.$emit('update:modelValue', '');
      return; 
    }

    // 3. Дальнейшая логика выполняется только если val НЕ пустой
    const parts = val.split('.');
    
    // Оставляем только первую точку, если их вдруг стало больше
    if (parts.length > 2) {
      val = parts[0] + '.' + parts.slice(1).join('');
    }

    // Ограничение до 2 знаков после точки
    if (parts.length === 2 && parts[1].length > 2) {
      val = parts[0] + '.' + parts[1].substring(0, 2);
    }

    // Защита "висящей" точки при вводе
    if (val.endsWith('.')) {
      return; 
    }

    // 4. Мгновенная валидация границ
    const num = parseFloat(val);
    if (!isNaN(num)) {
      if (this.view === 'district') {
        if (num < 1) val = '1';
        else if (num > 3) val = '3';
      } else {
        if (num < 0.1) val = '0.1';
        else if (num > 100) val = '100';
      }
    }

    this.dataField = val;
    this.$emit('update:modelValue', this.dataField);
  },
