<script src="https://smartcaptcha.yandexcloud.net/captcha.js?render=explicit" defer></script>

<div :id="captchaContainerId" class="captcha-container" v-once></div>

export default {
  data() {
    return {
      captchaId: null, // Хранит ID виджета после инициализации
      // Генерируем уникальный ID, чтобы 15 приложений не конфликтовали друг с другом
      captchaContainerId: `yandex-captcha-${Math.random().toString(36).substr(2, 9)}`,
      isSending: false, // Флаг для защиты от двойного клика по кнопке
    };
  },

  // Очистка при удалении компонента (обязательно!)
  beforeUnmount() { // Используйте beforeDestroy, если у вас Vue 2
    if (this.captchaId !== null && window.smartCaptcha) {
      try {
        window.smartCaptcha.reset(this.captchaId);
      } catch (e) {}
      this.captchaId = null;
    }
  },

  methods: {
    // 1. Метод инициализации (вызывается только при необходимости)
    initCaptcha() {
      // Если уже инициализирована или скрипт не загружен — выходим
      if (this.captchaId !== null || !window.smartCaptcha) return;

      const container = document.getElementById(this.captchaContainerId);
      if (!container) return;

      // "Ядерная" очистка на случай, если что-то пошло не так
      container.innerHTML = '';

      try {
        this.captchaId = window.smartCaptcha.render(this.captchaContainerId, {
          sitekey: conf?.smartcaptcha_key,
          invisible: true,
          callback: (token) => {
            this.submitToApi(token);
          },
        });
      } catch (e) {
        console.error('[SmartCaptcha] Ошибка render:', e);
      }
    },

    // 2. Обработчик клика по кнопке "Отправить"
    async handleSend() {
      if (this.isSending || !this.btn_active) return;
      this.isSending = true;

      // Блокируем кнопку визуально (опционально, через ваш store или локально)
      // store.dispatch('ActionButtonActive', false);

      // Если капча еще не инициализирована, инициализируем её прямо сейчас
      if (this.captchaId === null) {
        this.initCaptcha();
        
        // Даем React-движку Яндекса ~50 мс на то, чтобы "схватить" контейнер.
        // Это безопаснее, чем большие setTimeout, и незаметно для пользователя.
        setTimeout(() => {
          if (this.captchaId !== null && window.smartCaptcha) {
            window.smartCaptcha.execute(this.captchaId);
          } else {
            // Если капча так и не инициализировалась (например, нет интернета), 
            // решайте сами: либо разрешить отправку без неё, либо показать ошибку.
            console.warn('Капча не инициализировалась');
            this.isSending = false;
          }
        }, 50);
        return;
      }

      // Если капча УЖЕ инициализирована (второй и последующие клики), выполняем сразу
      if (window.smartCaptcha) {
        window.smartCaptcha.execute(this.captchaId);
      }
    },

    // 3. Метод, который вызывается после успешного прохождения капчи
    submitToApi(token) {
  this.$refs.selectListWrapper?.classList.remove('open');
  this.closeBlock();
  
  store.dispatch('ActionGetResult', [this.oktmo, token]);
  
  // Сброс сразу — просто и работает
  if (this.captchaId !== null && window.smartCaptcha) {
    window.smartCaptcha.reset(this.captchaId);
    this.captchaId = null;
  }
  
  this.isSending = false;
}
