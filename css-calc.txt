for (const _select of _selects) {
  const _select_el = _select.querySelector('.js--openlist-btn');
  if (!_select_el) continue; // Пропускаем, если кнопки нет

  const placeholder = _select_el.getAttribute('data-placeholder');
  if (!placeholder) continue; // Пропускаем, если нет placeholder
  const targets = _select_el.querySelectorAll('.select__substituted-text, .js--selected-item-text');
  targets.forEach(target => {
    target.classList.add('input-emty');
    target.textContent = placeholder;
  });
}
