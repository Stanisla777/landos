<script setup>
import { ref, computed } from 'vue'
import { DsTooltip } from '@ringsites/ui-components'

// 1. Пропсы
const props = defineProps({
  tableData: {
    type: Object,
    default: () => ({ headers: [], rows: [] })
  },
  sameColumn: {
    type: Boolean,
    default: false,
  },
  sortingRow: {
    type: Boolean,
    default: false,
  },
  alignRight: {
    type: Boolean,
    default: false,
  },
  numberRight: {
    type: Boolean,
    default: true,
  },
  numberMarker: {
    type: String,
    default: ''
  },
  breakThousandths: {
    type: Boolean,
    default: false,
  },
  normalizeColumn: {
    type: Boolean,
    default: true
  },
  headerAlways: {
    type: Boolean,
    default: true
  }
})

// 2. Состояние (реактивные переменные)
const activeSortColumn = ref(null)
const sortDirection = ref(null) // null, 'asc', 'desc'
const isExpanded = ref(true)

// 3. Вспомогательные функции (чистые, без this)
const extractValue = (text) => {
  if (!text) return { type: 'string', value: '' }
  let trimmed = String(text).trim().replace(/<[^>]*>/g, '').trim()
  if (!trimmed) return { type: 'string', value: '' }

  const dateMatch = trimmed.match(/^(\d{2})\.(\d{2})\.(\d{4})$/)
  if (dateMatch) {
    const [, day, month, year] = dateMatch
    return { type: 'date', value: new Date(year, month - 1, day).getTime() }
  }

  const cleaned = trimmed.replace(/\s/g, '').replace(/[^\d.,-]/g, '')
  if (cleaned && !isNaN(cleaned)) {
    const number = parseFloat(cleaned.replace(',', '.'))
    if (!isNaN(number)) return { type: 'number', value: number }
  }
  return { type: 'string', value: trimmed.toLowerCase() }
}

const extractNumberAndMarker = (text) => {
  if (!text) return { number: null, originalMarker: '' }
  const cleaned = String(text).trim().replace(/<[^>]*>/g, '').trim()
  if (!cleaned) return { number: null, originalMarker: '' }

  const match = cleaned.match(/^([\d\s.,-]+)\s*(.*)$/)
  if (!match) return { number: null, originalMarker: cleaned }

  const numberPart = match[1].trim()
  const originalMarker = match[2].trim()
  const cleanedNumber = numberPart.replace(/\s/g, '').replace(/[^\d.,-]/g, '')
  
  if (!cleanedNumber || isNaN(cleanedNumber)) return { number: null, originalMarker: cleaned }

  const number = parseFloat(cleanedNumber.replace(',', '.'))
  if (isNaN(number)) return { number: null, originalMarker: cleaned }

  return { number, originalMarker }
}

const isExternalLink = (link) => {
  if (!link) return false
  return link.startsWith('http://') || link.startsWith('https://')
}

const isImagePath = (icon) => {
  if (!icon || typeof icon !== 'string') return false
  const trimmed = icon.trim()
  if (trimmed.toLowerCase().startsWith('<svg')) return false

  const imageExtensions = ['.svg', '.jpg', '.jpeg', '.png', '.webp', '.gif', '.avif']
  const lowerIcon = trimmed.toLowerCase()
  if (imageExtensions.some(ext => lowerIcon.includes(ext))) return true

  if (trimmed.startsWith('/') || trimmed.startsWith('./') || trimmed.startsWith('../') || trimmed.startsWith('http://') || trimmed.startsWith('https://')) {
    return true
  }
  return false
}

const getMaxCellsCount = () => {
  const rows = props.tableData.rows || []
  return rows.reduce((max, row) => {
    const cells = Array.isArray(row) ? row : (row.cells || [])
    return Math.max(max, cells.length)
  }, 0)
}

// 4. Вычисляемые свойства (Computed)
const normalizedHeaders = computed(() => {
  if (!props.normalizeColumn) {
    if (props.tableData.headers && props.tableData.headers.length > 0) return props.tableData.headers
    if (props.headerAlways) {
      const maxCells = getMaxCellsCount()
      return Array.from({ length: maxCells }, (_, i) => ({ id: `auto-header-${i}`, label: '', sortable: false }))
    }
    return []
  }

  const hasHeaders = props.tableData.headers && props.tableData.headers.length > 0
  const maxCellsInRows = getMaxCellsCount()

  if (!hasHeaders && !props.headerAlways) return []

  const targetCount = hasHeaders ? Math.max(props.tableData.headers.length, maxCellsInRows) : maxCellsInRows
  if (targetCount === 0) return []

  let headers = hasHeaders ? [...props.tableData.headers] : []
  while (headers.length < targetCount) {
    headers.push({ id: `empty-header-${headers.length}`, label: '', sortable: false })
  }

  return headers.map((header, index) => ({
    id: header.id ?? `header-${index}`,
    label: header.label || '',
    sortable: header.sortable ?? false,
    ...header
  }))
})

const normalizedRows = computed(() => {
  const headerCount = props.normalizeColumn ? normalizedHeaders.value.length : 0

  return (props.tableData.rows || []).map((row, rowIndex) => {
    if (Array.isArray(row)) {
      return {
        id: `row-${rowIndex}`,
        cells: row.map((cell, cellIndex) => {
          if (cell === null || cell === undefined) return { id: `cell-${rowIndex}-${cellIndex}`, label: '' }
          if (typeof cell === 'string' || typeof cell === 'number') return { id: `cell-${rowIndex}-${cellIndex}`, label: String(cell) }
          return { ...cell, id: cell?.id ?? `cell-${rowIndex}-${cellIndex}`, label: cell?.label ?? '' }
        })
      }
    }

    const cells = row.cells || []
    const hasAllIds = row.id && cells.every(cell => cell?.id)
    const correctCellCount = !props.normalizeColumn || headerCount === 0 || cells.length === headerCount

    if (hasAllIds && correctCellCount) return row

    let normalizedCells = props.normalizeColumn && headerCount > 0 ? [...cells].slice(0, headerCount) : [...cells]

    if (props.normalizeColumn && headerCount > 0) {
      while (normalizedCells.length < headerCount) {
        normalizedCells.push({ id: `empty-${rowIndex}-${normalizedCells.length}`, label: '', link: '', icon: '' })
      }
    }

    normalizedCells = normalizedCells.map((cell, cellIndex) => {
      if (cell === null || cell === undefined) return { id: `cell-${rowIndex}-${cellIndex}`, label: '' }
      if (typeof cell === 'string' || typeof cell === 'number') return { id: `cell-${rowIndex}-${cellIndex}`, label: String(cell) }
      return { ...cell, id: cell?.id ?? `cell-${rowIndex}-${cellIndex}`, label: cell?.label ?? '' }
    })

    return { id: row.id ?? `row-${rowIndex}`, ...row, cells: normalizedCells }
  })
})

const sortedRows = computed(() => {
  if (!props.sortingRow || activeSortColumn.value === null || sortDirection.value === null) {
    return props.normalizeColumn ? normalizedRows.value : (props.tableData.rows || [])
  }

  const rows = props.normalizeColumn ? normalizedRows.value : (props.tableData.rows || [])
  const col = activeSortColumn.value
  const direction = sortDirection.value === 'asc' ? 1 : -1

  return [...rows].sort((a, b) => {
    const valA = extractValue(a.cells[col]?.label || '')
    const valB = extractValue(b.cells[col]?.label || '')
    
    if (valA.type !== valB.type) return direction * String(valA.value).localeCompare(String(valB.value), 'ru')
    if (valA.type === 'number' || valA.type === 'date') return direction * (valA.value - valB.value)
    return direction * valA.value.localeCompare(valB.value, 'ru')
  })
})

const columnTypes = computed(() => {
  if (!props.numberRight) return []
  const types = []
  const rows = props.normalizeColumn ? normalizedRows.value : (props.tableData.rows || [])
  const headers = props.normalizeColumn ? normalizedHeaders.value : (props.tableData.headers || [])
  const colCount = headers.length

  if (rows.length === 0 || colCount === 0) return []

  for (let colIndex = 0; colIndex < colCount; colIndex++) {
    let numberCount = 0
    let totalCount = 0

    rows.forEach(row => {
      const cell = row.cells[colIndex]
      if (cell && cell.label) {
        const extracted = extractValue(cell.label)
        totalCount++
        if (extracted.type === 'number') numberCount++
      }
    })

    types.push({ isNumeric: totalCount > 0 && numberCount > totalCount / 2 })
  }
  return types
})

const hasNestedContent = computed(() => {
  const rows = props.normalizeColumn ? normalizedRows.value : (props.tableData.rows || [])
  return rows.some(row => {
    const label = row.nesting?.label
    return label && String(label).trim() !== ''
  })
})

// 5. Методы (функции, изменяющие состояние или использующие другие функции)
const toggleExpand = () => {
  isExpanded.value = !isExpanded.value
}

const isNumericColumn = (columnIndex) => {
  if (!props.numberRight) return false
  return columnTypes.value[columnIndex]?.isNumeric || false
}

const formatNumber = (value, colIndex) => {
  if (!isNumericColumn(colIndex)) return value
  if (!props.breakThousandths && !props.numberMarker) return value

  const tagsMatch = String(value).match(/^(<[^>]*>)(.*?)(<\/[^>]*>)$/)
  const hasTags = tagsMatch !== null
  const openTag = hasTags ? tagsMatch[1] : ''
  const closeTag = hasTags ? tagsMatch[3] : ''
  const innerContent = hasTags ? tagsMatch[2] : String(value)

  const { number, originalMarker } = extractNumberAndMarker(innerContent)
  if (number === null) return value

  let formatted = props.breakThousandths ? number.toLocaleString('ru-RU').replace(/,/g, ' ') : String(number)
  let marker = (props.numberMarker && props.numberMarker.trim()) ? props.numberMarker.trim() : originalMarker

  if (marker) formatted = `${formatted} ${marker}`
  return hasTags ? `${openTag}${formatted}${closeTag}` : formatted
}

const getDisplayValue = (cell, colIndex) => {
  if (!cell || !cell.label) return ''
  if (!props.numberRight) return cell.label
  return formatNumber(cell.label, colIndex)
}

const handleSort = (columnIndex) => {
  if (activeSortColumn.value !== columnIndex) {
    activeSortColumn.value = columnIndex
    sortDirection.value = 'asc'
  } else if (sortDirection.value === 'asc') {
    sortDirection.value = 'desc'
  } else {
    activeSortColumn.value = null
    sortDirection.value = null
  }
}
</script>
