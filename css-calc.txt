export default {
  name: 'component-district-coefficient',
  props: {
    item: { type: Object, required: true },
    modelValue: { type: [String, Number, Array], default: '' },
    view: { type: String, default: 'district' } // 'district' или 'northern'
  },
  data() {
    return {
      dataField: '',
      dropDown: false,
      isUserInteracted: false,
      isMobile: null,
      originalWindowHeight: null,
    }
  },

 methods: {
  logEvent(event) {
      this.isUserInteracted = true; // Пользователь начал взаимодействовать

      let val = event.target.value.replace(/[^\d.,]/g, '').replace(',', '.');
      const parts = val.split('.');

      // Ограничение до 2 знаков после точки
      if (parts.length === 2 && parts[1].length > 2) {
        val = parts[0] + '.' + parts[1].substring(0, 2);
      }

      const num = parseFloat(val);

      if (!isNaN(num)) {
        if (this.view === 'district') {
          if (num < 1) val = '1';
          else if (num >= 3) val = '3'; // Если введено 3 или больше, фиксируем на 3
        } else {
          if (num < 0.1) val = '0.1';
          else if (num >= 100) val = '100';
        }
      }

      this.dataField = val;
      this.$emit('update:modelValue', this.dataField);
    },
}

watch: {
    modelValue: {
      handler(newVal) {
        let normalizedVal = '';
        if (Array.isArray(newVal) && newVal.length > 0) {
          normalizedVal = String(newVal[0].label || newVal[0].value || '');
        } else {
          normalizedVal = String(newVal || '');
        }

        // Обновляем поле и сбрасываем флаг ТОЛЬКО если значение пришло извне
        // и отличается от текущего (например, Store очистил поле при смене таба).
        // Если пользователь просто печатает, родитель вернет то же значение,
        // условие this.dataField !== normalizedVal будет false, и флаг НЕ сбросится!
        if (this.dataField !== normalizedVal) {
          this.dataField = normalizedVal;
          if (normalizedVal === '') {
            this.isUserInteracted = false;
          }
        }
      },
      immediate: true
    }
  },

data(){
    return {
      //поля средний доход по годам, заполняется в created
      fieldsIncome: {},
      isInitialized: false,
      skipNextWatch: true,

      //поле дополнительные доходы


    }
  },
  methods:{
    syncFieldsFromStore() {
      const fields = {};
      this.array_all_income_period.forEach(tab => {
        fields[tab.modelInput] = tab.value || '';

        if (tab.additionalCoefficient) {
          const baseCoeffKey = tab.additionalCoefficient.modelInput;

          // НОРМАЛИЗАЦИЯ: Преобразуем массив (от старого селекта) в строку для нового инпута
          const distVal = tab.additionalCoefficient.districtCoefficient.value;
          fields[baseCoeffKey + '_district'] = Array.isArray(distVal) ? (distVal[0]?.label || '') : (distVal || '');

          const northVal = tab.additionalCoefficient.northernAllowance.value;
          fields[baseCoeffKey + '_northern'] = Array.isArray(northVal) ? (northVal[0]?.label || '') : (northVal || '');
        }
      });
      this.fieldsIncome = fields;
    }
  },
  computed:{
    data_year() {
      return store.getters.DATAYEAR;
    },

    array_all_income_period(){
      if (this.view === 'man') {
        return store.getters.ARRAYALLINCOMEPERIODMAN
      } else {
        return store.getters.ARRAYALLINCOMEPERIODWIFE
      }

    },

    //этот компьютед не только получает со стор массив полей, но и вычисляет какие инпуты показывать в зависимости от выбранных лет
    //на основе него по циклу формируются поля с доходом
    visibleBlockIncomePeriod() {
      // Ищем индекс первого элемента, у которого isActive === true
      const activeIndex = this.array_all_income_period.findIndex(tab => tab.isActive === true)

      // Если такой элемент не найден — возвращаю пустой массив
      if (activeIndex === -1) return []

      // Беру все элементы от начала до активного включительно
      return this.array_all_income_period.slice(0, activeIndex + 1)
    },

  },
created() {

    this.debouncedIncomeUpdate = debounce((newFields) => {
      if (this.skipNextWatch || !this.isInitialized) return;

      const cleanedFields = {};
      for (const key in newFields) {
        const val = newFields[key];
        // Теперь val для коэффициентов будет просто строкой (например, "1.5"),
        // а не массивом. Умная очистка это корректно обработает.
        if (Array.isArray(val) || (typeof val === 'object' && val !== null)) {
          cleanedFields[key] = val;
        } else {
          const cleanStr = String(val).replace(/\s/g, '');
          cleanedFields[key] = cleanStr === '' ? '' : (isNaN(parseFloat(cleanStr)) ? cleanStr : parseFloat(cleanStr));
        }
      }
      console.log('created')

      store.dispatch('ActionShowLoader', true);
      const actionName = this.view === 'man' ? 'ActionInputIncomeMan' : 'ActionInputIncomeWife';

      store.dispatch(actionName, cleanedFields).finally(() => {
        store.dispatch('ActionShowLoader', false);
      });
    }, 500);
  },

mounted(){
    this.skipNextWatch = true;
    this.$nextTick(() => {
      this.syncFieldsFromStore();
      this.$nextTick(() => {
        this.skipNextWatch = false;
        this.isInitialized = true;
      });
    });


  },
  watch: {
    array_all_income_period: {
      handler() {
        this.skipNextWatch = true;
        this.$nextTick(() => {
          this.syncFieldsFromStore();
          this.$nextTick(() => {
            this.skipNextWatch = false;
            this.isInitialized = true;
          });
        });
      },
      deep: true
    },
    fieldsIncome: {
      handler(newValue) {
        if (this.skipNextWatch || !this.isInitialized) return;
        this.debouncedIncomeUpdate(newValue);
      },
      deep: true,
      immediate: false
    },
  },
