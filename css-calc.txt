import { watch, nextTick, unref } from 'vue';

export function useMaxValue(modelRef, maxValueRef, inputRef = null) {
  watch(modelRef, async (newValue) => {
    if (!newValue) return;

    // unref() позволяет передать сюда И обычное число (60000000), И ref/computed
    const maxVal = unref(maxValueRef);
    if (!maxVal) return;

    const cleanStr = String(newValue).replace(/\s/g, '');
    const num = parseFloat(cleanStr);

    // Если значение корректно, ничего не делаем
    if (isNaN(num) || num <= maxVal) {
      return;
    }

    const maxFormatted = Number(maxVal).toLocaleString('ru-RU');

    // Защита от бесконечного цикла
    if (newValue === maxFormatted) return;

    // Принудительно устанавливаем максимум
    modelRef.value = maxFormatted;

    // Исправление позиции курсора для сложного объекта компонента
    if (inputRef && inputRef.value) {
      await nextTick();
      
      const componentInstance = inputRef.value;
      
      // componentInstance.$el — это корневой DOM-элемент компонента <DsInput>.
      // Мы ищем внутри него настоящий тег <input>.
      const nativeInput = componentInstance.$el 
        ? componentInstance.$el.querySelector('input') 
        : componentInstance;

      if (nativeInput && typeof nativeInput.setSelectionRange === 'function') {
        nativeInput.focus();
        // requestAnimationFrame гарантирует, что маска DsInput уже отрисовала новое значение
        requestAnimationFrame(() => {
          nativeInput.setSelectionRange(maxFormatted.length, maxFormatted.length);
        });
      }
    }
  });
}

setup() {
    const amountInput = ref(null);

    const { dataField } = useDebounceStoreSync(
      'ActionApartmentPrice',
      () => store.getters.APARTMENTPRICE
    );

    // --- Вспомогательная функция (аналог метода, но доступна в setup) ---
    const getMaxAmountForYear = (year, limits) => {
      if (!year || !limits || !Array.isArray(limits)) return 0;

      const numericYear = Number(year);

      const rule = limits.find((item) => {
        const start = Number(item.begin);
        // Если end пустой, null или undefined, считаем его бесконечностью
        const end = item.end === "" || item.end === null || item.end === undefined 
          ? Infinity 
          : Number(item.end);

        return numericYear >= start && numericYear <= end;
      });

      return rule ? Number(rule.maxAmount) : 0;
    };

    // --- Реактивное вычисление текущего лимита ---
    // Мы используем computed, чтобы при смене года в сторе, лимит пересчитался автоматически
    const currentMaxAmount = computed(() => {
      // Обратите внимание: в setup мы обращаемся к computed через store.getters напрямую, 
      // либо можем использовать this, если вынесем эту логику в options API computed.
      // Но самый надежный способ в setup — взять значения из store.getters:
      const year = store.getters.DATAYEAR;
      const limits = store.getters.MAX_AMOUNT_LIMITS || [];
      
      return getMaxAmountForYear(year, limits);
    });

    // --- Подключаем ограничение ---
    // Передаем currentMaxAmount (это ref). В других компонентах тут может быть просто число 60000000.
    useMaxValue(dataField, currentMaxAmount, amountInput);

    return {
      dataField,
      amountInput
