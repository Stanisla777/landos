<script>
import AddClassBody from '../../../../modules/redesign-site/disallow-body-scrolling';
import RemoveClassBody from '../../../../modules/redesign-site/allow-body-scrolling';
import universalForm from '../../../../modules/universal-form/universal-form';
import store from "../../../FamilyMortgageCalculator/store";
export default {
  name: "ModalWeThingNewCourses",
  data() {
    return {
      showModal:false,
      modalData:{},
      parseData: {}
    }
  },
  methods: {
    handleOpen(event) {
      this.modalData = event.detail || {};
      this.parseDataMethod();
      this.$nextTick(() => {
        this.setupEventListeners();
        // this.showModal=true
      })
      this.showModal=true
      document.body.classList.add(
        'modal-opened'
      );
      AddClassBody()
    },
    safeJsonParse(value) {
      // 1. Если это null, undefined или пустая строка
      if (!value) return null;

      // 2. Если это уже объект или массив, возвращаем как есть
      if (typeof value === 'object') {
        return value;
      }

      // 3. Если это строка
      if (typeof value === 'string') {
        const trimmed = value.trim();

        // Если строка пустая
        if (!trimmed) return null;

        // Если строка НЕ начинается с { или [, это точно не JSON-объект/массив
        if (!trimmed.startsWith('{') && !trimmed.startsWith('[')) {
          return value; // Возвращаем обычную строку "bti" как есть
        }

        // Если начинается с { или [, пытаемся распарсить
        try {
          return JSON.parse(trimmed);
        } catch (e) {
          console.warn('Ошибка парсинга JSON, возвращаем исходную строку:', value, e);
          return value;
        }
      }
      return value;
    },

    parseDataMethod(){
      // 1. Парсим базовые данные (контейнер)
      const containerData = this.safeJsonParse(this.modalData.htmlTooltipContainer);

      // 2. Парсим отдельные данные (элемент)
      const itemData = this.safeJsonParse(this.modalData.htmlTooltip);

      //3 Парсим название купса для скрытого input в форме
      const itemCourse = this.safeJsonParse(this.modalData.course);

      // 3. Объединяем: itemData имеет приоритет при совпадении ключей
      this.parseData = { ...containerData, ...itemData, itemCourse };
    },
    closeModal(el){
      const element = el.currentTarget;
      const block = element.closest('#another-thing-speaker');
      if (block) {
        block.setAttribute('style', 'z-index:7;');
      }

      const windowMainContent = this.$refs.refModal.querySelector('.js--modal-main-content')
      const windowSuccessnContent = this.$refs.refModal.querySelector('.js--modal-wr-success')
      setTimeout(()=>{
        if (windowSuccessnContent && windowMainContent && windowMainContent.classList.contains('unactive')) {
          windowSuccessnContent.classList.remove('active')
          windowMainContent.classList.remove('unactive')
        }
      },500)


      this.showModal=false
      document.body.classList.remove(
        'modal-opened'
      );
      RemoveClassBody();
    },
    setupEventListeners() {
      const element = this.$refs.refModal.querySelector('.js--another-thing-close-modal');
      if (element) {
        element.addEventListener('click', this.closeModal)
      }
    },
  },
  mounted(){
    window.addEventListener('open-universal-modal', this.handleOpen);
    universalForm();
  },
  beforeUnmount() {
    window.removeEventListener('open-universal-modal', this.handleOpen);
  }
}
</script>

disallow-body-scrolling.js
export default function AddClassBody() {
  document.body.classList.add('body-modal');
  document.body.classList.add('body-additional-class');
  document.body.setAttribute('style', `top:-${window.scrollY}px;position: fixed;`);
  document.ontouchmove = (e) => {
    e.preventDefault();
  };
}
