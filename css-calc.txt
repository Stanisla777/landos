// disallow-body-scrolling.js
export default function AddClassBody() {
  // 1. Запоминаем текущую позицию скролла
  const scrollY = window.scrollY;

  // 2. Вычисляем и применяем компенсацию ширины скроллбара 
  // (ОБЯЗАТЕЛЬНО делаем это ДО position: fixed)
  const scrollBarWidth = window.innerWidth - document.documentElement.clientWidth;
  document.body.style.paddingRight = `${scrollBarWidth}px`;

  // 3. Применяем фиксацию ПО ОТДЕЛЬНОСТИ (НЕ через setAttribute!)
  // Это предотвращает стирание padding-right и резкий сдвиг контента
  document.body.style.position = 'fixed';
  document.body.style.top = `-${scrollY}px`;
  document.body.style.width = '100%'; // Критически важно для iOS, чтобы контент не сжимался

  document.body.classList.add('body-modal');
  document.body.classList.add('body-additional-class');

  // 4. Блокируем скролл на уровне тач-событий
  document.ontouchmove = (e) => {
    e.preventDefault();
  };
}
