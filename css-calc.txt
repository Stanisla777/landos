visibleBlockIncomePeriodReversed() {
      // slice() создаёт копию, чтобы не мутировать исходный массив
      return this.visibleBlockIncomePeriod.slice().reverse();
    }


---------------

getDisplayYear(index) {
      const totalItems = this.visibleBlockIncomePeriod.length;
      // Формула: последний год в исходном массиве минус текущий index
      return this.data_year + (totalItems - 1 - index);
    }
