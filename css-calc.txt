 on: {
    slideChange: (swiper) => {
      this.activeIndex = swiper.activeIndex;
      // Обновляем сразу для случаев без анимации
      this.updateControls(swiper);
    },
    transitionEnd: (swiper) => {
      // Обновляем после завершения анимации,
      // чтобы getBoundingClientRect вернул финальные значения
      this.updateControls(swiper);
    },
    resize: (swiper) => {
      swiper.update();
      this.updateControls(swiper);
    },
    imagesReady: (swiper) => {
      swiper.update();
      this.updateControls(swiper);
    }
  }
