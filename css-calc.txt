const isColumnSortable = (columnIndex) => {
  // Если глобально включено - можно всё
  if (props.sortingRow) return true
  
  // Иначе ищем разрешение в additionalColumnParameters
  const param = props.additionalColumnParameters?.find(p => p.column === columnIndex)
  return param?.sorting === true
}

const sortedRows = computed(() => {
  const activeCol = activeSortColumn.value
  const dir = sortDirection.value

  // Блокируем сортировку, если:
  // 1. Нет активной колонки или направления
  // 2. ИЛИ для этой конкретной колонки сортировка запрещена
  if (activeCol === null || dir === null || !isColumnSortable(activeCol)) {
    return props.normalizeColumn ? normalizedRows.value : (props.tableData.rows || [])
  }

  const rows = props.normalizeColumn ? normalizedRows.value : (props.tableData.rows || [])
  const direction = dir === 'asc' ? 1 : -1

  return [...rows].sort((a, b) => {
    const valA = extractValue(a.cells[activeCol]?.label || '')
    const valB = extractValue(b.cells[activeCol]?.label || '')

    if (valA.type !== valB.type) {
      return direction * String(valA.value).localeCompare(String(valB.value), 'ru')
    }
    if (valA.type === 'number' || valA.type === 'date') {
      return direction * (valA.value - valB.value)
    }
    return direction * valA.value.localeCompare(valB.value, 'ru')
  })
})
