updateControls(swiper) {
  if (!swiper) return;

  this.isBeginning = Boolean(swiper.isBeginning);

  const lastRealSlideIndex = this.flatSlides.length - 1;

  // Если активный слайд уже последний реальный, то точно конец
  if (swiper.activeIndex >= lastRealSlideIndex) {
    this.isEnd = true;
    return;
  }

  // Проверяем, виден ли последний реальный слайд полностью
  const lastRealSlide = swiper.slides[lastRealSlideIndex];
  
  if (lastRealSlide) {
    const swiperRect = swiper.el.getBoundingClientRect();
    const slideRect = lastRealSlide.getBoundingClientRect();

    // Если правый край последнего слайда <= правому краю контейнера,
    // значит все блоки последнего года видны и листать некуда
    if (slideRect.right <= swiperRect.right + 1) {
      this.isEnd = true;
    } else {
      this.isEnd = false;
    }
  } else {
    this.isEnd = Boolean(swiper.isEnd);
  }
},

nextSlide() {
  if (!this.swiper) return;
  
  // Не листаем, если уже в конце
  if (!this.isEnd) {
    this.swiper.slideNext();
  }
},

prevSlide() {
  if (!this.swiper) return;
  this.swiper.slidePrev();
}
