updateControls(swiper) {
  if (!swiper) return;

  this.isBeginning = Boolean(swiper.isBeginning);

  const lastRealSlideIndex = this.flatSlides.length - 1;
  const activeSlide = this.flatSlides[swiper.activeIndex];
  const lastSlide = this.flatSlides[lastRealSlideIndex];

  if (swiper.activeIndex >= lastRealSlideIndex) {
    this.isEnd = true;
  } else if (activeSlide && lastSlide && activeSlide.year !== lastSlide.year) {
    this.isEnd = false;
  } else {
    const lastRealSlide = swiper.slides[lastRealSlideIndex];

    if (lastRealSlide) {
      const swiperRect = swiper.el.getBoundingClientRect();
      const slideRect = lastRealSlide.getBoundingClientRect();

      this.isEnd = slideRect.right <= swiperRect.right + 1;
    } else {
      this.isEnd = false;
    }
  }

  // Блокируем свайп мышкой и пальцем
  swiper.allowSlideNext = !this.isEnd;
  swiper.allowSlidePrev = !this.isBeginning;
}
