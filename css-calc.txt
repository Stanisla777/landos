// Подключаем новый миксин капчи в этот миксин
mixins: [captchaMixin], 

methods: {
  sendingResult(flag) {
    // Убираем старый вызов this.captchaInit(flag)
    
    // Вызываем метод из нового миксина и передаем туда функцию, 
    // которая знает про flag благодаря замыканию
    this.runCaptcha((token) => {
      if (flag === 'result') {
        this.sendingResultToApi(token);
      } else if (flag === 'mail') {
        this.sendingMailToApi(token);
      }
    });
  },
