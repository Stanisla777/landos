&.hidden{
    max-height: 0;
    overflow: hidden;
    opacity: 0;
    display: block;
    padding-top: 0;
    padding-bottom: 0;
    border-bottom-color: transparent; // Анимируем только color, НЕ width
    
    transition: 
      max-height 0.5s ease,
      opacity 0.5s ease,
      padding-bottom 0.5s ease,
      border-bottom-color 0.5s ease;
  }

-----------------
btn.addEventListener('click', () => {
  const hiddenItems = parent.querySelectorAll('.js-accordion-parent.hidden');

  // 1. Измеряем все высоты заранее
  const heights = Array.from(hiddenItems).map(item => {
    const original = item.style.maxHeight;
    item.style.maxHeight = 'none';
    const height = item.scrollHeight;
    item.style.maxHeight = original;
    return height;
  });

  // 2. Анимируем каждый элемент
  hiddenItems.forEach((item, index) => {
    const targetHeight = heights[index];
    
    // Начальное состояние через inline
    item.style.maxHeight = '0';
    item.style.opacity = '0';
    item.style.paddingBottom = '0';
    
    // Убираем класс
    item.classList.remove('hidden');
    
    // Запускаем анимацию
    requestAnimationFrame(() => {
      item.style.maxHeight = `${targetHeight}px`;
      item.style.opacity = '1';
      item.style.paddingBottom = ''; // CSS вернёт 24px/16px
    });

    // Очистка после анимации
    item.addEventListener('transitionend', function handler(e) {
      if (e.propertyName !== 'max-height') return;
      item.style.maxHeight = '';
      item.style.opacity = '';
      item.style.paddingBottom = '';
      item.removeEventListener('transitionend', handler);
    });
  });

  btn.classList.add('btn-accordion-hidden');
});
