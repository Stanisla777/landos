methods: {
  setupCaptchaListeners() {
      // Использую хелпер из миксина, чтобы подписаться на закрытие окна
      this.subscribeCaptchaEvent('challenge-hidden', () => {
        console.log('Пользователь закрыл окно капчи');
        store.dispatch('ActionStandingButtonFalse');
        store.dispatch('ActionButtonActive', true);
      });

      // При желании можно добавить и другие события, если нужно
      this.subscribeCaptchaEvent('token-expired', () => {
        console.warn('Токен капчи истёк');
        store.dispatch('ActionStandingButtonFalse');
        store.dispatch('ActionButtonActive', true);
      });
    },
  },
  watch: {
    
    captcha_id(newId) {
      if (newId !== null) {
        this.setupCaptchaListeners();
      }
    }

// Подписывается на события капчи

export function subscribeCaptcha(container, eventName, handler) {
  const state = widgetStates.get(container);

  if (state && window.smartCaptcha) {
    try {
      window.smartCaptcha.subscribe(state.widgetId, eventName, handler);
    } catch (e) {}
  }
}
