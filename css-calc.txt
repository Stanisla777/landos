<template>
  <div class="history__wrapper-slider-arrow-subtitle">
    <h2 class="h2_subtitle">
      Развитие ДОМ.РФ — это последовательность решений, которые сформировали текущий масштаб и роль компании
      на рынке
    </h2>
  </div>

  <div class="history__slider">

    <section class="history-timeline">
      <template v-if="flatSlides.length">
        <div v-if="currentYear" class="history-timeline__header">
          <div class="history-timeline__year">
            {{ currentYear }}
          </div>

          <div class="history-timeline__track">
            <span class="history-timeline__dot"></span>
            <span class="history-timeline__line"></span>
          </div>
        </div>

        <div
          ref="swiperContainer"
          class="swiper-container history-timeline__swiper"
        >
          <div class="swiper-wrapper">
            <div
              v-for="(slide, index) in flatSlides"
              :key="`${slide.year}-${index}`"
              class="swiper-slide history-timeline__slide"
              :class="{
              'history-timeline__slide--year-first': slide.isFirstInYear,
              'history-timeline__slide--year-last': slide.isLastInYear
            }"
            >
              <article class="history-card">
                <div
                  v-if="slide.picture"
                  class="history-card__image"
                >
                  <img
                    :src="slide.picture"
                    :alt="slide.title || ''"
                    loading="lazy"
                  >
                </div>

                <h3 class="history-card__title">
                  {{ slide.title }}
                </h3>

                <div
                  class="history-card__text"
                  v-html="slide.description"
                ></div>
              </article>
            </div>
          </div>
        </div>

        <div
          v-if="flatSlides.length > 1"
          class="history-timeline__controls"
        >
          <button
            type="button"
            class="history-timeline__button"
            :disabled="isBeginning"
            @click="prevSlide"
          >
            Назад
          </button>

          <button
            type="button"
            class="history-timeline__button"
            :disabled="isEnd"
            @click="nextSlide"
          >
            Вперёд
          </button>
        </div>
      </template>

      <p
        v-else
        class="history-timeline__empty"
      >
        История пока не передана.
      </p>
    </section>




  </div>


</template>

<script>
import Swiper, { Navigation, Pagination } from 'swiper';
Swiper.use([Navigation, Pagination]);
let mySwiper;

export default {
  name: "HistoryCompany",
  data() {
    return {
      swiper: null,
      activeIndex: 0,
      isBeginning: true,
      isEnd: false,
      history: [
        {
          title: '1997',
          items: [
            {
              title: 'создание Консультационного центра ДОМ.РФ',
              description: '<p>Ежегодный просветительский проект по финансовой грамотности в жилищной сфере</p>',
              picture: '/upload/file.png'
            }
          ]
        },
        {
          title: '2008',
          items: [
            {
              title: 'создание Консультационного центра ДОМ.РФ',
              description: '<p>Ежегодный просветительский проект по финансовой грамотности в жилищной сфере</p>',
              picture: '/upload/file.png'
            },
            {
              title: 'создание Консультационного центра ДОМ.РФ',
              description: '<p>Ежегодный просветительский проект по финансовой грамотности в жилищной сфере</p>',
              picture: '/upload/file.png'
            }
          ]
        }
      ]
    };
  },
  //
  computed: {
    flatSlides() {
      if (!Array.isArray(this.history)) {
        return [];
      }

      return this.history.reduce((acc, year) => {
        const yearTitle = year && year.title !== undefined ? String(year.title) : '';
        const items = Array.isArray(year.items) ? year.items : [];

        items.forEach((item, index) => {
          acc.push({
            year: yearTitle,
            title: item.title || '',
            description: item.description || '',
            picture: item.picture || '',
            isFirstInYear: index === 0,
            isLastInYear: index === items.length - 1
          });
        });

        return acc;
      }, []);
    },

    currentYear() {
      const activeSlide = this.flatSlides[this.activeIndex];

      return activeSlide ? activeSlide.year : '';
    }
  },

  watch: {
    history: {
      deep: true,
      handler() {
        this.activeIndex = 0;

        this.$nextTick(() => {
          this.initSwiper();
        });
      }
    }
  },

  mounted() {
    this.$nextTick(() => {
      this.initSwiper();
    });
  },

  beforeUnmount() {
    this.destroySwiper();
  },

  methods: {
    initSwiper() {
      this.destroySwiper();

      if (!this.$refs.swiperContainer || this.flatSlides.length === 0) {
        return;
      }

      this.swiper = new Swiper(this.$refs.swiperContainer, {
        slidesPerView: 'auto',
        spaceBetween: 16,
        slidesPerGroup: 1,
        watchOverflow: true,
        observer: true,
        observeParents: true,
        grabCursor: true,
        keyboard: {
          enabled: true
        },
        on: {
          slideChange: (swiper) => {
            this.activeIndex = swiper.activeIndex;
            this.updateControls(swiper);
          },
          resize: (swiper) => {
            swiper.update();
            this.updateControls(swiper);
          },
          imagesReady: (swiper) => {
            swiper.update();
            this.updateControls(swiper);
          }
        }
      });

      this.activeIndex = this.swiper.activeIndex || 0;
      this.updateControls(this.swiper);
    },

    destroySwiper() {
      if (this.swiper && typeof this.swiper.destroy === 'function') {
        this.swiper.destroy(true, true);
      }

      this.swiper = null;
    },

    updateControls(swiper) {
      if (!swiper) {
        return;
      }

      this.isBeginning = Boolean(swiper.isBeginning);
      this.isEnd = Boolean(swiper.isEnd);
    },

    nextSlide() {
      if (!this.swiper) {
        return;
      }

      this.swiper.slideNext();
    },

    prevSlide() {
      if (!this.swiper) {
        return;
      }

      this.swiper.slidePrev();
    }
  }
}
</script>
