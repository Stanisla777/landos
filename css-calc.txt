isImagePath(icon) {
    if (!icon || typeof icon !== 'string') return false;
    
    const trimmed = icon.trim();
    
    // Если это inline SVG (начинается с тега <svg)
    if (trimmed.toLowerCase().startsWith('<svg')) {
      return false;
    }
    
    // Если содержит расширения изображений (с учётом query-параметров)
    const imageExtensions = ['.svg', '.jpg', '.jpeg', '.png', '.webp', '.gif', '.avif'];
    const lowerIcon = trimmed.toLowerCase();
    
    if (imageExtensions.some(ext => lowerIcon.includes(ext))) {
      return true;
    }
    
    // Если это путь к ресурсу (начинается с /, ./, ../ или http)
    if (
      trimmed.startsWith('/') ||
      trimmed.startsWith('./') ||
      trimmed.startsWith('../') ||
      trimmed.startsWith('http://') ||
      trimmed.startsWith('https://')
    ) {
      return true;
    }
    
    // По умолчанию считаем inline SVG
    return false;
  }
