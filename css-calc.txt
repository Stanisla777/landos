filteredVariants() {
      // 1. Безопасно получаем исходный массив
      const variants = this.item?.additionalCoefficient?.districtCoefficient?.variant || [];
      
      // 2. Приводим текущее значение инпута к строке и убираем пробелы
      // (toLowerCase() здесь не нужен, так как у нас только цифры и точки)
      const searchQuery = String(this.dataField).trim();

      // 3. Если поле пустое или содержит только точку, показываем весь список
      if (!searchQuery || searchQuery === '.') {
        return variants;
      }

      // 4. ⭐ ФИЛЬТРАЦИЯ ПО НАЧАЛУ СТРОКИ
      return variants.filter(item2 => {
        return String(item2.label).startsWith(searchQuery);
      });
    }
