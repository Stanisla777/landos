<script setup>
import { ref } from 'vue';
import DsChip from './DsChip.vue';

const playgroundProps = ref({
  modelValue: false,
  label: 'Chips',
  showClose: true,
  disabled: false,
  darkMode: false,
  iconLeft: false,
});

const rooms = ['Студия', '1-комнатная', '2-комнатная', '3-комнатная', '4+'];
const selectedRooms = ref(['1-комнатная']);
const toggleRoom = (room, checked) => {
  selectedRooms.value = checked
    ? [...selectedRooms.value, room]
    : selectedRooms.value.filter((item) => item !== room);
};

const states = [
  { title: 'Default', active: false, disabled: false },
  { title: 'Active', active: true, disabled: false },

            :key="room"
            :label="room"
            :model-value="selectedRooms.includes(room)"
            @update:model-value="toggleRoom(room, $event)"
          />
        </div>
        <p class="p2-medium chip-story__info">Выбрано: {{ selectedRooms.join(', ') || '—' }}</p>
