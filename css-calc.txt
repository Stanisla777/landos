<div ref="captchaWrapper"></div>

export default {
  data() {
    return {
      captchaId: null,
      // Генерируем уникальный ID
      containerId: `yandex-captcha-${Math.random().toString(36).substr(2, 9)}`,
      isMobile: null,
      flag: 0,
    };
  },

  mounted() {
    // 1. Создаем абсолютно чистый DOM-элемент вручную
    const container = document.createElement('div');
    container.id = this.containerId;
    // ВАЖНО: Не добавляйте сюда классы вроде 'smart-captcha'! 
    // Это триггерит авто-инициализацию Яндекса и вызывает двойной рендер.
    container.className = 'captcha-element'; 
    
    // 2. Вставляем его в обертку, которой управляет Vue
    if (this.$refs.captchaWrapper) {
      this.$refs.captchaWrapper.appendChild(container);
    }

    // 3. Даем браузеру один кадр на отрисовку этого чистого узла, 
    // прежде чем отдавать его React-движку Яндекса. 
    // requestAnimationFrame надежнее setTimeout для таких задач.
    requestAnimationFrame(() => {
      this.initCaptcha();
    });
  },

  beforeUnmount() { // Используйте beforeDestroy, если у вас Vue 2
    if (this.captchaId !== null && window.smartCaptcha) {
      try {
        window.smartCaptcha.reset(this.captchaId);
      } catch (e) {
        // Игнорируем ошибки при уничтожении
      }
      this.captchaId = null;
    }
  },

  methods: {
    initCaptcha() {
      if (this.captchaId !== null) return;
      if (!window.smartCaptcha) {
        console.warn('[SmartCaptcha] Скрипт еще не загружен');
        return;
      }

      const container = document.getElementById(this.containerId);
      if (!container) return;

      // "Ядерная" очистка на всякий случай (защита от авто-инициализации)
      while (container.firstChild) {
        container.removeChild(container.firstChild);
      }

      try {
        this.captchaId = window.smartCaptcha.render(this.containerId, {
          sitekey: conf?.smartcaptcha_key,
          invisible: true,
          callback: (token) => {
            this.sendingResultToApi(token);
          },
        });
        console.log('[SmartCaptcha] Успешно инициализирована, ID:', this.captchaId);
      } catch (e) {
        console.error('[SmartCaptcha] Ошибка render:', e);
      }
    },

    sendingResult() {
      if (!this.btn_active) return;

      store.dispatch('ActionStandingButton');
      store.dispatch('ActionButtonActive', false);

      if (this.captchaId !== null && window.smartCaptcha) {
        window.smartCaptcha.execute(this.captchaId);
      } else {
        // Fallback: пробуем инициализировать прямо сейчас
        this.initCaptcha();
        setTimeout(() => {
          if (this.captchaId !== null && window.smartCaptcha) {
            window.smartCaptcha.execute(this.captchaId);
          }
        }, 100);
      }
    },

    sendingResultToApi(token) {
      this.$refs.selectListWrapper?.classList.remove('open');
      this.closeBlock();
      store.dispatch('ActionGetResult', [this.oktmo, token]);
      
      // Сброс после отправки
      if (this.captchaId !== null && window.smartCaptcha) {
        window.smartCaptcha.reset(this.captchaId);
      }
    }
  }
}
