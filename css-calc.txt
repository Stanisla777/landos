export default {
  data() {
    return {
      captchaWidgetId: null,    // Только одна переменная для ID
      captchaContainer: null,   // Только одна переменная для DOM-элемента
      isSending: false,         // Защита от двойного клика
    };
  },

  beforeUnmount() { // beforeDestroy для Vue 2
    this.destroyCaptcha();
  },

  methods: {
    // 1. Обработчик клика
    sendingResult() {
      if (!this.btn_active || this.isSending) return;
      
      this.isSending = true;
      store.dispatch('ActionStandingButton');
      store.dispatch('ActionButtonActive', false);

      if (!this.captchaWidgetId) {
        // Первый клик: создаем виджет
        this.initCaptcha();
        // Даем React Яндекса 150мс на инициализацию перед вызовом execute
        setTimeout(() => this.executeCaptcha(), 150);
      } else {
        // Повторный клик: виджет уже есть, просто запускаем
        this.executeCaptcha();
      }
    },

    // 2. Создание виджета
    initCaptcha() {
      if (!window.smartCaptcha) {
        console.error('Скрипт капчи не загружен');
        this.isSending = false;
        return;
      }

      const sitekey = conf?.smartcaptcha_key;
      if (!sitekey || typeof sitekey !== 'string') {
        console.error('КРИТИЧЕСКАЯ ОШИБКА: smartcaptcha_key отсутствует!');
        this.isSending = false;
        return;
      }

      // Создаем элемент вне Vue
      const el = document.createElement('div');
      el.id = `yandex-captcha-ext-${Date.now()}`;
      el.style.position = 'absolute';
      el.style.left = '-9999px';
      el.style.opacity = '0';
      
      document.body.appendChild(el);
      this.captchaContainer = el;

      try {
        this.captchaWidgetId = window.smartCaptcha.render(el.id, {
          sitekey: sitekey,
          invisible: true, // ⚠️ ОБЯЗАТЕЛЬНО true, так как мы используем .execute()
          callback: (token) => {
            this.submitToApi(token);
          },
        });
      } catch (e) {
        console.error('Ошибка render:', e);
        this.destroyCaptcha();
        this.isSending = false;
      }
    },

    // 3. Запуск проверки
    executeCaptcha() {
      if (this.captchaWidgetId !== null && this.captchaWidgetId !== undefined && window.smartCaptcha) {
        window.smartCaptcha.execute(this.captchaWidgetId);
      } else {
        console.warn('Не удалось выполнить капчу: widgetId невалиден', this.captchaWidgetId);
        this.isSending = false;
      }
    },

    // 4. Обработка токена
    submitToApi(token) {
      this.$refs.selectListWrapper?.classList.remove('open');
      this.closeBlock();

      store.dispatch('ActionGetResult', [this.oktmo, token])
        .then(() => {
          console.log('Успешная отправка');
        })
        .catch((err) => {
          console.error('Ошибка отправки:', err);
        })
        .finally(() => {
          // ⚠️ ВАЖНО: Мы делаем ТОЛЬКО reset. 
          // Мы НЕ удаляем элемент из DOM (не делаем removeChild) здесь!
          // Удаление DOM-элемента во время работы callback'а вызывает ошибку React #418.
          if (this.captchaWidgetId !== null && window.smartCaptcha) {
            try {
              window.smartCaptcha.reset(this.captchaWidgetId);
            } catch (e) {}
          }
          
          // Обнуляем ID, чтобы следующий клик создал свежий виджет
          this.captchaWidgetId = null; 
          this.isSending = false;
          // store.dispatch('ActionButtonActive', true); // Раскомментируйте, если нужно
        });
    },

    // 5. Полная очистка (вызывается ТОЛЬКО при уничтожении компонента Vue)
    destroyCaptcha() {
      if (this.captchaWidgetId !== null && window.smartCaptcha) {
        try {
          window.smartCaptcha.reset(this.captchaWidgetId);
        } catch (e) {}
      }

      // Вот здесь мы безопасно удаляем DOM-элемент, так как компонент Vue умирает
      if (this.captchaContainer && this.captchaContainer.parentNode) {
        this.captchaContainer.parentNode.removeChild(this.captchaContainer);
      }

      this.captchaWidgetId = null;
      this.captchaContainer = null;
    }
  }
}
