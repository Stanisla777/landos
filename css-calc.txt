function CommentsFormValidationAnswer() {
  const form = document.querySelectorAll('.js--comments__item-form-answer:not(.js--comments__item-form-answer-init)')
  if (form) {
    form.forEach((item) => {
item.addEventListener('submit', (event) => {
        event.preventDefault();
if (!errors.name && !errors.check && !errors.text) {
          // Создаём уникальный ID для каждой формы ответа

          const containerId = item.dataset.captchaContainerId;
          if (!containerId) {
            console.error('Контейнер капчи не найден');
            return;
          }

          const container = item.querySelector(`#${containerId}`);
          if (!container) {
            console.error('Элемент контейнера капчи не найден');
            return;
          }

          // Запускаем капчу через наш модуль
          CaptchaCore.runCaptcha(container, (token) => {
            let formData = new FormData(item);
            formData.delete('smart-token');

            // Добавляем токен (реальный или fallback)
            if (token !== null && token !== undefined) {
              formData.append('smart-token', token);
            }

            sendReply(form, formData, errors);
          }, {
            sitekey: conf?.smartcaptcha_key,

            // Подписываемся на события капчи
            onWidgetCreated: (widgetId) => {
              // Пользователь закрыл окно капчи
              CaptchaCore.subscribeCaptcha(container, 'challenge-hidden', () => {
                if (item.querySelector('.form-comments__button')) {
                  item.querySelector('.form-comments__button').classList.remove('submitting')
                  item.querySelector('.form-comments__button').removeAttribute('disabled','disabled')
                }
                CaptchaCore.resetCaptcha(container, false);
              });

              // Токен истёк
              CaptchaCore.subscribeCaptcha(container, 'token-expired', () => {
                console.warn('Токен капчи истёк');
                if (item.querySelector('.form-comments__button')) {
                  item.querySelector('.form-comments__button').classList.remove('submitting')
                  item.querySelector('.form-comments__button').removeAttribute('disabled','disabled')
                }
                CaptchaCore.resetCaptcha(container, false);
              });
            }
          });
        }
      })
    })
  }
}

function sendReply(item, formData, errors) {
  let request = BX.ajax.runComponentAction('local:comments.form', 'save', {
    mode: 'class',
    data: formData
  });

  request.then(function(responce){
    commentsPopupShow(responce.data.title, responce.data.message, responce.status);
    item.reset();

    //Сбрасываю состояние кнопки
    const button = item.querySelector('.form-comments__button');
    if (button) {
      button.classList.add('form-comments__button_disabled');
      button.classList.remove('submitting');
      button.setAttribute('disabled', 'disabled');
    }

    // Удаляю форму после успешной отправки
    removeReplyForm(item);

    // 5. Сбрасываем локальные ошибки, хоть форму и удалил уже
    errors = {
      name: false,
      text: false,
      check: false
    };
    updateCountComments(responce);
    updateCommentsList(responce.data.id, formData.get('parent_id'));

  }, function (responce) {
    // При ошибке форму не удаляю
    commentsPopupShow('Ошибка', responce.errors, responce.status);

    // Разблокирую кнопку, если она была в состоянии submitting
    const button = item.querySelector('.form-comments__button');
    if (button) {
      button.classList.remove('submitting');
    }
    const container = form.querySelector('.captcha-element');
    if (container) {
      CaptchaCore.resetCaptcha(container, false);
    }
  });
}

function removeReplyForm(form) {
  if (!form) return;


  //сбрасываю и удаляю контейнер каптчи
  const containerId = form.dataset.captchaContainerId;
  if (containerId) {
    const container = form.querySelector(`#${containerId}`);
    if (container) {
      CaptchaCore.resetCaptcha(container, true); // true = удалить DOM-элемент
    }
  }

  // Добавить в стили
  if (form.style.transition) {
    form.style.transition = 'opacity 0.2s ease';
    form.style.opacity = '0';
    setTimeout(() => {
      if (form.parentNode) {
        form.remove();
      }
    }, 200);
  } else {
    // Мгновенное удаление, если анимация не нужна
    form.remove();
  }
}
