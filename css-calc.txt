// 🔥 НОВАЯ ФУНКЦИЯ для запуска капчи
function runCaptchaForAction(actionCallback) {
    CaptchaCore.runCaptcha(captchaContainer, (token) => {
        actionCallback(token);
    }, {
        sitekey: conf?.smartcaptcha_key,
        
        onWidgetCreated: (widgetId) => {
            // Пользователь закрыл окно капчи
            CaptchaCore.subscribeCaptcha(captchaContainer, 'challenge-hidden', () => {
                console.log('Пользователь закрыл окно капчи');
                // Разблокируем все кнопки
                $check_status_button_start_request.attr('disabled', false);
                $check_status_repeat_smscode.attr('disabled', false);
                $check_status_button_check_request.attr('disabled', false);
                CaptchaCore.resetCaptcha(captchaContainer, false);
            });

            // Токен истёк
            CaptchaCore.subscribeCaptcha(captchaContainer, 'token-expired', () => {
                console.warn('Токен капчи истёк');
                $check_status_button_start_request.attr('disabled', false);
                $check_status_repeat_smscode.attr('disabled', false);
                $check_status_button_check_request.attr('disabled', false);
                CaptchaCore.resetCaptcha(captchaContainer, false);
            });
        }
    });
}

function submitFormStartRequest(token) {
    let order_number = $check_status_input.val().trim();
    if (!!order_number) {
        $check_status_button_start_request.attr('disabled', true);
        $check_status_input.removeClass('error');
        $check_status_hint_error.html('').hide();
        
        const requestData = { 'number': order_number };
        if (token !== null && token !== undefined) {
            requestData['smart-token'] = token;
        }
        
        $.ajax({
            type: 'POST',
            url: url_happy_family_rest + 'start/',
            dataType: 'json',
            data: requestData,
            headers: {
                'X-Bitrix-Csrf-Token': window.BX.bitrix_sessid(),
            },
            success: function (response) {
                if (!!response?.result) {
                    if (!!response.result?.phone) {
                        $check_status_phone.text(response.result.phone);
                    }
                    resetTimer();
                    showStep(2);
                } else {
                    $check_status_input.addClass('error');
                    if (!!response?.description) {
                        $check_status_hint_error.html(response.description).show();
                    } else {
                        console.log('Success response has not result', response);
                    }
                    if (!!response?.debug) {
                        console.error(response.debug);
                    }
                    $check_status_button_start_request.attr('disabled', false);
                }
                // 🔥 Сбрасываем капчу
                CaptchaCore.resetCaptcha(captchaContainer, false);
            },
            error: function (error) {
                let errorMessage = error.responseJSON?.description;
                if (!!errorMessage) {
                    $check_status_input.addClass('error');
                    $check_status_hint_error.html(errorMessage).show();
                } else {
                    console.log('Error response has not description', error);
                }
                $check_status_button_start_request.attr('disabled', false);
                // 🔥 Сбрасываем капчу
                CaptchaCore.resetCaptcha(captchaContainer, false);
            }
        });
    }
}

function submitFormCheckRequest(token) {
    let pin = $check_status_pin.val().trim();
    if (!!pin) {
        $check_status_button_check_request.attr('disabled', true);
        $check_status_hint_error.html('').hide();
        
        const requestData = { 'code': pin };
        if (token !== null && token !== undefined) {
            requestData['smart-token'] = token;
        }
        
        $.ajax({
            type: 'POST',
            url: url_happy_family_rest + 'check/',
            dataType: 'json',
            data: requestData,
            headers: {
                'X-Bitrix-Csrf-Token': window.BX.bitrix_sessid(),
            },
            success: function (response) {
                if (!!response?.result) {
                    if (!!response.result?.data) {
                        if (!!response.result.data?.text)
                            $check_status_response_text.html(response.result.data.text);
                        if (!!response.result.data?.hint)
                            $check_status_response_hint.html(response.result.data.hint);
                        if (!!response.result.data?.date) {
                            $check_status_response_date.find('span').text(response.result.data.date);
                            $check_status_response_date.show();
                        }
                        if (!!response.result.data?.time) {
                            $check_status_response_time.find('span').text(response.result.data.time);
                            $check_status_response_time.show();
                        }
                        if (!!response.result.data?.certificate_for_status_paid) {
                            certificate_data = response.result.data.certificate_for_status_paid;
                            $information_mail_block.show();
                        }

                    } else {
                        $check_status_hint_error.html('Ошибка сервера при получении данных.').show();
                    }
                    showStep(3);

                } else {
                    $check_status_input.addClass('error');
                    if (!!response?.description) {
                        $check_status_hint_error.html(response.description).show();
                    } else {
                        console.log('Success response has not result', response);
                    }
                    if (!!response?.debug) {
                        console.error(response.debug);
                    }
                    $check_status_button_check_request.attr('disabled', false);
                }
                // 🔥 Сбрасываем капчу
                CaptchaCore.resetCaptcha(captchaContainer, false);
            },
            error: function (error) {
                let errorMessage = error.responseJSON?.description;
                let errorCode = error.responseJSON?.code;

                if (!!errorMessage) {
                    $check_status_input.addClass('error');
                    $check_status_hint_error.html(errorMessage).show();
                } else {
                    console.log('Error response has not description', error);
                }
                if (errorCode !== 406) {
                    showStep(3);
                }
                $check_status_button_check_request.attr('disabled', false);
                // 🔥 Сбрасываем капчу
                CaptchaCore.resetCaptcha(captchaContainer, false);
            }
        });
    }
}
